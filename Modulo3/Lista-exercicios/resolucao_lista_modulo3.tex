  \begin{block}{Enunciado}
  Considere o amostrador ideal $g_s(t)=g(t)\,\delta_{T_s}(t)$, com $\delta_{T_s}(t)=\sum_{n}\delta(t-nT_s)$ e $g(t)$ limitado a $|f|<W$.
  \end{block}
  \vspace{0.1cm}
  \textbf{Pede-se:}
  \begin{enumerate}[(a)]
    \item Expandindo o trem de impulsos em série de Fourier exponencial, demonstre que \[ G_s(f)=f_s\sum_{k=-\infty}^{\infty}G(f-kf_s),\qquad f_s=1/T_s . \]
    \item A partir do espectro acima e de um filtro passa-baixas ideal de corte $W$, demonstre a fórmula de interpolação \[ g(t)=\sum_{n=-\infty}^{\infty} g(nT_s)\, \sinc\!\left(\frac{t-nT_s}{T_s}\right). \]
    \item Prove que a curva interpolada passa exatamente pelas amostras, isto é, avalie a soma em $t=mT_s$ e mostre que resulta $g(mT_s)$.
  \end{enumerate}
\end{frame}

\begin{frame}{Q1 (a) --- espectro do sinal amostrado}
  \footnotesize
  \begin{columns}[c]
    \column{0.30\textwidth}\centering
    \begin{tikzpicture}
      \node[blk] (samp) {Amostrador\\$\times\,\delta_{T_s}$};
      \draw[arw] ($(samp.north)+(0,0.5)$) node[above,font=\scriptsize]{$g(t)$} -- (samp);
      \draw[arw] (samp) -- ($(samp.south)+(0,-0.5)$) node[below,font=\scriptsize]{$g_s(t)$};
      \node[hl, fit=(samp)] {};
    \end{tikzpicture}

    \column{0.70\textwidth}
    \onslide<1->{\passo{1.} O pente é periódico de período $T_s$, logo admite série
      de Fourier; o coeficiente sai pela propriedade de filtragem:
      \[ \delta_{T_s}(t)=\sum_k c_k e^{j2\pi kf_st},\quad
         c_k=\tfrac1{T_s}\!\int_{-T_s/2}^{T_s/2}\!\delta(t)e^{-j2\pi kf_st}dt=f_s. \]}
    \onslide<2->{\passo{2.} Substituindo no sinal amostrado:
      \[ g_s(t)=g(t)\,\delta_{T_s}(t)=f_s\sum_k g(t)\,e^{j2\pi kf_st}. \]}
    \onslide<3->{\passo{3.} Cada $e^{j2\pi kf_st}$ desloca o espectro de $kf_s$:
      \[ \boxed{\,G_s(f)=f_s\sum_k G(f-kf_s)\,}. \]}
  \end{columns}
\end{frame}

    \onslide<3->{\passo{3.} A reconstrução é $g=g_s*h$:

% ---- resolução item (c) ----
\begin{frame}{Q1 (c) --- a curva passa pelas amostras}
  \footnotesize
  \onslide<1->{\passo{1.} Avaliamos a soma de interpolação no instante de amostra
    $t=mT_s$:
    \[ g(mT_s)=\sum_n g(nT_s)\,\sinc\!\Big(\tfrac{mT_s-nT_s}{T_s}\Big)
              =\sum_n g(nT_s)\,\sinc(m-n). \]}
  \onslide<2->{\passo{2.} O $\sinc$ vale $1$ em zero e $0$ nos demais inteiros:
    $\sinc(m-n)=\delta[m-n]$.}
  \onslide<3->{\passo{3.} Só o termo $n=m$ sobrevive:
    \[ \boxed{\,g(mT_s)=g(mT_s)\,} \]
    --- a curva interpolada passa exatamente pelas amostras.}
\end{frame}

\begin{frame}{Síntese}
  \small
  \begin{itemize}
    \item \textbf{Amostragem e quantização} fixam a taxa de bits
          $R_b=n f_s$ e o $\SQNR\approx6{,}02n+1{,}76$ dB.
    \item \textbf{Companding, DM e DPCM} reduzem bits explorando a estatística
          e a correlação do sinal.
    \item \textbf{Formatação de pulso e $M$-PAM} ajustam banda e potência:
          $W=(1+\alpha)R_s/2$, $E_s=(M^2-1)d^2/3$.
    \item O \textbf{projeto integrado} mostra o acoplamento entre qualidade,
          banda ocupada e potência exigida.
  \end{itemize}
\end{frame}
