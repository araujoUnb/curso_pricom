\subsection{Quantizador Uniforme}
% ------------------------------------------------------------------

\begin{frame}{Onde Estamos na Cadeia?}
  \drawpipeline{quant}
  \vspace{0.1cm}
  {\small Nesta seção: \textbf{Quantizador} — o sinal já está discreto no
  tempo; agora discretizamos a amplitude e analisamos o ruído introduzido.}
\end{frame}

\begin{frame}{Por Que Quantizar?}
  Após a amostragem, temos $g[n] = g(nT_s)$:
  discreto no tempo, mas com \textbf{amplitude ainda contínua}.

  \vspace{0.15cm}
  \begin{columns}[T]
    \column{0.50\textwidth}
    \textbf{O problema:}
    \begin{itemize}
      \item Um processador só armazena valores \alert{discretos}
      \item Com $n$ bits há apenas $2^n$ valores representáveis
      \item Amplitudes contínuas \textbf{precisam ser arredondadas}
    \end{itemize}

    \column{0.47\textwidth}
    \textbf{Analogia cotidiana:}
    \begin{itemize}
      \item Régua com marcas a cada 1 mm
      \item $23{,}731$ mm $\;\to\;$ $23{,}7$ mm
      \item O erro é inevitável — só reduzível
    \end{itemize}
  \end{columns}

  \vspace{0.15cm}
  \textbf{A questão central:}
  \begin{enumerate}
    \item Quanto ruído a quantização introduz?
    \item Como esse ruído depende do número de bits?
  \end{enumerate}
\end{frame}

% ------------------------------------------------------------------

\begin{frame}{O Quantizador Uniforme — Passo 1: A Faixa de Amplitude}
  O sinal amostrado $g[n]$ assume valores dentro de uma faixa finita
  $[V_{\min}, V_{\max}]$. A largura total é a tensão \textbf{pico a pico}:
  \[
    V_{pp} = V_{\max} - V_{\min}.
  \]

  \vspace{0.4cm}
  \begin{center}
  \begin{tikzpicture}[>=stealth, every node/.style={font=\scriptsize}]
    \draw[->,gray!55] (-0.5,0) -- (7.4,0) node[right,black]{$g$};
    \fill[UNB_GOLD!16] (0,0) rectangle (6.4,0.7);
    \draw[UNB_BLUE,very thick] (0,-0.12) -- (0,0.95);
    \draw[UNB_BLUE,very thick] (6.4,-0.12) -- (6.4,0.95);
    \node[below,black] at (0,-0.14) {$V_{\min}$};
    \node[below,black] at (6.4,-0.14) {$V_{\max}$};
    \draw[<->,thick,black] (0,1.25) -- (6.4,1.25)
        node[midway,above]{$V_{pp} = V_{\max}-V_{\min}$};
  \end{tikzpicture}
  \end{center}

  \vspace{0.3cm}
  \textit{Toda a informação de amplitude está contida nessa faixa; é ela
  que vamos dividir em níveis discretos.}
\end{frame}

% ------------------------------------------------------------------

\begin{frame}{O Quantizador Uniforme — Passo 2: Particionar em $L$ Intervalos}
  Dividimos a faixa $V_{pp}$ em \textbf{$L$ intervalos de mesma largura}.
  Como os $L$ degraus, somados, cobrem toda a faixa:
  \[
    L \cdot \Delta = V_{pp}
    \quad\Longrightarrow\quad
    \boxed{\;\Delta = \dfrac{V_{pp}}{L}\;}
  \]

  \vspace{0.35cm}
  \begin{center}
  \begin{tikzpicture}[>=stealth, every node/.style={font=\scriptsize}]
    \draw[->,gray!55] (-0.5,0) -- (7.4,0) node[right,black]{$g$};
    \fill[UNB_GOLD!12] (0,0) rectangle (6.4,0.7);
    \foreach \i in {0,...,8}
      \draw[gray!70, thick] (\i*0.8,0) -- (\i*0.8,0.7);
    \draw[UNB_BLUE,very thick] (0,-0.12)--(0,0.7);
    \draw[UNB_BLUE,very thick] (6.4,-0.12)--(6.4,0.7);
    \node[below,black] at (0,-0.14) {$V_{\min}$};
    \node[below,black] at (6.4,-0.14) {$V_{\max}$};
    \draw[<->,thick,UNB_BLUE] (0,1.0) -- (0.8,1.0)
        node[midway,above,black]{$\Delta$};
    \node[gray!60!black] at (3.2,-0.5) {$L$ intervalos iguais\, (aqui $L=8$)};
  \end{tikzpicture}
  \end{center}
\end{frame}

\begin{frame}{O Quantizador Uniforme — Passo 3: Os Níveis de Quantização}
  Cada intervalo é representado por um único valor: o seu \textbf{ponto
  médio}. O $k$-ésimo intervalo
  $[V_{\min}+k\Delta,\; V_{\min}+(k{+}1)\Delta)$ recebe o nível
  \[
    \hat{g} = V_{\min} + \left(k+\tfrac12\right)\Delta,
    \qquad k = 0,1,\dots,L-1.
  \]

  {\footnotesize Com $n$ bits há $L = 2^n$ níveis, cada um com seu próprio

\end{frame}

\subsection{Ruído de Quantização}
% ------------------------------------------------------------------

\begin{frame}{O Erro de Quantização}
  O \textbf{erro de quantização} é a diferença entre o valor
  quantizado e o valor real:
  \[
    e_q[n] = \hat{g}[n] - g[n]
  \]

  \vspace{0.3cm}
  Como o quantizador mapeia cada amostra para o
  \textbf{ponto médio} de seu intervalo $\Delta$,
  o erro fica sempre limitado a:
  \[
    -\frac{\Delta}{2} \;\le\; e_q[n] \;\le\; +\frac{\Delta}{2}
  \]

  \vspace{0.4cm}
  \textbf{Compromisso fundamental:}
  \begin{itemize}
    \item Erro menor $\Rightarrow$ $\Delta$ menor
          $\Rightarrow$ mais níveis $L = V_{pp}/\Delta$
          $\Rightarrow$ mais bits $n = \log_2 L$
    \item Aumentar $n$ em 1 bit $\Rightarrow$ $\Delta$ cai pela metade
          $\Rightarrow$ erro máximo cai pela metade
  \end{itemize}
\end{frame}

% ------------------------------------------------------------------

\begin{frame}{Modelo Estatístico do Erro de Quantização}
  \textbf{Hipóteses do modelo de ruído de quantização}
  (válidas para $L$ grande e sinal variando rapidamente):

  \begin{enumerate}
    \item $e_q[n]$ é uma \textbf{variável aleatória uniforme} em
          $[-\Delta/2,\; +\Delta/2]$
    \item Erros em amostras consecutivas são \textbf{não-correlacionados}
    \item O erro é \textbf{independente} do sinal de entrada
  \end{enumerate}

  \vspace{0.3cm}
  Sob essas hipóteses, a função de densidade de probabilidade (PDF) é:
  \[
    p(e_q) =
    \begin{cases}
      \dfrac{1}{\Delta}, & \text{se } -\dfrac{\Delta}{2}
                            \le e_q \le \dfrac{\Delta}{2} \\[8pt]
      0,                 & \text{caso contrário}
    \end{cases}
  \]

  Cada valor do erro é \textbf{igualmente provável} dentro do intervalo.
\end{frame}

% ------------------------------------------------------------------

\begin{frame}{PDF do Erro de Quantização}
  \begin{center}
    \includegraphics[width=0.60\textwidth]{figures/cap7/quantization_error_pdf}
  \end{center}
  \begin{center}
  {\footnotesize Erro $e_q$ uniformemente distribuído em

  \end{center}
\end{frame}

% ------------------------------------------------------------------

\begin{frame}{PDF do Erro de Quantização — Verificações}
  O modelo uniforme $p(e_q) = 1/\Delta$ em $[-\Delta/2,\,+\Delta/2]$
  satisfaz as duas propriedades esperadas de uma densidade:

  \vspace{0.6cm}
  \textbf{1. Área unitária} (é uma densidade de probabilidade válida):
  \[
    \int_{-\Delta/2}^{\Delta/2}\frac{1}{\Delta}\,de_q = 1
    \qquad(\checkmark)
  \]

  \vspace{0.4cm}
  \textbf{2. Média nula} (erro sem viés — tão provável para cima quanto
  para baixo):
  \[
    E[e_q] = \int_{-\Delta/2}^{\Delta/2} e_q \cdot\frac{1}{\Delta}\,de_q = 0
    \qquad(\checkmark)
  \]
\end{frame}

% ------------------------------------------------------------------
\subsection{Derivação de $P_q = \Delta^2/12$}
% ------------------------------------------------------------------

\begin{frame}{Derivação de $P_q$ — Passo 1: Definição Formal}
  A \textbf{potência do ruído de quantização} $P_q$ é o valor esperado
  do \emph{quadrado} do erro:
  \[
    P_q = E\!\left[e_q^2\right]
  \]

  \vspace{0.3cm}
  Para uma variável aleatória contínua, o valor esperado de uma função
  $f(e_q)$ é calculado como:
  \[
    E\!\left[f(e_q)\right] = \int_{-\infty}^{+\infty} f(e_q)\,p(e_q)\,de_q
  \]

  \vspace{0.3cm}
  Portanto, a definição formal de $P_q$ é:
  \[
    P_q = \int_{-\infty}^{+\infty} e_q^2 \; p(e_q) \; de_q
  \]

  \textit{Interpretação:} pesamos $e_q^2$ pela sua
  probabilidade $p(e_q)$ e somamos (integramos) sobre todos os valores.
\end{frame}

% ------------------------------------------------------------------

\begin{frame}{Derivação de $P_q$ — Passo 2: Substituição da PDF}
  Como $p(e_q) = 1/\Delta$ somente no intervalo
  $[-\Delta/2, +\Delta/2]$ e zero fora dele,
  podemos restringir os limites da integral:
  \[
    P_q = \int_{-\Delta/2}^{+\Delta/2} e_q^2 \cdot \frac{1}{\Delta} \; de_q
  \]

  \vspace{0.6cm}
  A constante $1/\Delta$ não depende de $e_q$ e pode sair da integral:
  \[
    P_q = \frac{1}{\Delta}
          \int_{-\Delta/2}^{+\Delta/2} e_q^2 \; de_q
  \]
\end{frame}

% ------------------------------------------------------------------

\begin{frame}{Derivação de $P_q$ — Passo 3: Resolver a Integral}
  Partindo de
  $P_q = \dfrac{1}{\Delta}\displaystyle\int_{-\Delta/2}^{+\Delta/2} e_q^2 \, de_q$,
  a primitiva de $e_q^2$ é $e_q^3/3$.

  \vspace{0.5cm}
  Aplicando o teorema fundamental do cálculo:
  \[
    P_q = \frac{1}{\Delta}
          \left[\frac{e_q^3}{3}\right]_{-\Delta/2}^{+\Delta/2}
        = \frac{1}{\Delta}
          \cdot\frac{\left(\Delta/2\right)^3 - \left(-\Delta/2\right)^3}{3}
  \]
\end{frame}

% ------------------------------------------------------------------

\begin{frame}{Derivação de $P_q$ — Passo 4: Simplificação}
  Calculamos os cubos:
  \[
    \left(\frac{\Delta}{2}\right)^{\!3} = \frac{\Delta^3}{8},
    \qquad
    \left(-\frac{\Delta}{2}\right)^{\!3} = -\frac{\Delta^3}{8}
  \]

  \vspace{0.3cm}
  Substituindo na expressão anterior:
  \[
    P_q = \frac{1}{\Delta}
          \cdot\frac{\;\dfrac{\Delta^3}{8} -
                     \left(-\dfrac{\Delta^3}{8}\right)}{3}
        = \frac{1}{\Delta}
          \cdot\frac{\;2\cdot\dfrac{\Delta^3}{8}\;}{3}
        = \frac{1}{\Delta}\cdot\frac{\Delta^3}{12}
  \]

  \vspace{0.3cm}
  Cancelando um fator $\Delta$, chegamos ao resultado:
  \[
    \boxed{\,P_q = \frac{\Delta^2}{12}\,}
  \]
\end{frame}

% ------------------------------------------------------------------

\begin{frame}{Potência do Ruído de Quantização — Interpretação}
  \begin{block}{Potência do Ruído de Quantização}
    \[
      P_q = \frac{\Delta^2}{12}
    \]
    O ruído depende \textbf{apenas} do passo $\Delta$ — quanto mais fino
    o degrau, menor a potência do erro.
  \end{block}

  \vspace{0.4cm}
  \textbf{Consequência prática:}
  \begin{itemize}
    \item Para reduzir $P_q$ à metade ($-3$ dB), reduz-se $\Delta$ por
          $\sqrt{2}$, ou seja, dobra-se o número de níveis $L$.
    \item Dobrar $L$ corresponde a adicionar \textbf{1 bit} de resolução.
    \item Exemplo: um ADC de 12 bits tem $\Delta$ 16$\times$ menor que um
          de 8 bits $\Rightarrow$ $P_q$ é $256\times$ menor ($-24$ dB).
  \end{itemize}
\end{frame}

\subsection{SQNR e Regra dos 6 dB}
% ------------------------------------------------------------------

\begin{frame}{Relação Sinal-Ruído de Quantização (SQNR)}
  A qualidade da quantização é medida pelo \textbf{SQNR}
  (\textit{Signal-to-Quantization-Noise Ratio}):
  \[
    \SQNR = \frac{P_s}{P_q}
           = \frac{P_s}{\Delta^2/12}
           = \frac{12\,P_s}{\Delta^2}
  \]

  \vspace{0.3cm}
  Expressando $\Delta$ em termos do número de bits $n$
  ($L = 2^n$ níveis, faixa $V_{pp}$):
  \[
    \Delta = \frac{V_{pp}}{2^n}
    \;\Longrightarrow\;
    \SQNR = \frac{12\,P_s}{\left(V_{pp}/2^n\right)^2}
           = \frac{12\,P_s}{V_{pp}^2}\cdot 2^{2n}
  \]

  \vspace{0.3cm}
  \textbf{Observação:} $2^{2n} = 4^n$.
  Adicionar 1 bit multiplica o SQNR por 4 (ou seja, $+6$ dB).
\end{frame}

% ------------------------------------------------------------------

\begin{frame}{SQNR para um Sinal Senoidal — Derivação}
  \textbf{Caso prático:} senoide $g(t) = A\cos(\omega t)$ que ocupa
  \textbf{toda a faixa} do quantizador.

  \vspace{0.3cm}
  Parâmetros:
  \[
    V_{pp} = A - (-A) = 2A,
    \qquad
    P_s = \frac{A^2}{2}
    \quad\text{(potência média da senoide)}
  \]

  \vspace{0.3cm}
  Substituindo na expressão geral do SQNR:
  \[
    \SQNR = \frac{12\cdot(A^2/2)}{(2A)^2}\cdot 2^{2n}
           = \frac{6A^2}{4A^2}\cdot 2^{2n}
           = \frac{3}{2}\cdot 2^{2n}
           = 1{,}5\cdot 4^n
  \]

  \vspace{0.3cm}
  Convertendo para decibéis ($X_{\mathrm{dB}} = 10\log_{10}X$):
  \[
    \SQNR_{\mathrm{dB}}
    = 10\log_{10}\!\left(1{,}5\cdot 4^n\right)
    = 10\log_{10}(1{,}5) + n\cdot 10\log_{10}(4)
  \]
\end{frame}

% ------------------------------------------------------------------

\begin{frame}{A Regra dos 6 dB por Bit}
  Calculando os valores numéricos dos logaritmos:
  \[
    10\log_{10}(1{,}5) \approx 1{,}76\;\text{dB},
    \qquad
    10\log_{10}(4)     \approx 6{,}02\;\text{dB}
  \]

  \vspace{0.2cm}
  Portanto:
  \[
    \SQNR_{\mathrm{dB}} \approx 1{,}76 + 6{,}02\,n
  \]

  \begin{block}{Regra dos 6 dB/bit (senoide em faixa cheia)}
    \[
      \SQNR_{\mathrm{dB}} \approx 1{,}76 + 6{,}02\,n\;\text{dB}
    \]
    Cada bit adicional eleva o SQNR em \textbf{$\approx 6$ dB}.
  \end{block}
\end{frame}

% ------------------------------------------------------------------

\begin{frame}{Regra dos 6 dB/bit — Exemplos de Engenharia}
  Aplicando $\SQNR_{\mathrm{dB}} \approx 1{,}76 + 6{,}02\,n$ a sistemas reais:

  \vspace{0.3cm}
  \begin{itemize}
    \item \textbf{Telefonia PCM} ($n = 8$ bits): $\SQNR \approx 50\,\text{dB}$
          — suficiente para voz inteligível na rede telefônica.
    \item \textbf{CD de áudio} ($n = 16$ bits): $\SQNR \approx 98\,\text{dB}$
          — cobre toda a faixa dinâmica da audição humana.
    \item \textbf{Estúdio profissional} ($n = 24$ bits):
          $\SQNR \approx 146\,\text{dB}$ — margem para edição e mixagem
          sem acúmulo audível de ruído.
  \end{itemize}

  \vspace{0.3cm}
  \begin{block}{Critério de projeto}
    Para atingir um SQNR-alvo, o número de bits necessário é
    $n \ge \dfrac{\SQNR_{\mathrm{dB}} - 1{,}76}{6{,}02}$.
    Ex.: para 70 dB $\Rightarrow$ $n \ge 11{,}3 \Rightarrow$ 12 bits.
  \end{block}
\end{frame}

\begin{frame}[fragile]{Implementação em Python}
\begin{lstlisting}[language=Python]
import numpy as np

def quantize_uniform(x, n_bits, v_min=-1.0, v_max=1.0):
    L     = 2**n_bits                        # numero de niveis
    delta = (v_max - v_min) / L             # tamanho do passo
    x_c   = np.clip(x, v_min, v_max)        # limitar faixa
    idx   = np.floor((x_c - v_min)/delta).astype(int)
    idx   = np.clip(idx, 0, L-1)
    return v_min + (idx + 0.5) * delta       # ponto medio

# Calcular SQNR para varios numeros de bits
t = np.linspace(0, 100, 1_000_000)
g = np.sin(2*np.pi*t)                       # senoide em faixa cheia

for n in [4, 8, 16]:
    g_q  = quantize_uniform(g, n)
    sqnr = 10 * np.log10(np.mean(g**2) / np.mean((g_q-g)**2))
    teo  = 1.76 + 6.02*n
    print(f"n={n:2d}: SQNR = {sqnr:.1f} dB  (teoria: {teo:.1f} dB)")
\end{lstlisting}
\end{frame}
