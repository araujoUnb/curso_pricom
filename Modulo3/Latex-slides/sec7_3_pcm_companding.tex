\begin{frame}{Codificação Binária: PCM}
  Após a quantização, cada nível $\hat{g}[n]$ recebe uma
  \textbf{palavra binária de $n$ bits} (representação PCM).

  \vspace{0.2cm}
  Com $n$ bits podemos representar $L = 2^n$ níveis:
  \[
    L = 2^n
    \quad\Longleftrightarrow\quad
    n = \log_2 L
  \]

  \vspace{0.15cm}
  \textbf{Exemplo com $n = 3$ bits ($L = 8$ níveis):}

  \vspace{0.1cm}
  \begin{center}
  \begin{tabular}{ccc}
    \hline
    Índice $k$ & Código binário & Nível $\hat{g}$\\
    \hline
    0 & \texttt{000} & $V_{\min} + 0{,}5\Delta$ \\
    1 & \texttt{001} & $V_{\min} + 1{,}5\Delta$ \\
    $\vdots$ & $\vdots$ & $\vdots$ \\
    7 & \texttt{111} & $V_{\min} + 7{,}5\Delta$ \\
    \hline
  \end{tabular}
  \end{center}
\end{frame}

\begin{frame}{Taxa de Bits e Largura de Banda PCM}
  A \textbf{taxa de bits} (bit rate) do canal PCM é:
  \[
    R_b = n \cdot f_s \quad \text{[bits/s]}
  \]
  onde $n$ é o número de bits por amostra e
  $f_s$ é a taxa de amostragem.

  \vspace{0.35cm}
  A largura de banda mínima necessária para transmitir o PCM
  (sem inter-símbolos) é:
  \[
    B_{\min} = \frac{R_b}{2} = \frac{n\,f_s}{2}
  \]

  \vspace{0.35cm}
  \textbf{Compromisso clássico:}
  \begin{itemize}
    \item Qualidade alta $\Rightarrow$ $n$ grande $\Rightarrow$ $R_b$ alto
          $\Rightarrow$ mais largura de banda
    \item Largura de banda limitada $\Rightarrow$ $n$ reduzido
          $\Rightarrow$ mais ruído de quantização
  \end{itemize}
\end{frame}

\begin{frame}{De Onde Vem o Fator 2? — Critério de Nyquist}
  O fator $2$ vem da \textbf{sinalização no canal}: um canal de banda $B$
  transmite até \textbf{$2B$ pulsos/s} sem interferência entre símbolos
  (ISI).

  \vspace{0.05cm}
  \begin{center}
  \begin{tikzpicture}[>=stealth, every node/.style={font=\scriptsize},
      declare function={ms(\t)= abs(\t)<0.0008 ? 1 : sin(180*\t)/(pi*\t);}]
    \def\T{1.35}\def\A{0.82}
    \draw[->,gray!55] (-2.3,0) -- (5.9,0) node[right,black]{$t$};
    \draw[->,gray!55] (0,-0.4) -- (0,1.08);
    % três pulsos sinc (símbolos vizinhos)
    \draw[gray!55, thick]
       plot[domain=-2.7:2.5, samples=130] (\x*\T, {\A*ms(\x+1)});
    \draw[UNB_BLUE, very thick]
       plot[domain=-1.7:3.5, samples=130] (\x*\T, {\A*ms(\x)});
    \draw[UNB_GREEN!70!black, thick]
       plot[domain=-0.7:4.5, samples=130] (\x*\T, {\A*ms(\x-1)});
    % instantes de símbolo
    \foreach \k in {-1,0,1} \draw[densely dotted,gray] (\k*\T,-0.1) -- (\k*\T,1.05);
    \fill[UNB_BLUE] (0,\A) circle (1.7pt);
    \fill (-\T,0) circle (1.3pt);  \fill (\T,0) circle (1.3pt);
    % espaçamento entre símbolos
    \draw[<->,thick,black] (0,-0.35) -- (\T,-0.35)
       node[midway,below]{$T=\dfrac{1}{2B}$};
    \node[UNB_BLUE] at (0.35,1.18) {1};
  \end{tikzpicture}
  \end{center}

  \vspace{-0.05cm}
  Os \textbf{cruzamentos por zero} ficam a $T = 1/(2B)$ um do outro.
  Pondo um símbolo em cada zero, nenhum interfere no vizinho, logo a taxa
  máxima é $R_s = 1/T = 2B$ símbolos/s:
  \[
    R_s \le 2B \;\Longrightarrow\; B \ge \frac{R_s}{2}
    \;\xrightarrow{\;R_s = R_b\;}\;
    B_{\min} = \frac{R_b}{2} = \frac{n f_s}{2}.
  \]
\end{frame}

\begin{frame}{Dualidade: Amostragem $\leftrightarrow$ Sinalização}
  O mesmo fator $2$ aparece na amostragem e na transmissão — é a
  \textbf{dualidade tempo $\leftrightarrow$ frequência}:

  \vspace{0.3cm}
  \begin{center}
  \begin{tikzpicture}[>=stealth,
      box/.style={draw, rounded corners=4pt, align=center,
                  minimum width=5.2cm, minimum height=1.7cm, font=\small}]
    \node[box, fill=UNB_BLUE!10, draw=UNB_BLUE] (a)
      {\textbf{Amostragem}\\[2pt]
       sinal de banda $W$ exige\\
       $f_s \ge 2W$ amostras/s};
    \node[box, fill=UNB_GREEN!10, draw=UNB_GREEN, right=1.6cm of a] (b)
      {\textbf{Sinalização}\\[2pt]
       banda $B$ comporta\\
       $R_s \le 2B$ símbolos/s};
    \draw[<->, very thick, UNB_GOLD] (a) -- (b)
       node[midway, above, font=\scriptsize\itshape, black] {dual};
  \end{tikzpicture}
  \end{center}

  \vspace{0.3cm}
  \begin{itemize}
    \item Amostrar: preciso de \textbf{2 amostras por ciclo} da maior
          frequência $W$.
    \item Transmitir: cabem \textbf{2 símbolos por ciclo} da banda $B$ do
          canal.
  \end{itemize}

  \vspace{0.15cm}
  {\footnotesize O critério de sinalização é detalhado no Capítulo 6
  (pulsos de Nyquist e diagrama de olho).}
\end{frame}

\begin{frame}{Exemplo de Projeto PCM: Telefonia Digital}
  \textbf{Parâmetros do canal de voz (ITU-T G.711):}

  \vspace{0.2cm}
  \begin{itemize}
    \item Largura de banda da voz: $W = 4\,\text{kHz}$
    \item Taxa de amostragem escolhida: $f_s = 8\,\text{kHz}$
          \quad (garante $f_s > 2W$, com margem)
    \item Resolução: $n = 8$ bits/amostra ($L = 256$ níveis)
  \end{itemize}

  \vspace{0.35cm}
  \textbf{Taxa de bits por canal:}
  \[
    R_b = n \cdot f_s = 8 \times 8\,000 = 64\,\text{kb/s}
  \]

  \textbf{SQNR máximo (senoide):}
  \[
    \SQNR_{\mathrm{dB}} \approx 1{,}76 + 6{,}02 \times 8 \approx 50\,\text{dB}
  \]

  \vspace{0.2cm}
  \textbf{Observação:} 50 dB é adequado para inteligibilidade da voz,
  mas ainda abaixo da qualidade de áudio de alta fidelidade (CD: $\sim$98 dB).
\end{frame}

% ------------------------------------------------------------------

\begin{frame}{Exemplo Atual: Voz HD na Telefonia Móvel (VoLTE)}
  \textbf{Parâmetros da voz \emph{wideband}} (``HD Voice'', base do
  codec AMR-WB usado no 4G/5G):

  \vspace{0.15cm}
  \begin{itemize}
    \item Largura de banda da voz: $W = 7\,\text{kHz}$
          \quad (vs.\ $3{,}4$ kHz na telefonia fixa)
    \item Taxa de amostragem: $f_s = 16\,\text{kHz}$
    \item Resolução PCM de entrada: $n = 16$ bits/amostra
  \end{itemize}

  \vspace{0.25cm}
  \textbf{Taxa de bits PCM bruta:}
  \[
    R_b = n \cdot f_s = 16 \times 16\,000 = 256\,\text{kb/s}
    \qquad (\SQNR \approx 1{,}76 + 6{,}02 \times 16 \approx 98\,\text{dB})
  \]

  \vspace{0.1cm}
  \textbf{Observação:} $256$ kb/s é alto demais para o rádio móvel. A voz
  é então \textbf{comprimida} por um codec — AMR-WB ($\sim$12--24 kb/s) ou
  EVS no 5G — que preserva a qualidade percebida com uma fração dos bits.
\end{frame}

\begin{frame}{Como Calcular os Níveis Ótimos: Lloyd-Max}
  O quantizador ótimo minimiza o erro quadrático médio
  $E\!\left[(x-\hat{x})^2\right]$ dada a densidade $p(x)$, satisfazendo
  \textbf{duas condições}:

  \vspace{0.15cm}
  \textbf{1. Cada nível é o centróide (média) do seu intervalo:}
  \[
    \hat{y}_k = \frac{\displaystyle\int_{b_{k-1}}^{b_k} x\,p(x)\,dx}
                     {\displaystyle\int_{b_{k-1}}^{b_k} p(x)\,dx}
  \]

  \textbf{2. Cada fronteira é o ponto médio entre níveis vizinhos:}
  \[
    b_k = \frac{\hat{y}_k + \hat{y}_{k+1}}{2}
  \]

  \vspace{-0.05cm}
  {\footnotesize \textbf{Algoritmo de Lloyd:} alterna entre as duas
  condições até convergir.}
\end{frame}

% ------------------------------------------------------------------

\begin{frame}{Demonstração — Condição 1: Nível é o Centróide}
  A distorção (erro quadrático médio) é a soma sobre os intervalos:
  \[
    D = \sum_{k} \int_{b_{k-1}}^{b_k} \left(x - \hat{y}_k\right)^2 p(x)\,dx.
  \]

  Com as fronteiras fixas, fazemos $\partial D / \partial \hat{y}_k = 0$:
  \[
    \int_{b_{k-1}}^{b_k} -2\left(x - \hat{y}_k\right) p(x)\,dx = 0
    \;\Longrightarrow\;
    \int_{b_{k-1}}^{b_k}\! x\,p(x)\,dx = \hat{y}_k \int_{b_{k-1}}^{b_k}\! p(x)\,dx.
  \]

  Isolando $\hat{y}_k$, obtemos o \textbf{centróide}:
  \[
    \boxed{\;\hat{y}_k = \left.\int_{b_{k-1}}^{b_k} x\,p(x)\,dx \;\middle/\;
             \int_{b_{k-1}}^{b_k} p(x)\,dx\;\right.}
  \]
\end{frame}

% ------------------------------------------------------------------

\begin{frame}{Demonstração — Condição 2: A Derivada de Leibniz}
  A fronteira $b_k$ é \textbf{limite superior} do 1º termo e
  \textbf{limite inferior} do 2º termo da distorção:
  \[
    D_k = \int_{b_{k-1}}^{b_k}\!\!\left(x-\hat{y}_k\right)^2 p(x)\,dx
        + \int_{b_k}^{b_{k+1}}\!\!\left(x-\hat{y}_{k+1}\right)^2 p(x)\,dx.
  \]

  \textbf{Regra de Leibniz:} derivar em relação a um limite devolve o
  integrando avaliado nele — com sinal $+$ no superior, $-$ no inferior:
  \[
    \frac{\partial}{\partial b_k}\!\int_{a}^{b_k}\! f(x)\,dx = f(b_k),
    \qquad
    \frac{\partial}{\partial b_k}\!\int_{b_k}^{c}\! f(x)\,dx = -\,f(b_k).
  \]

  Aqui o integrando é $f(x) = (x-\hat{y})^2 p(x)$, avaliado em $x=b_k$:
  \[
    \frac{\partial D}{\partial b_k}
    = \underbrace{\left(b_k-\hat{y}_k\right)^2 p(b_k)}_{\text{1º termo }(+)}
    \;-\;
    \underbrace{\left(b_k-\hat{y}_{k+1}\right)^2 p(b_k)}_{\text{2º termo }(-)}.
  \]
\end{frame}

% ------------------------------------------------------------------

\begin{frame}{Demonstração — Condição 2: O Ponto Médio}
  No ótimo, $\dfrac{\partial D}{\partial b_k} = 0$, logo:
  \[
    \left(b_k-\hat{y}_k\right)^2 p(b_k)
    = \left(b_k-\hat{y}_{k+1}\right)^2 p(b_k).
  \]

  Como $p(b_k) > 0$, cancelamos a densidade:
  \[
    \left(b_k-\hat{y}_k\right)^2 = \left(b_k-\hat{y}_{k+1}\right)^2.
  \]

  Como $\hat{y}_k < b_k < \hat{y}_{k+1}$, temos $b_k-\hat{y}_k > 0$ e
  $b_k-\hat{y}_{k+1} < 0$; a raiz válida é
  $b_k-\hat{y}_k = \hat{y}_{k+1}-b_k$. Isolando $b_k$ — o
  \textbf{ponto médio}:
  \[
    \boxed{\;b_k = \frac{\hat{y}_k + \hat{y}_{k+1}}{2}\;}
  \]
\end{frame}

\begin{frame}{Exemplo: Nível Ótimo de Um Intervalo}
  Seja o intervalo $[0, a]$ com densidade \textbf{decrescente}
  $p(x) = \dfrac{2(a-x)}{a^2}$ (amplitudes pequenas mais prováveis).

  \vspace{0.2cm}
  O nível ótimo é o \textbf{centróide} do intervalo:
  \[
    \hat{y} = \frac{\displaystyle\int_0^a x\,p(x)\,dx}
                   {\displaystyle\int_0^a p(x)\,dx}
            = \frac{\dfrac{2}{a^2}\displaystyle\int_0^a x\,(a-x)\,dx}{1}
            = \frac{2}{a^2}\cdot\frac{a^3}{6}
            = \boxed{\dfrac{a}{3}}
  \]

  \vspace{0.15cm}
  \textbf{Interpretação:} o nível ótimo cai em $a/3$, \textbf{deslocado
  para perto de zero} — e não no meio geométrico $a/2$ — acompanhando a
  concentração da densidade.
\end{frame}

\begin{frame}{Exemplo Prático: Níveis Ótimos (Centróides)}
  Quantizador de \textbf{2 níveis} (1 bit), partindo da fronteira uniforme
  $b = 2$ V. Cada nível é o \textbf{centróide} do seu lado:
  \[
    \hat{y}_1 = \frac{\displaystyle\int_0^2 x\,p\,dx}{\displaystyle\int_0^2 p\,dx}
              = \frac{2/3}{3/4} = \frac{8}{9} \approx 0{,}89\ \text{V},
    \quad
    \hat{y}_2 = \frac{\displaystyle\int_2^4 x\,p\,dx}{\displaystyle\int_2^4 p\,dx}
              = \frac{2/3}{1/4} = \frac{8}{3} \approx 2{,}67\ \text{V}.
  \]

  \vspace{0.05cm}
  \begin{center}
  \begin{tikzpicture}[>=stealth, x=1.25cm, y=1.7cm,
      every node/.style={font=\scriptsize}]
    \draw[->,gray!55] (-0.2,0) -- (4.7,0) node[right,black]{$x$ (V)};
    \fill[UNB_GOLD!12] (0,0) -- (0,0.5) -- (4,0) -- cycle;
    \draw[UNB_GOLD!85!black, very thick] (0,0.5) -- (4,0);
    \draw[UNB_BLUE, densely dashed, thick] (2,0) -- (2,0.5)
         node[above,black,font=\tiny]{$b=2$};
    % níveis ótimos (verde)
    \fill[UNB_GREEN] (0.89,0) circle (1.7pt);
    \fill[UNB_GREEN] (2.67,0) circle (1.7pt);
    \node[UNB_GREEN!50!black,font=\tiny,below=1pt] at (0.89,0) {$\hat y_1$};
    \node[UNB_GREEN!50!black,font=\tiny,below=1pt] at (2.67,0) {$\hat y_2$};
    % meios geométricos (cinza)
    \draw[gray!70] (1,0.05)--(1,-0.05); \node[gray,font=\tiny,above] at (1,0.05){$1{,}0$};

  \vspace{-0.05cm}
  {\footnotesize Os níveis (verde) ficam \textbf{abaixo} dos meios
  geométricos (cinza, $1{,}0$ e $3{,}0$ V): migram para onde há mais sinal.}
\end{frame}

\begin{frame}{Exemplo Prático: Atualizar a Fronteira (Lloyd)}
  \textbf{Atualizar a fronteira} (ponto médio dos novos níveis):
  \[
    b = \frac{\hat{y}_1 + \hat{y}_2}{2}
      = \frac{1}{2}\!\left(\frac{8}{9} + \frac{8}{3}\right)
      = \frac{16}{9} \approx 1{,}78\ \text{V}.
  \]

  A fronteira \textbf{migrou de $2{,}0$ para $1{,}78$ V} — para perto de
  zero, onde o sinal passa mais tempo. \textbf{Iterando} (recalcular
  centróides $\to$ atualizar fronteira), o algoritmo de Lloyd converge.

  \vspace{0.2cm}
  \begin{block}{}
  O resultado é um quantizador \textbf{não-uniforme} com degraus finos
  perto de zero e largos nas amplitudes altas — exatamente o efeito que o
  \textbf{companding} ($\mu$-law / A-law) produz na prática.
  \end{block}
\end{frame}

\begin{frame}{O Conceito de Companding}
  \textbf{Companding = Compressão + Quantização Uniforme + Expansão}

  \vspace{0.35cm}
  {\centering
  \begin{tikzpicture}[
    cmpblk/.style={draw, rectangle, rounded corners=4pt,
                   minimum width=2.0cm, minimum height=0.85cm,
                   text centered, fill=unbprimary!12,
                   draw=unbprimary, font=\small},
    arw/.style={->, thick, >=stealth, color=unbprimary},
    lbl/.style={font=\scriptsize, color=unbsecondary}
  ]
    \node[cmpblk]                       (comp)  {Compressor};
    \node[cmpblk, right=0.9cm of comp]  (quant) {Quant. Uniforme};
    \node[cmpblk, right=0.9cm of quant] (chan)  {Canal};
    \node[cmpblk, right=0.9cm of chan]  (exp)   {Expansor};

    \draw[arw] (-1.5,0) node[left, font=\small]{$g[n]$} -- (comp);
    \draw[arw] (comp)  -- (quant) node[midway,above,lbl]{comprimido};
    \draw[arw] (quant) -- (chan)  node[midway,above,lbl]{bits PCM};
    \draw[arw] (chan)  -- (exp)   node[midway,above,lbl]{bits};
    \draw[arw] (exp)   -- ++(1.5,0) node[right,font=\small]{$\hat{g}[n]$};

    \node[lbl, below=0.4cm of comp]  {TX};
    \node[lbl, below=0.4cm of exp]   {RX};
  \end{tikzpicture}
  \par}

  \vspace{0.3cm}
  \begin{itemize}
    \item \textbf{Compressor:} amplifica amplitudes pequenas e comprime
          amplitudes grandes antes da quantização
    \item \textbf{Expansor:} aplica a função inversa no receptor
          para restaurar as amplitudes originais
  \end{itemize}
\end{frame}

\begin{frame}{Lei $\mu$: Definição Matemática}
  Padrão nos EUA, Canadá e Japão (ITU-T G.711, $\mu = 255$):
  \[
    C_\mu(x) = \operatorname{sgn}(x) \,
               \frac{\ln\!\left(1 + \mu\,|x|\right)}{\ln(1 + \mu)},
               \qquad |x| \le 1
  \]

  \vspace{0.2cm}
  \textbf{Propriedades:}
  \begin{itemize}
    \item Para $\mu|x| \gg 1$ (sinal grande):
          $C_\mu(x) \approx \operatorname{sgn}(x)\,\dfrac{\ln(\mu|x|)}{\ln(1+\mu)}$
          — compressão logarítmica
    \item Para $\mu|x| \ll 1$ (sinal pequeno):
          $C_\mu(x) \approx x$ — aproximadamente linear
    \item Caso limite: $\mu \to 0$ recupera a quantização uniforme
  \end{itemize}

  \vspace{0.2cm}
  A \textbf{inversão} (expansor) é:
  \[
    C_\mu^{-1}(y) = \operatorname{sgn}(y) \,
                    \frac{(1+\mu)^{|y|} - 1}{\mu}
  \]
\end{frame}

\begin{frame}{Lei A: Definição Matemática}
  Padrão na Europa e em sistemas legados no Brasil
  (ITU-T G.711, $A = 87{,}6$):
  \[
    C_A(x) = \operatorname{sgn}(x)\cdot
    \begin{cases}
      \dfrac{A\,|x|}{1 + \ln A},             & 0 \le |x| < \dfrac{1}{A}\\[8pt]
      \dfrac{1 + \ln\!\left(A\,|x|\right)}{1 + \ln A},
                                              & \dfrac{1}{A} \le |x| \le 1
    \end{cases}
  \]

  \vspace{0.2cm}
  \textbf{Diferença em relação à lei $\mu$:}
  \begin{itemize}
    \item A lei A é \textbf{linear} para amplitudes muito pequenas
          ($|x| < 1/A$) e logarítmica para amplitudes maiores
    \item A lei $\mu$ é logarítmica em toda a faixa
    \item Ambas garantem SQNR quase constante
          em uma ampla faixa de níveis de sinal
  \end{itemize}
\end{frame}

\begin{frame}{Resumo: Conversão Analógico-Digital}
  \begin{block}{Resultado 1 — Amostragem (Nyquist-Shannon)}
    \[
      f_s > 2W
      \qquad\Longrightarrow\qquad
      G_s(f) = f_s\sum_{k=-\infty}^{\infty} G(f - k\,f_s)
    \]
  \end{block}

  \vspace{0.25cm}

  \begin{block}{Resultado 2 — Ruído de Quantização}
    \[
      P_q = \frac{\Delta^2}{12},
      \qquad
      \Delta = \frac{V_{pp}}{2^n}
    \]
  \end{block}

  \vspace{0.25cm}

  \begin{block}{Resultado 3 — SQNR (senoide em faixa cheia)}
    \[
      \SQNR_{\mathrm{dB}} \approx 1{,}76 + 6{,}02\,n \;\text{dB}
      \qquad\longleftarrow\quad 6\text{ dB por bit}
    \]
  \end{block}

  \vspace{0.2cm}
  \textbf{Companding} ($\mu$-law / A-law) mantém SQNR quase constante
  para sinais com distribuição de amplitude não-uniforme (ex.: voz).
\end{frame}
