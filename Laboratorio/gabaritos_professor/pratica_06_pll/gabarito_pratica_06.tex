\section*{Respostas esperadas}

\begin{enumerate}[label=\textbf{Q\arabic*.}, leftmargin=*, itemsep=1em]

  \item \textbf{Descreva os três componentes fundamentais de um PLL e a função de cada um.}

    \textbf{Resposta esperada:} (1) Detector de Fase (PD): multiplica o sinal de entrada
    pelo sinal do VCO e filtra, produzindo um sinal proporcional a
    $\sin(\phi_i - \phi_o) \approx \phi_e$ para pequenos erros de fase.
    (2) Filtro de Laço (LF): filtro passa-baixas que remove altas frequências e
    determina a largura de banda do sistema (velocidade vs.\ supressão de ruído).
    (3) VCO: oscilador com frequência $\omega_0 + K_v \cdot e(t)$; integra o sinal
    de erro para ajustar a fase. A realimentação fecha o laço: PD detecta erro, LF
    suaviza, VCO corrige.

    \begin{nota}[Erros comuns]
      Descrever apenas o VCO e o comparador de fase sem mencionar o papel do filtro
      de laço na determinação da largura de banda e na supressão de ruído.
    \end{nota}

  \item \textbf{Como se detecta o estado de lock do PLL na saída do \texttt{PLL Freq Det}?}

    \textbf{Resposta esperada:} O PLL está em lock quando a saída do \texttt{PLL Freq Det}
    converge para um valor constante (proporcional à frequência do sinal de entrada).
    Na forma de onda, a saída muda de oscilatória/crescente para uma linha horizontal
    estável. No GNU Radio, o lock se manifesta quando o sinal de saída para de variar
    e estabiliza em um nível DC correspondente à frequência de entrada.

    \begin{nota}[Erros comuns]
      Confundir a convergência da saída com o desaparecimento completo de oscilações
      transitórias. O sobresinal (overshoot) pode persistir brevemente mesmo em lock.
    \end{nota}

  \item \textbf{Como a largura de banda do laço (\texttt{loop\_bw}) afeta a velocidade de lock e o jitter de fase?}

    \textbf{Resposta esperada:} \texttt{loop\_bw} estreita (0,01): convergência lenta
    (muitas amostras), baixo sobresinal, boa supressão de ruído pois o LF atenua
    flutuações de alta frequência. \texttt{loop\_bw} larga (0,5): convergência rápida,
    possível sobresinal significativo (resposta subamortecida), maior jitter de fase
    pois o LF passa mais ruído. O valor ótimo ($\approx 0{,}062$\,rad/amostra) representa
    o compromisso entre velocidade de lock e robustez ao ruído.

    \begin{nota}[Erros comuns]
      Afirmar que \texttt{loop\_bw} larga é sempre preferível por convergir mais rápido,
      sem mencionar o aumento de jitter de fase e a possível instabilidade.
    \end{nota}

  \item \textbf{Qual é a faixa de captura teórica do PLL para \texttt{loop\_bw}$= 0{,}062$\,rad/amostra e $f_s = 32000$\,Hz?}

    \textbf{Resposta esperada:} A faixa de captura teórica é
    $\Delta\omega_\text{cap} \approx 2\omega_n$, onde $\omega_n$ é a frequência natural
    do laço. Para \texttt{loop\_bw}$= 0{,}062$\,rad/amostra e $f_s = 32000$\,Hz:
    $\omega_n \approx 0{,}062 \times f_s = 1984$\,rad/s; logo
    $\Delta f_\text{cap} \approx 2 \times 1984/(2\pi) \approx 631$\,Hz. Discrepâncias
    ocorrem devido à não-linearidade do detector de fase e ao comportamento do filtro
    discreto.

    \begin{nota}[Erros comuns]
      Calcular a faixa de captura em rad/amostra sem converter para Hz, ou usar
      diretamente \texttt{loop\_bw} como a faixa de captura sem a conversão correta.
    \end{nota}

  \item \textbf{Como o ruído externo afeta o jitter de fase do PLL em função de \texttt{loop\_bw}?}

    \textbf{Resposta esperada:} O ruído na entrada do PLL se propaga pelo detector de
    fase e é parcialmente filtrado pelo LF. Um \texttt{loop\_bw} larga (passa mais ruído)
    resulta em maior jitter de fase, pois o VCO responde a flutuações de ruído de alta
    frequência. Um \texttt{loop\_bw} estreita suprime o ruído melhor, mas se o ruído
    for muito intenso, pode fazer o PLL perder o lock completamente. A densidade espectral
    de jitter de saída é proporcional à densidade de ruído de entrada e à largura de
    banda do laço.

    \begin{nota}[Erros comuns]
      Afirmar que \texttt{loop\_bw} estreita é sempre melhor para ruído, sem mencionar
      o tradeoff com a velocidade de aquisição de lock e o risco de perda de lock com
      ruído intenso.
    \end{nota}

\end{enumerate}
