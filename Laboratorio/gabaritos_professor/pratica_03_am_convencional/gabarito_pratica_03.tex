\section*{Respostas esperadas}

\begin{enumerate}[label=\textbf{Q\arabic*.}, leftmargin=*, itemsep=1em]

  \item \textbf{Como calcular o índice de modulação $\mu$ a partir das amplitudes medidas no espectro?}

    \textbf{Resposta esperada:} A relação entre amplitudes espectrais é: a portadora em
    $f_c$ tem amplitude $A_c$; cada banda lateral tem amplitude $\mu A_c/2$. Portanto
    $\mu = 2 \times A_\text{BL} / A_\text{portadora}$. Para $\mu=0{,}5$:
    $A_\text{BL}/A_\text{portadora} = 0{,}25$. Para $\mu=1{,}0$:
    $A_\text{BL}/A_\text{portadora} = 0{,}5$. Esta relação pode ser verificada
    diretamente nas medições do espectro do GNU Radio.

    \begin{nota}[Erros comuns]
      Inverter a relação (escrever $\mu = A_\text{portadora}/(2A_\text{BL})$) ou esquecer
      o fator 2 na fórmula, levando a valores de $\mu$ com erro de fator 2.
    \end{nota}

  \item \textbf{Qual é a diferença visual entre os regimes de submodulação ($\mu<1$), modulação plena ($\mu=1$) e sobremodulação ($\mu>1$) no domínio do tempo?}

    \textbf{Resposta esperada:} Submodulação ($\mu<1$): envoltória sempre positiva,
    variando entre $A_c(1-\mu)$ e $A_c(1+\mu)$. Modulação plena ($\mu=1$): envoltória
    toca zero mas não cruza; amplitude mínima $= 0$, máxima $= 2A_c$. Sobremodulação
    ($\mu>1$): envoltória cruza zero, sinal sofre inversão de fase, criando cruzamentos
    que o detector de envoltória interpreta incorretamente.

    \begin{nota}[Erros comuns]
      Descrever apenas ``picos maiores/menores'' sem identificar o critério de cruzamento
      zero como distinção fundamental entre modulação plena e sobremodulação.
    \end{nota}

  \item \textbf{Por que o detector de envoltória falha quando $\mu > 1$?}

    \textbf{Resposta esperada:} O detector de envoltória extrai $|1 + \mu m_n(t)|$.
    Para $\mu > 1$, existem instantes $t^*$ onde $1 + \mu m_n(t^*) < 0$ (sinal cruza
    zero). O detector calcula o módulo, que inverte a região negativa, gerando distorção
    harmônica grave. A mensagem recuperada apresenta clipping na região inferior,
    tornando-a irreconhecível para o receptor.

    \begin{nota}[Erros comuns]
      Dizer apenas que ``há distorção'' sem explicar o mecanismo matemático:
      $|x| \neq x$ para $x < 0$, o que dobra a frequência aparente na região negativa.
    \end{nota}

  \item \textbf{Qual é a eficiência de potência do AM convencional em função de $\mu$? Compare com o DSB-SC.}

    \textbf{Resposta esperada:} A eficiência é
    $\eta = \tfrac{\mu^2/2}{1 + \mu^2/2}$. Para $\mu=0{,}5$:
    $\eta \approx 11{,}1\%$. Para $\mu=1{,}0$: $\eta \approx 33{,}3\%$.
    O DSB-SC tem $\eta = 100\%$ pois toda a potência está nas bandas laterais (sem
    portadora transmitida). A principal desvantagem do AM convencional é a fração
    de potência desperdiçada na portadora.

    \begin{nota}[Erros comuns]
      Calcular $\eta$ como $\mu^2$ em vez de $\mu^2/2\,/\,(1+\mu^2/2)$, ou afirmar
      que DSB-SC tem $\eta = 50\%$.
    \end{nota}

  \item \textbf{Como a relação entre amplitude da portadora e das bandas laterais muda conforme $\mu$ varia?}

    \textbf{Resposta esperada:} As amplitudes são: portadora $= A_c$, cada banda lateral
    $= \mu A_c/2$. À medida que $\mu$ cresce, as bandas laterais crescem
    proporcionalmente enquanto a portadora permanece constante. Para $\mu=1$, as bandas
    laterais têm metade da amplitude da portadora. Maior portadora em relação às bandas
    laterais implica menor índice de modulação e menor eficiência de transmissão.

    \begin{nota}[Erros comuns]
      Afirmar que a portadora ``some'' ou diminui ao aumentar $\mu$, o que não ocorre
      no AM convencional (a portadora é constante).
    \end{nota}

\end{enumerate}
