\section{Objetivos}

\begin{itemize}
  \item Gerar e demodular sinais FM de faixa estreita (NBFM) e larga
    (WBFM) usando os blocos \texttt{WBFM Transmit} e \texttt{WBFM Receive}.
  \item Medir a largura de banda do sinal FM no espectro para diferentes
    desvios de frequência $\Delta f$ e frequências de mensagem $f_m$.
  \item Verificar experimentalmente a \textbf{Regra de Carson}:
    $B_T \approx 2(\Delta f + f_m)$.
  \item Comparar a qualidade do sinal recuperado para NBFM ($\beta < 1$) e
    WBFM ($\beta \gg 1$) com a adição de ruído.
\end{itemize}

% ======================================================
\section{Fundamentos teóricos (síntese)}

O sinal FM para mensagem senoidal $m(t) = A_m\cos(2\pi f_m t)$ é:
\begin{equation}
  s(t) = A_c\cos\!\left[2\pi f_c t + \frac{\Delta f}{f_m}\sin(2\pi f_m t)\right],
\end{equation}
onde o \textbf{índice de modulação} é $\beta = \Delta f / f_m$.

A \textbf{Regra de Carson} fornece a largura de banda prática:
\begin{equation}
  B_T \approx 2(\beta + 1)f_m = 2(\Delta f + f_m).
  \label{eq:carson}
\end{equation}

O espectro da FM é composto por linhas em $f_c \pm n f_m$ ponderadas pelas
funções de Bessel de primeiro tipo $J_n(\beta)$.  Para $\beta \ll 1$
(NBFM), predominam os termos $n = 0, \pm1$ e o espectro se assemelha ao AM.
Para $\beta \gg 1$ (WBFM), muitos termos contribuem e a largura de banda
cresce com $\beta$.

\section{Roteiro de montagem e execução (50 min)}

\subsection*{Parâmetros iniciais}

\begin{center}
\begin{tabular}{l l}
  \texttt{samp\_rate} = 480\,000\,Hz & \texttt{audio\_rate} = 48\,000\,Hz \\
  $f_m$ = 2\,000\,Hz (variável) & $\Delta f$ = 5\,000\,Hz (variável) \\
  Amplitude mensagem = 1.0 & \\
\end{tabular}
\end{center}

\begin{atencao}[Atenção -- taxa de amostragem]
O bloco \texttt{WBFM Transmit} opera
com taxa de saída $=$ \texttt{samp\_rate}, enquanto \texttt{WBFM Receive}
decima para \texttt{audio\_rate}.  Use \texttt{samp\_rate = 480000} e
\texttt{audio\_rate = 48000} (fator de decimação = 10).
\end{atencao}

\subsection*{Etapa 1 -- Modulador FM e espectro básico (0--14 min)}

\begin{instrucao}[Geração e visualização do sinal FM]
\begin{enumerate}[label=\arabic*.]
  \item Crie variáveis:
    \begin{itemize}
      \item \texttt{samp\_rate = 480000}
      \item \texttt{audio\_rate = 48000}
      \item \texttt{max\_dev = 5000} (desvio máximo em Hz)
    \end{itemize}
  \item Crie \texttt{fm\_freq} como \texttt{QT GUI Range}
    (min: 500, max: 10000, passo: 500, padrão: 2000).
  \item Adicione \textbf{Signal Source}: Cosine, freq = \texttt{fm\_freq},
    amplitude = 1.0, samp\_rate = \texttt{audio\_rate}.
  \item Adicione \texttt{WBFM Transmit}:
    \begin{itemize}
      \item Interpolation = \texttt{samp\_rate / audio\_rate} = 10
      \item Max Deviation = \texttt{max\_dev}
    \end{itemize}
  \item Adicione \texttt{Throttle} (samp\_rate = \texttt{samp\_rate}) e
    \texttt{QT GUI Freq Sink} (FFT Size = 4096, samp\_rate = \texttt{samp\_rate}).
  \item Execute. Observe o espectro FM em torno de $f_c = 0$ (banda base).
  \item Meça a largura de banda visual para $f_m = 2000$\,Hz,
    $\Delta f = 5000$\,Hz ($\beta = 2.5$).
\end{enumerate}
\end{instrucao}

\subsection*{Etapa 2 -- Verificação da Regra de Carson (14--28 min)}

\begin{instrucao}[Medir largura de banda para diferentes $f_m$ e $\Delta f$]
\begin{enumerate}[label=\arabic*.]
  \item Crie variável \texttt{max\_dev} como \texttt{QT GUI Range}
    (min: 1000, max: 20000, passo: 1000, padrão: 5000).
  \item Preencha a Tabela~\ref{tab:carson} variando $f_m$ e $\Delta f$:
    \begin{itemize}
      \item Para cada combinação, registre a $B_T$ medida no espectro.
      \item Calcule $B_T$ pela Regra de Carson (Eq.~\eqref{eq:carson})
        e compare.
    \end{itemize}
\end{enumerate}
\end{instrucao}

\subsection*{Etapa 3 -- Demodulação FM (28--50 min)}

\begin{instrucao}[Construção do demodulador WBFM]
\begin{enumerate}[label=\arabic*.]
  \item Adicione \texttt{WBFM Receive}:
    \begin{itemize}
      \item Decimation = 10
    \end{itemize}
  \item Conecte: WBFM Transmit $\to$ Throttle $\to$ WBFM Receive
    $\to$ QT GUI Time Sink.
  \item Adicione um segundo canal no Time Sink para a mensagem original.
  \item Execute. Compare a mensagem original com a demodulada. A forma de
    onda deve ser praticamente idêntica.
  \item Mude $f_m$ e $\Delta f$ com os sliders e verifique que a recuperação
    continua fiel para diferentes valores de $\beta$.
  \item Para $f_m = 2\,000$\,Hz, meça no Time Sink e preencha a
    Tabela~\ref{tab:fidelidade} com os critérios objetivos para três valores
    de $\Delta f$. 
\end{enumerate}

\section{Perguntas para o relatório}

\begin{enumerate}[label=\textbf{Q\arabic*.}, leftmargin=*]

  \item Qual é a relação entre o desvio de frequência $\Delta f$ e o
    comportamento do sinal FM no domínio do tempo? O que muda visualmente
    quando $\Delta f$ aumenta?

  \item Com base na Tabela~\ref{tab:carson}, a Regra de Carson se confirma
    experimentalmente? Discuta as fontes de erro entre $B_T^\text{Carson}$
    e $B_T^\text{medida}$.

  \item Qual é a diferença entre FM de faixa estreita, NBFM com $\beta < 1$,
    e de faixa larga, WBFM com $\beta \gg 1$, em termos de espectro e
    complexidade de implementação?

\end{enumerate}
