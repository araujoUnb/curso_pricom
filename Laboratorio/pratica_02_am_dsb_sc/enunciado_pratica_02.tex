\section{Objetivos}

\begin{itemize}
  \item Implementar no GNU Radio a modulação AM DSB-SC (Double Sideband
    Suppressed Carrier) e verificar a ausência da portadora no espectro.
  \item Identificar as bandas laterais superior e inferior no espectro do
    sinal modulado.
  \item Realizar a demodulação coerente (síncrona) e avaliar o efeito de
    erros de fase do oscilador local.
  \item Determinar os parâmetros críticos do filtro passa-baixas para a
    recuperação fiel do sinal mensagem.
\end{itemize}

\section{Fundamentos teóricos (síntese)}

Na modulação \textbf{AM DSB-SC}, o sinal modulado é:
\begin{equation}
  s(t) = m(t)\cos(2\pi f_c t),
  \label{eq:dsbsc}
\end{equation}
onde $m(t)$ é o sinal mensagem e $f_c$ é a frequência da portadora.
Para uma mensagem senoidal $m(t) = A_m\cos(2\pi f_m t)$, o espectro resulta em:
\begin{equation}
  S(f) = \frac{A_m}{2}\bigl[\delta(f - (f_c - f_m)) + \delta(f - (f_c + f_m))\bigr]
         + \text{réplicas em } -f,
\end{equation}
mostrando \textbf{duas bandas laterais} em $f_c \pm f_m$ e
\textbf{sem componente na portadora} $f_c$.

A demodulação coerente multiplica $s(t)$ por $\cos(2\pi f_c t + \theta)$
e aplica um filtro passa-baixas (LPF):
\begin{equation}
  s(t)\cdot\cos(2\pi f_c t + \theta)
  = \frac{A_m}{2}\cos(\theta)\cos(2\pi f_m t)
  + \text{termos em } 2f_c \text{ (removidos pelo LPF)}.
\end{equation}
Um erro de fase $\theta \neq 0$ atenua o sinal recuperado pelo fator
$\cos\theta$; para $\theta = 90°$ o sinal é completamente suprimido.

O produto acima coloca energia em duas regiões: a mensagem em $f_m$ e um
termo de frequência dupla em $2f_c \pm f_m$.  O filtro passa-baixas precisa,
portanto, satisfazer simultaneamente
\begin{equation}
  f_m < f_\text{corte} < 2f_c - f_m,
  \label{eq:criterio-lpf}
\end{equation}
e é esse intervalo que a Etapa 4 explora experimentalmente.

\subsection*{Etapa 1 -- Modulador DSB-SC (0--12 min)}

\begin{instrucao}[Construção do modulador]
\begin{enumerate}[label=\arabic*.]
  \item Crie a variável \texttt{samp\_rate = 48000} e a variável
    \texttt{fc = 5000}.  Crie \texttt{fm} já como \texttt{QT GUI Range}
    (min: 200, max: 6000, passo: 200, padrão: 1000), porque ela será
    variada na Etapa 2.
  \item Adicione \textbf{Signal Source 1} (mensagem):
    Waveform = Cosine, Frequency = \texttt{fm}, Amplitude = 1.0.
  \item Adicione \textbf{Signal Source 2} (portadora):
    Waveform = Cosine, Frequency = \texttt{fc}, Amplitude = 1.0.
  \item Adicione um bloco \texttt{Multiply} com Type = Float e
    Num Inputs = 2.  Conecte as saídas dos dois \texttt{Signal Source} a ele.
  \item Adicione \texttt{Throttle} com Type = Float e conecte a saída do
    \texttt{Multiply}.
  \item Adicione um \texttt{QT GUI Time Sink} com Type = Float e
    Number of Inputs = 2.  Conecte \textbf{Signal Source 1} à entrada 1 e o
    \texttt{Throttle} à entrada 2.
  \item Adicione um \texttt{QT GUI Freq Sink} com Type = Float à saída do
    \texttt{Throttle}.
  \item Execute (\texttt{F6}). \textbf{Verifique:}
    \begin{itemize}
      \item No tempo: envoltória variando ao ritmo de $f_m$ (batimento).
      \item No espectro: duas componentes em $f_c - f_m = 4\,000\,\text{Hz}$
        e $f_c + f_m = 6\,000\,\text{Hz}$; \emph{sem componente em
        $f_c = 5\,000\,\text{Hz}$}.
    \end{itemize}
\end{enumerate}
\end{instrucao}

\subsection*{Etapa 2 -- Variação da frequência de mensagem (12--22 min)}

\begin{instrucao}[Observar movimento das bandas laterais]
\begin{enumerate}[label=\arabic*.]
  \item Execute. Deslize \texttt{fm} para 500, 1000, 2000 e 4000\,Hz.
  \item \textbf{Registre} na Tabela~2 a posição das bandas laterais em cada
    caso.  Confirme que estão em $f_c \pm f_m$.
  \item Continue subindo \texttt{fm} até 5000 e 6000\,Hz e observe o que
    acontece com a banda lateral inferior quando $f_m$ alcança e ultrapassa
    $f_c$.  Registre a observação para a questão Q2.
\end{enumerate}
\end{instrucao}

\subsection*{Etapa 3 -- Demodulador coerente (22--38 min)}

\begin{instrucao}[Construção do demodulador]
\begin{enumerate}[label=\arabic*.]
  \item Crie uma variável \texttt{phase\_deg} como \texttt{QT GUI Range}
    (min: 0, max: 90, passo: 5, padrão: 0).
  \item Converta para radianos: crie uma variável
    \texttt{phase\_rad = phase\_deg * 3.14159265 / 180}.
  \item Adicione um bloco \texttt{Import} com \texttt{import math} e crie
    duas variáveis de ganho:
    \texttt{ganho\_i = math.cos(phase\_rad)} e
    \texttt{ganho\_q = -math.sin(phase\_rad)}.
  \item Monte o oscilador local em quadratura, que realiza a identidade
    $\cos(2\pi f_c t + \theta) = \cos\theta\,\cos(2\pi f_c t)
      - \sin\theta\,\sin(2\pi f_c t)$:
    \begin{itemize}
      \item \textbf{Signal Source 3}: Waveform = Cosine, Frequency =
        \texttt{fc}, Phase Offset = 0, seguido de \texttt{Multiply Const}
        com IO Type = Float e constante \texttt{ganho\_i}.
      \item \textbf{Signal Source 4}: Waveform = Sine, Frequency =
        \texttt{fc}, Phase Offset = 0, seguido de \texttt{Multiply Const}
        com IO Type = Float e constante \texttt{ganho\_q}.
      \item Some as duas saídas com um bloco \texttt{Add} de Type = Float
        e Num Inputs = 2.
    \end{itemize}
  \item Adicione um segundo \texttt{Multiply}, também com Type = Float e
    Num Inputs = 2.  Conecte o sinal DSB-SC, tomado na saída do Throttle,
    e a saída do \texttt{Add}.
  \item Adicione um \texttt{Low Pass Filter}:
    \begin{itemize}
      \item Decimation = 1
      \item Cutoff Freq = \texttt{lpf\_cutoff}, criada como
        \texttt{QT GUI Range} (min: 200, max: 12000, passo: 100,
        padrão: 1500)
      \item Transition Width = 300\,Hz
      \item Window = Hamming
    \end{itemize}
  \item Altere o \texttt{QT GUI Time Sink} para Number of Inputs = 3 e
    conecte a saída do LPF à entrada 3.  O osciloscópio passa a mostrar
    a mensagem no canal 1, o sinal DSB-SC no canal 2 e a mensagem
    recuperada no canal 3.
  \item Adicione um segundo \texttt{QT GUI Freq Sink} com Type = Float à
    saída do LPF, para acompanhar o espectro do sinal demodulado.
  \item Execute. Compare a mensagem original, no canal 1, com a recuperada,
    no canal 3, quando \texttt{phase\_deg = 0}.
  \item Ajuste \texttt{phase\_deg} para 0°, 30°, 60° e 90°.  Em cada valor,
    meça a amplitude da mensagem recuperada no osciloscópio e registre na
    Tabela~3.
\end{enumerate}
\end{instrucao}

\subsection*{Etapa 4 -- Ajuste do LPF (38--46 min)}

\begin{instrucao}[Otimizar o filtro passa-baixas]
\begin{enumerate}[label=\arabic*.]
  \item Com \texttt{phase\_deg = 0}, teste os seguintes valores de
    \texttt{lpf\_cutoff}: 500, 1500, 3000, 9500 e 12000\,Hz.
  \item Observe no osciloscópio e no espectro as diferenças entre os casos.
    Registre em quais frequências há energia na saída do filtro.
  \item Preencha a Tabela~4 com as observações.
\end{enumerate}
\end{instrucao}

\subsection*{Etapa 5 -- Registro e conclusões (46--50 min)}

\begin{enumerate}[label=\arabic*.]
  \item Tire capturas de tela para: espectro do sinal modulado, espectro do
    sinal demodulado com \texttt{phase\_deg = 0}, forma de onda com erro
    de fase de 90° e saída demodulada com \texttt{lpf\_cutoff = 9500}\,Hz.
  \item Salve o flowgraph.
\end{enumerate}

\section{Perguntas para o relatório}

\begin{enumerate}[label=\textbf{Q\arabic*.}, leftmargin=*]

  \item Como o espectro do sinal DSB-SC confirmou a ausência da portadora?
    Explique em termos da equação~\eqref{eq:dsbsc} por que não há energia em
    $f_c$.

  \item Quais bandas laterais foram observadas? Como elas se relacionam
    matematicamente com $f_m$ e $f_c$? Usando os dados da Tabela~2,
    confirme a relação $f_\text{BL} = f_c \pm f_m$ e explique o que
    observou ao levar $f_m$ até 5000 e 6000\,Hz.

  \item O que ocorreu na saída demodulada quando foi introduzido erro de fase
    no oscilador local? Calcule a atenuação teórica para 30°, 60° e 90° e
    compare com o medido.

  \item A partir da Tabela~4, determine o intervalo de frequências de corte
    que recupera fielmente a mensagem e compare com o critério
    $f_m < f_\text{corte} < 2f_c - f_m$.  Descreva separadamente o que
    aconteceu com o corte abaixo de $f_m$ e com o corte acima de
    $2f_c - f_m$, indicando em qual frequência aparece o termo indesejado.

  \item Compare o sinal mensagem original com o sinal recuperado (com corte
    e fase ideais): houve distorções? Qual é o fator de escala esperado na
    saída do demodulador coerente?

\end{enumerate}
