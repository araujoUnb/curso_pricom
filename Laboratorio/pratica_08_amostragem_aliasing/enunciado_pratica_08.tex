\section{Objetivos}

\begin{itemize}
  \item Verificar experimentalmente o Teorema de Nyquist-Shannon: para
    reconstrução sem distorção a taxa de amostragem deve satisfazer
    $f_s \geq 2 f_\text{max}$.
  \item Observar o aliasing quando a condição de Nyquist é violada e
    calcular a frequência aparente do sinal subamostrado.
  \item Implementar um quantizador uniforme PCM e medir o SQNR no GNU
    Radio, confirmando a regra de $6{,}02$\,dB por bit.
  \item Relacionar número de bits, passo de quantização, potência do
    ruído e taxa de bits de um sistema PCM.
  \item Reconstruir o sinal com um segurador de ordem zero, ZOH, medir o
    droop $\text{sinc}$ e corrigi-lo com um equalizador inverse-sinc.
\end{itemize}

A prática cobre a cadeia completa de conversão analógico-digital vista no
Módulo 3: amostragem na Parte~I, quantização na Parte~II e reconstrução na
Parte~III.

\section{Fundamentos teóricos (síntese)}

\subsection*{Amostragem e aliasing}

Um sinal de banda limitada com frequência máxima $f_\text{max}$ pode ser
reconstruído sem distorção se amostrado com $f_s \geq 2 f_\text{max}$. Quando
$f_s < 2 f_\text{max}$ ocorre aliasing: o espectro se dobra em torno de
$f_s/2$. Para um tom de frequência $f_0$, a frequência aparente após a
amostragem é
\begin{equation}
  f_\text{alias} = \left|f_0 - n f_s\right|, \quad
  n = \left\lfloor\frac{f_0 + f_s/2}{f_s}\right\rfloor,
  \label{eq:alias}
\end{equation}
isto é, $f_\text{alias}$ é o reflexo de $f_0$ na frequência de Nyquist
$f_s/2$. O filtro anti-aliasing é um passa-baixas aplicado antes da
decimação, com frequência de corte $\leq f_s/2$. No GNU Radio a
subamostragem é feita pelo bloco \texttt{Keep 1 in N}, que mantém 1
amostra a cada $N$.

\subsection*{Quantização PCM}

Após a amostragem, a amplitude contínua é discretizada em $L = 2^n$ níveis
igualmente espaçados dentro da faixa $V_{pp} = V_\text{max}-V_\text{min}$. O
passo de quantização é
\begin{equation}
  \Delta = \frac{V_{pp}}{L} = \frac{V_{pp}}{2^n}.
  \label{eq:delta}
\end{equation}
O erro de quantização fica limitado a $[-\Delta/2,\,+\Delta/2]$ e, sob o
modelo uniforme, tem potência
\begin{equation}
  P_q = \frac{\Delta^2}{12}.
  \label{eq:pq}
\end{equation}
A razão sinal-ruído de quantização é $\text{SQNR}=P_s/P_q$. Para uma
senoide que ocupa todo o fundo de escala,
\begin{equation}
  \text{SQNR}_\text{dB} \approx 1{,}76 + 6{,}02\,n,
  \label{eq:sqnr}
\end{equation}
ou seja, cada bit adicional eleva o SQNR em $\approx 6$\,dB. Para atingir
um SQNR-alvo, o número mínimo de bits é
\begin{equation}
  n \geq \frac{\text{SQNR}_\text{dB} - 1{,}76}{6{,}02}.
  \label{eq:nbits}
\end{equation}
Por fim, a taxa de bits de um sistema PCM com $C$ canais é
\begin{equation}
  R = f_s \cdot n \cdot C \quad\text{[bits/s]}.
  \label{eq:taxa}
\end{equation}

\subsection*{Reconstrução com segurador de ordem zero}

Um conversor digital-analógico real não gera impulsos: ele segura cada
amostra durante $T_s = 1/f_s$, comportamento chamado de zero-order hold, ZOH.
A resposta em frequência do ZOH é
\begin{equation}
  |H_\text{ZOH}(f)| = \bigl|\text{sinc}(f\,T_s)\bigr|
  = \left|\frac{\sin(\pi f T_s)}{\pi f T_s}\right|,
  \label{eq:zoh}
\end{equation}
cujo envelope atenua as altas frequências, o chamado efeito de abertura. No
pior caso, na borda da banda $f = f_s/2$, a atenuação é
$|\text{sinc}(0{,}5)| = 2/\pi \approx -3{,}9$\,dB. Corrige-se esse droop com um
filtro de compensação de resposta proporcional a $1/\text{sinc}(f T_s)$ na
banda de passagem, o equalizador inverse-sinc. No GNU Radio o ZOH é o bloco
\texttt{Repeat}, que mantém cada amostra por $N$ períodos.

\begin{instrucao}[Montagem do flowgraph com Keep 1 in N]

  \item Crie \texttt{decim} como \texttt{QT GUI Range}, tipo Int, mínimo 2,

\subsection*{Etapa 3 -- Cálculo da frequência de aliasing}

\begin{instrucao}[Prever e confirmar o aliasing]
\begin{enumerate}[label=\arabic*.]
  \item Para os cenários C, E e F, calcule $f_\text{alias}$ pela
    Eq.~\eqref{eq:alias}. Exemplo do cenário C: $f_0 = 7000$, $f_s = 12000$,
    logo $f_\text{alias} = |7000 - 12000| = 5000$\,Hz.
  \item Compare o valor calculado com o pico observado e preencha a última
    coluna da Tabela~\ref{tab:nyquist}.
  \item No domínio do tempo, confirme que o sinal aparenta a frequência
    de alias e não a frequência original.
\end{enumerate}
\end{instrucao}

\subsection*{Etapa 4 -- Filtro anti-aliasing}

\begin{instrucao}[Prevenção de aliasing com passa-baixas]
\begin{enumerate}[label=\arabic*.]
  \item No cenário C, insira um \texttt{Low Pass Filter} antes do
    \texttt{Keep 1 in N}, com Cutoff $= f_{s,\text{ef}}/2 = 6000$\,Hz e
    Transition Width $= 500$\,Hz.
  \item Execute e verifique se o pico aliasado em 5000\,Hz desaparece.
  \item Repita para o cenário E, com $f_\text{sinal} = 2500$\,Hz e
    $f_{s,\text{ef}} = 4800$\,Hz. Observe que, como $2500 > 2400 = f_s/2$, o
    próprio sinal de interesse é atenuado pelo filtro.
\end{enumerate}
\end{instrucao}

\subsection*{Etapa 5 -- Sinal multitonal}

\begin{instrucao}[Aliasing com dois tons]
\begin{enumerate}[label=\arabic*.]
  \item Gere um sinal com dois tons, $f_1 = 1000$\,Hz e $f_2 = 7000$\,Hz,
    somando dois \texttt{Signal Source} com um \texttt{Add}.
  \item Aplique \texttt{decim = 4}, logo $f_{s,\text{ef}} = 12000$\,Hz.
  \item Verifique que o tom de 1000\,Hz é preservado e o de 7000\,Hz
    aparece em 5000\,Hz. Aplique o filtro anti-aliasing e compare.
\end{enumerate}
\end{instrucao}

\subsection*{Etapa 6 -- Montagem do quantizador e formas de onda}

\begin{instrucao}[Quantizador PCM no GNU Radio]
\begin{enumerate}[label=\arabic*.]
  \item Crie as variáveis \texttt{vmax = 1.0}, \texttt{f\_sinal = 997} e
    \texttt{ma\_len = 16384}.
  \item Crie \texttt{n\_bits} como \texttt{QT GUI Range}, tipo Int, mínimo 2,
    máximo 16, passo 1, padrão 4.
  \item Adicione \texttt{Signal Source}: Float, Cosine, freq = \texttt{f\_sinal},
    amplitude = \texttt{vmax}.
  \item Adicione um \texttt{Embedded Python Block} e cole o código do
    quantizador do Apêndice~\ref{ap:codigos}. Os parâmetros do bloco são
    \texttt{n\_bits} e \texttt{vmax}. A saída é o sinal quantizado
    $x_q[n] = \Delta\left(\lfloor x/\Delta\rfloor + 0{,}5\right)$.
  \item Adicione \texttt{Subtract} para formar o erro $e[n]=x[n]-x_q[n]$.
  \item Ligue original, quantizado e erro a um \texttt{QT GUI Time Sink} com
    3 entradas. Execute e varie \texttt{n\_bits} de 2 a 8, observando os
    degraus do sinal quantizado e a amplitude do erro.
\end{enumerate}
\end{instrucao}

\subsection*{Etapa 7 -- Medição do SQNR no GNU Radio}

\begin{instrucao}[Cadeia de medição de potência]
\begin{enumerate}[label=\arabic*.]

\section{Parte III -- Reconstrução: ZOH e equalização}

\begin{instrucao}[Segurador de ordem zero com o bloco Repeat]

  \item Adicione \texttt{Signal Source}: Float, Cosine, freq = \texttt{f0},

  \item Adicione \texttt{Repeat} com Interpolation = \texttt{N\_zoh}. A saída é
    a escada do ZOH na taxa \texttt{samp\_rate}.
  \item Ligue a um \texttt{QT GUI Time Sink} para ver a escada e a um
    \texttt{QT GUI Freq Sink} com bw = \texttt{samp\_rate}.
  \item Observe no espectro o droop $\text{sinc}$ na banda de passagem e as
    imagens em torno de múltiplos de \texttt{fs\_low}, atenuadas pelos nulos
    do $\text{sinc}$.

\end{instrucao}

\subsection*{Etapa 10 -- Equalização inverse-sinc}

\begin{instrucao}[Compensação do droop do ZOH]

  \item Compare os dois espectros: o equalizador achata a banda de passagem e
    aumenta a atenuação das imagens. Varie \texttt{f0} e confirme que o droop
    cresce com a frequência segundo a Eq.~\eqref{eq:zoh}.
\end{enumerate}
\end{instrucao}

  \item Pela Eq.~\eqref{eq:zoh}, calcule a atenuação do ZOH,

  \item Na borda da banda, $f = f_\text{low}/2 = 3000$\,Hz, qual é a atenuação

\end{enumerate}

Na Parte I, quando o sinal é decimado, o \texttt{QT GUI Freq Sink} do
caminho decimado deve usar a taxa efetiva $f_{s,\text{ef}} = $
\texttt{samp\_rate/decim}; caso contrário o eixo de frequência fica
incorreto. Na Parte II, o \texttt{Moving Average} estima a potência em uma
janela de \texttt{ma\_len} amostras; janelas muito curtas deixam a leitura
do SQNR ruidosa, enquanto janelas longas tornam a convergência mais lenta.
\end{atencao}
