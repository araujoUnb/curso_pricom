\section{Objetivos}

\begin{itemize}
  \item Construir cadeias completas de modulação-canal-demodulação para
    AM (DSB-SC) e FM com canal AWGN no GNU Radio.
  \item Variar o nível de ruído no canal e observar a degradação da
    qualidade do sinal recuperado.
  \item Comparar a robustez ao ruído dos demoduladores AM e FM para as
    mesmas condições de SNR de entrada.
  \item Identificar o \textbf{limiar de FM} (\textit{FM threshold}):
    abaixo do qual a vantagem do FM é perdida.
  \item Discutir a metodologia de comparação justa entre AM e FM.
\end{itemize}

\section{Fundamentos teóricos (síntese)}

Para um canal AWGN com densidade espectral de potência $N_0/2$, as
relações SNR de saída para os demoduladores são:

\textbf{AM DSB-SC (coerente):}
\begin{equation}
  \text{SNR}_{o,\text{AM}} = \text{SNR}_c = \frac{S_T}{N_0 W},
\end{equation}
onde $W$ é a largura de banda da mensagem e $S_T$ é a potência do sinal
transmitido.

\textbf{FM (WBFM):}
\begin{equation}
  \text{SNR}_{o,\text{FM}} = 3\beta^2(\beta + 1)\,\text{SNR}_c \gg \text{SNR}_{o,\text{AM}},
  \quad \text{para } \beta \gg 1,
\end{equation}
mostrando que o FM compra ganho de SNR à custa de maior largura de banda.

O \textbf{limiar de FM} ocorre quando o SNR de entrada cai abaixo de
$\approx 10$\,dB; abaixo deste ponto, o demodulador FM gera ruído de
impulsivo (cliques) e a vantagem é destruída.

\subsection*{Etapa 1 -- Cadeia AM com canal AWGN (0--15 min)}

\begin{instrucao}[Modulador e demodulador AM com ruído]
\begin{enumerate}[label=\arabic*.]
  \item Crie variáveis: \texttt{samp\_rate = 48000}, \texttt{fc = 10000},
    \texttt{fm = 1000}.
  \item Crie \texttt{noise\_sigma} como \texttt{QT GUI Range}
    (min: 0, max: 1.0, passo: 0.05, padrão: 0).
  \item \textbf{Modulador AM (DSB-SC):}
    \begin{itemize}
      \item Signal Source (msg, Cosine, fm) $\times$ Signal Source
        (portadora, Cosine, fc) = sinal modulado $s_\text{AM}(t)$.
    \end{itemize}
  \item \textbf{Canal AWGN:}
    Adicione Noise Source (Float, Gaussian, amplitude = \texttt{noise\_sigma})
    e \texttt{Add} para somar ao sinal AM.
  \item \textbf{Demodulador AM coerente:}
    \begin{itemize}
      \item Multiply: sinal recebido $\times$ portadora local.
      \item Low Pass Filter: cutoff = $1.5 \times f_m$.
    \end{itemize}
  \item Adicione \texttt{Throttle} e \texttt{QT GUI Time Sink} com 3 canais:
    mensagem original, sinal AM com ruído, mensagem recuperada.
  \item Execute. Com \texttt{noise\_sigma = 0}, verifique a recuperação fiel.
  \item Aumente gradualmente \texttt{noise\_sigma}: observe a degradação.
\end{enumerate}
\end{instrucao}

\subsection*{Etapa 2 -- Cadeia FM com canal AWGN (15--28 min)}

\begin{instrucao}[Modulador e demodulador FM com ruído]
\begin{enumerate}[label=\arabic*.]
  \item \textbf{Modulador FM:}
    \begin{itemize}
      \item Use a mesma mensagem (Signal Source, \texttt{fm = 1000}).
      \item Adicione \texttt{WBFM Transmit} (samp\_rate = 480000,
        audio\_rate = 48000, max\_dev = 10000).
    \end{itemize}
  \item \textbf{Canal AWGN:}
    Adicione Noise Source (Complex, Gaussian, amplitude = \texttt{noise\_sigma\_fm})
    e \texttt{Add}.
  \item \textbf{Demodulador FM:}
    \texttt{WBFM Receive} (decimação = 10).
  \item Conecte ao \texttt{QT GUI Time Sink}: mensagem original e
    mensagem FM recuperada.
  \item Execute. Compare o comportamento com o da cadeia AM.
\end{enumerate}
\end{instrucao}

\subsection*{Etapa 3 -- Comparação AM vs FM (28--42 min)}

\begin{instrucao}[Medições comparativas]
\begin{enumerate}[label=\arabic*.]
  \item Para garantir comparação justa, mantenha a mesma potência de
    sinal transmitido para AM e FM ($A_c = 1.0$).
  \item Preencha a Tabela~\ref{tab:snr} variando \texttt{noise\_sigma}:
    \begin{itemize}
      \item Para cada nível, qualifique visualmente a saída:
        (0) sem ruído, (1) ruído leve, (2) degradação moderada,
        (3) sinal irreconhecível.
    \end{itemize}
  \item \textbf{Limiar de FM:} identifique o valor de \texttt{noise\_sigma}
    acima do qual a vantagem de FM desaparece e o FM degrada mais do que o AM.
\end{enumerate}
\end{instrucao}

\begin{instrucao}[Testar FM com $\beta$ baixo e alto]

  \item Repita o experimento com \texttt{max\_dev = 2000} ($\beta = 2$) e

\begin{atencao}[Dica -- comparação justa]
Para comparar AM e FM, certifique-se de usar a mesma amplitude de mensagem
e o mesmo \texttt{noise\_sigma} em ambas as cadeias.  Note que a cadeia FM
opera em \texttt{samp\_rate = 480000} enquanto a AM pode operar em
48\,000\,Hz; ajuste o \texttt{noise\_sigma} para refletir a mesma
densidade espectral de potência de ruído ($N_0$).
\end{atencao}
