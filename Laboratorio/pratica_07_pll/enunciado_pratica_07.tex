\section{Objetivos}

\begin{itemize}
  \item Demodular FM com o bloco \texttt{PLL Frequency Detector} e comparar
    com o discriminador de quadratura da Prática 05.
  \item Medir a resposta em frequência do PLL e a frequência natural $f_n$
    prevista a partir de \texttt{loop\_bw}.
  \item Medir o sobressinal e o tempo de pico da resposta a um degrau de
    frequência.
  \item Medir o efeito do limite de frequência do VCO e o ganho de relação
    tom-piso que o laço dá sobre o discriminador de quadratura.
\end{itemize}

\section{Fundamentos teóricos}

\textbf{Equações do bloco.} A cada amostra $n$, o \texttt{PLL Frequency
Detector} compara a fase da entrada $x[n]$ com a fase $\theta[n]$ do seu
oscilador, o VCO, e atualiza a frequência $\nu[n]$ do VCO, em rad/amostra:
\begin{equation}
  e[n] = \arg x[n] - \theta[n], \qquad
  \nu[n+1] = \nu[n] + \beta\, e[n], \qquad
  \theta[n+1] = \theta[n] + \nu[n+1] + \alpha\, e[n].
  \label{eq:laco}
\end{equation}
O erro $e[n]$ é reduzido ao intervalo $[-\pi, \pi)$, e $\nu[n]$ é limitada a
$\pm$\texttt{Max Phase/sample}. A saída do bloco é $\nu[n]$. Os ganhos vêm de
\texttt{loop\_bw} $= w$, com amortecimento fixo $\zeta = \sqrt2/2$:
\begin{equation}
  \alpha = \frac{4\zeta w}{D}, \qquad \beta = \frac{4w^2}{D}, \qquad
  D = 1 + 2\zeta w + w^2.
  \label{eq:ganhos}
\end{equation}

\textbf{Frequência natural.} Para $w$ pequeno, o laço da
equação~\eqref{eq:laco} é o laço de 2.ª ordem da aula, com
$\omega_n = \sqrt{\beta}\,f_s$ em rad/s. Como $\beta \approx 4w^2$,
\begin{equation}
  f_n = \frac{\sqrt{\beta}\,f_s}{2\pi} \approx \frac{w\,f_s}{\pi}\ \text{Hz}.
  \label{eq:fn}
\end{equation}

\textbf{Resposta em frequência.} Se a frequência instantânea da entrada é
$\Delta f\, m(t)$, a saída $\nu$ segue essa frequência através do passa-baixas
\begin{equation}
  H(f) = \frac{1}{1 - (f/f_n)^2 + j\,2\zeta f/f_n}, \qquad
  |H(f)|^2 = \frac{1}{1 + (f/f_n)^4}\ \text{para}\ \zeta = \tfrac{\sqrt2}{2}.
  \label{eq:H}
\end{equation}
Multiplicada por $f_s/(2\pi\Delta f)$, a saída do PLL vale $m(t)$ filtrada por
$H$. O discriminador de quadratura não tem esse filtro. Por isso a diferença de
nível entre as duas saídas, no tom $f_m$, é
\begin{equation}
  A_\text{teo} = 10\log_{10}|H(f_m)|^2 = -10\log_{10}\!\big[1 + (f_m/f_n)^4\big]\ \text{dB},
  \label{eq:A}
\end{equation}
que vale $-3{,}01$\,dB em $f_m = f_n$.

\textbf{Resposta ao degrau.} Uma mensagem quadrada de $-1$ para $+1$ é um degrau
de frequência de $2\Delta f$. A saída normalizada vai de $-1$ a $+1$, passa do
valor final e volta. Para $\zeta = \sqrt2/2$, o sobressinal e o tempo de pico
são
\begin{equation}
  M_\text{teo} = 100\, e^{-\pi\zeta/\sqrt{1-\zeta^2}} = 100\,e^{-\pi} = 4{,}32\,\%,
  \qquad
  t_{p,\text{teo}} = \frac{\pi}{\omega_n\sqrt{1-\zeta^2}} = \frac{1}{\sqrt2\, f_n}.
  \label{eq:degrau}
\end{equation}
No Time Sink, com $y_\text{max}$ o valor máximo da curva do PLL depois da
subida e $t_\text{max}$ o instante desse máximo,
\begin{equation}
  M = 100\,\frac{y_\text{max} - 1}{2}\ \%, \qquad
  t_p = t_\text{max} - 1\ \text{ms},
  \label{eq:M}
\end{equation}
porque o degrau tem altura $2$ e o trigger coloca a subida em $t = 1$\,ms.

\textbf{Limite do VCO.} A frequência do VCO não passa de
$f_\text{lim} = 500$\,Hz. Se $\Delta f > f_\text{lim}$, a saída normalizada
satura em $f_\text{lim}/\Delta f$:
\begin{equation}
  y_\text{teo} = \min\!\left(1,\ \frac{f_\text{lim}}{\Delta f}\right).
  \label{eq:sat}
\end{equation}

\textbf{Ruído.} Em cada saída $X$, a relação tom-piso é
\begin{equation}
  R_X = L_X(f_m) - L_X(f_r)\ \text{dB}, \qquad f_r = 1\ \text{kHz},
  \label{eq:R}
\end{equation}
com $L_X(f)$ o nível lido no Frequency Sink. O ruído na saída do discriminador
de quadratura sai do PLL multiplicado por $|H(f)|^2$, e o tom por $|H(f_m)|^2$.
O ganho do PLL sobre a quadratura é então
\begin{equation}
  G = R_\text{PLL} - R_\text{Q}, \qquad
  G_\text{teo} = 10\log_{10}\frac{1 + (f_r/f_n)^4}{1 + (f_m/f_n)^4}\ \text{dB}.
  \label{eq:G}
\end{equation}

\begin{instrucao}[Leitura dos instrumentos]
No Frequency Sink, passe o mouse sobre o topo de cada componente para ler o
nível em dB e use só diferenças entre níveis. O eixo vai até 24\,kHz: arraste um retângulo com o mouse sobre a região de $0$ a $1{,}5$\,kHz para ampliar e ler os tons. No Time Sink, o trigger prende a
subida da mensagem em $t = 1$\,ms; passe o mouse sobre a curva para ler o
instante e o valor de cada ponto.
\end{instrucao}

\subsection*{Etapa 1 -- Montagem (0--10 min)}

\begin{instrucao}[Transmissor e dois demoduladores]
\begin{enumerate}[label=\arabic*.]
  \item Monte \texttt{Signal Source} $\to$ \texttt{Frequency Mod} $\to$
    \texttt{Throttle} $\to$ \texttt{Add}, com o \texttt{Noise Source} na
    segunda entrada, como na Prática 06.
  \item Ligue \texttt{Add} $\to$ \texttt{PLL Frequency Detector} $\to$
    \texttt{Multiply Const} e \texttt{Add} $\to$ \texttt{Quadrature Demod}.
  \item No Time Sink, ligue a mensagem no canal 0, o PLL no canal 1 e a
    quadratura no canal 2. No Frequency Sink, o PLL na entrada 0 e a quadratura
    na entrada 1.
  \item Execute com os valores padrão. As três curvas do Time Sink devem
    coincidir, com amplitude $1$.
\end{enumerate}
\end{instrucao}

\section{Perguntas para o relatório}

Todas as respostas devem ser numéricas, com as contas mostradas.

\begin{enumerate}[label=\textbf{Q\arabic*.}, leftmargin=*]

  \item Calcule $\alpha$, $\beta$ e $f_n$ para \texttt{loop\_bw} $= 0{,}02$
    pelas equações~\eqref{eq:ganhos} e~\eqref{eq:fn}, com $\sqrt\beta$ e com a
    aproximação $w f_s/\pi$. Qual o erro da aproximação, em \%?

  \item Na Tabela~\ref{tab:H}, interpole linearmente entre as duas linhas
    vizinhas de $A = -3{,}01$\,dB e dê o $f_m$ medido em que $A = -3{,}01$\,dB.
    Compare com $f_n$ da Q1. Qual a maior diferença $|A - A_\text{teo}|$ da
    tabela?

  \item Na Tabela~\ref{tab:degrau}, compare $M$ com $M_\text{teo} = 4{,}32\,\%$
    e $t_p$ com $t_{p,\text{teo}}$ em cada linha. Calcule o produto
    $t_p\cdot$\texttt{loop\_bw} nas três linhas: pela
    equação~\eqref{eq:degrau}, ele deveria valer $\pi/(\sqrt2 f_s)$. Quanto vale
    essa constante em ms?

  \item Na Tabela~\ref{tab:sat}, a partir de qual $\Delta f$ a saída deixa de
    chegar a $1$? Qual a maior diferença $|y_\text{pico} - y_\text{teo}|$?

  \item Na Tabela~\ref{tab:ruido}, calcule $G_\text{teo}$ e a diferença
    $G - G_\text{teo}$ em cada linha.

  \item Uma mensagem de $f_m = 300$\,Hz deve passar pelo PLL com no máximo
    $1$\,dB de atenuação. Pela equação~\eqref{eq:A}, qual o menor $f_n$ e o
    menor \texttt{loop\_bw} que atendem? Com esse \texttt{loop\_bw}, quanto
    vale $G_\text{teo}$ para $f_m = 300$\,Hz e $f_r = 1$\,kHz?

\end{enumerate}

\begin{atencao}[Atenção -- unidades do PLL]
No bloco \texttt{PLL Frequency Detector}, \textit{Loop Bandwidth},
\textit{Max Phase/sample} e \textit{Min Phase/sample} são dados em radianos por
amostra, não em Hz. A conversão é $\omega = 2\pi f/f_s$.
\end{atencao}
