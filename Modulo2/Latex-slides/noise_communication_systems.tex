\section{Por que estudar ruído?}

\begin{frame}{O problema}

Todo canal real adiciona \textbf{ruído} ao sinal transmitido. O receptor recupera uma versão corrompida da mensagem.

\vspace{0.3cm}

\begin{center}
\begin{tikzpicture}[node distance=1.3cm, font=\small,
  block/.style={rectangle, draw, rounded corners, minimum height=0.9cm, minimum width=1.8cm, align=center},
  sum/.style={circle, draw, inner sep=1pt}]
\node[block] (tx) {Transmissor};
\node[block, right=of tx] (ch) {Canal};
\node[sum, right=0.9cm of ch] (sum) {$+$};
\node[block, right=0.9cm of sum] (rx) {Receptor};
\node[below=0.4cm of sum] (n) {$n(t)$};
\draw[->] (tx) -- (ch);
\draw[->] (ch) -- (sum);
\draw[->] (sum) -- (rx);
\draw[->] (n) -- (sum);
\node[left=0.05cm of tx] {$m(t)$};
\node[right=0.05cm of rx] {$\hat m(t)$};
\end{tikzpicture}
\end{center}

\textbf{Perguntas centrais deste capítulo:}
\begin{itemize}
\item Como medir a qualidade do sinal recuperado?
\item Cada esquema de modulação reage da mesma forma ao ruído?
\item É possível trocar banda por imunidade a ruído? E vale a pena?
\end{itemize}

\end{frame}

% ============================================

\begin{frame}{De onde vem o ruído?}

Ruído é qualquer perturbação aleatória que se soma ao sinal e que o receptor não consegue prever. As fontes mais comuns são:

\begin{itemize}
\item ruído térmico: agitação dos elétrons em resistores, cabos e componentes acima do zero absoluto;
\item ruído shot: flutuações na corrente em dispositivos semicondutores e válvulas;
\item ruído de cintilação, do tipo $1/f$: dominante em baixas frequências em circuitos eletrônicos;
\item ruído atmosférico: descargas elétricas e fenômenos naturais, importantes em HF;
\item ruído cósmico e galáctico: radiação captada pela antena, relevante em micro-ondas;
\item interferência de outros transmissores e equipamentos.
\end{itemize}

\end{frame}

% ============================================

\begin{frame}{Modelo agregado de ruído}

Em vez de tratar cada fonte separadamente, somamos todas as contribuições em um único termo aleatório $n(t)$ visto na entrada do receptor.

\vspace{0.3cm}

Esse termo agregado tem três propriedades que serão usadas ao longo do capítulo:

\begin{itemize}
\item é aditivo: aparece somado ao sinal recebido;
\item tem média nula e é estatisticamente estacionário no intervalo de interesse;
\item é aproximadamente branco na banda do receptor, com densidade espectral de potência constante.
\end{itemize}

\vspace{0.3cm}

Esse modelo é conhecido como ruído branco gaussiano aditivo, ou AWGN, e será nossa hipótese de trabalho.

\end{frame}

% ============================================

\section{Ruído em sistema banda base}

\begin{frame}{Modelo de canal AWGN}

Sinal transmitido em \textbf{banda base}: $m(t)$ com banda $[-W, W]$.

\textbf{Sinal recebido:}
\[
r(t) = m(t) + n(t)
\]
$n(t)$: ruído branco gaussiano aditivo, PSD bilateral $S_n(f) = N_0/2$ em W/Hz.

\vspace{0.3cm}

\textbf{Receptor:} filtro passa-baixas ideal com largura de banda $W$.

\vspace{0.3cm}

\end{tikzpicture}
\end{center}

\end{frame}

% ============================================

\begin{frame}{De onde vem a potência do ruído?}

\textbf{Passo 1.} O ruído branco tem PSD bilateral constante:
\[
S_n(f) = \frac{N_0}{2}, \qquad \forall\, f \in \mathbb{R}
\]
Potência total seria infinita — todo sistema real tem filtro.

\textbf{Passo 2.} Após filtro passa-baixas ideal $|H(f)| = 1$ em $|f| \le W$, zero fora:
\[
S_{n,\text{out}}(f) = |H(f)|^2 S_n(f) = \begin{cases} N_0/2, & |f| \le W \\ 0, & \text{caso contrário} \end{cases}
\]

\textbf{Passo 3.} Potência = integral da PSD (Wiener-Khinchin):
\[
P_{\text{noise}} = \int_{-\infty}^{\infty} S_{n,\text{out}}(f)\, df = \int_{-W}^{W} \frac{N_0}{2}\, df = N_0 W
\]

\end{frame}

% ============================================

\end{frame}

% ============================================

\begin{frame}{Potências na saída}

\textbf{Sinal na saída:} $m(t)$ passa intacto pelo filtro $\Rightarrow$ potência $P_m$.

\textbf{Ruído na saída:} pela derivação anterior,
\[
P_{\text{noise}} = \int_{-W}^{W} \frac{N_0}{2}\, df = N_0 W
\]

\textbf{SNR de saída:}
\[
\left(\frac{S}{N}\right)_o = \frac{P_m}{N_0 W}
\]

\end{frame}

% ============================================

\begin{frame}{SNR de referência $\gamma$}

\begin{block}{Definição}
Como em banda base $P_r = P_m$, temos
\[
\left(\frac{S}{N}\right)_o = \gamma.
\]
Esta é a SNR de referência contra a qual mediremos todas as modulações.
\end{block}

\vspace{0.3cm}

Todas as fórmulas $(S/N)_o = \gamma$, $\eta\gamma$, $\tfrac{3}{2}\beta^2 \gamma$ que veremos a seguir usam esse mesmo $\gamma$.

\end{frame}

% ============================================

\begin{frame}{Aplicação: SNR de um sistema de voz}

\textbf{Enunciado.} Um sinal de voz com potência $P_m = 10$ mW e banda $W = 4$ kHz é transmitido em banda base por um canal AWGN com $N_0 = 10^{-12}$ W/Hz. Determine $(S/N)_o$ em dB.

% ============================================

\begin{frame}{Resolução: potências}

\textbf{Passo 1.} Potência de ruído após o filtro:
\[
P_{\text{noise}} = N_0 W = 10^{-12} \times 4 \times 10^{3} = 4 \text{ nW}
\]

\textbf{Passo 2.} SNR linear:
\[
\left(\frac{S}{N}\right)_o = \frac{P_m}{P_{\text{noise}}} = \frac{10 \times 10^{-3}}{4 \times 10^{-9}} = 2{,}5 \times 10^{6}
\]

\end{frame}

% ============================================

\begin{frame}{Resolução: conversão em dB}

\textbf{Passo 3.} Conversão para decibéis:
\[
\left(\frac{S}{N}\right)_o\Big|_{\text{dB}} = 10\log_{10}(2{,}5 \times 10^{6}) \approx 64 \text{ dB}
\]

\begin{block}{Interpretação}
Em condições ideais, a SNR de banda base é $\gamma = 64$ dB. Cada dB de perda no canal reduz esse valor diretamente.
\end{block}

\end{frame}

\section{Ruído em DSB-SC}

\begin{frame}{Sinal recebido em DSB-SC}

\textbf{Sinal transmitido:}
\[
u(t) = A_c\, m(t)\cos(2\pi f_c t), \quad \text{banda } 2W \text{ em torno de } f_c.
\]

\vspace{0.2cm}

\textbf{Sinal recebido}, com o ruído passa-banda em quadratura:
\[
r(t) = A_c m(t)\cos(2\pi f_c t) + n_c(t)\cos(2\pi f_c t) - n_s(t)\sin(2\pi f_c t).
\]

\vspace{0.2cm}

O receptor tem filtro passa-banda de largura $B_T = 2W$ centrado em $f_c$. O ruído após esse filtro tem PSD bilateral $N_0/2$ na faixa $|f-f_c|<W$ e potência total:
\[
P_n = \int_{-\infty}^{\infty} S_n(f)\,df = \tfrac{N_0}{2}\cdot 4W = 2 W N_0.
\]
As componentes em quadratura têm $\sigma^2_{n_c} = \sigma^2_{n_s} = P_n = 2 W N_0$.

\end{frame}

% ============================================

\begin{frame}{Demodulação coerente}

Multiplicar $r(t)$ por $\cos(2\pi f_c t)$ e filtrar em $W$:
\begin{align*}
r(t)\cos(2\pi f_c t)
&= \tfrac{1}{2} A_c m(t) + \tfrac{1}{2} A_c m(t)\cos(4\pi f_c t) \\
&\quad + \tfrac{1}{2} n_c(t) + \tfrac{1}{2}\bigl[n_c(t)\cos(4\pi f_c t) - n_s(t)\sin(4\pi f_c t)\bigr]
\end{align*}

O filtro passa-baixas rejeita os termos em $2f_c$:
\[
\boxed{y(t) = \tfrac{1}{2}\bigl[A_c m(t) + n_c(t)\bigr]}
\]

\vspace{0.2cm}

A componente em quadratura $n_s(t)$ é eliminada; apenas $n_c(t)$ aparece na saída.

\end{frame}

% ============================================

\begin{frame}{Potências na saída}

Da saída $y(t) = \frac{1}{2}[A_c m(t) + n_c(t)]$:
\begin{itemize}
\item sinal: $P_o = \dfrac{A_c^2}{4} P_M$;
\item ruído: $P_{n_o} = \dfrac{1}{4} P_{n_c} = \dfrac{1}{4} P_n = \dfrac{W N_0}{2}$.
\end{itemize}

\vspace{0.2cm}

Razão direta:
\[
\left(\frac{S}{N}\right)_o = \frac{P_o}{P_{n_o}} = \frac{A_c^2 P_M / 4}{W N_0 / 2} = \frac{A_c^2 P_M}{2 W N_0}.
\]

\end{frame}

% ============================================

\begin{frame}{Resultado em função de $\gamma$}

A potência recebida do sinal DSB-SC é $P_R = A_c^2 P_M / 2$, logo $A_c^2 P_M = 2 P_R$.

\begin{block}{Resultado}
\[
\left(\frac{S}{N}\right)_{o,\text{DSB}} = \frac{2 P_R}{2 W N_0} = \frac{P_R}{W N_0} = \gamma
\]
A SNR de saída do DSB-SC coerente é igual à SNR do sistema banda base de referência: não há ganho, mas também não há perda.
\end{block}

\vspace{0.2cm}

A demodulação coerente rejeita $n_s(t)$, garantindo desempenho equivalente ao banda base apesar de o ruído passa-banda ter o dobro da potência total.

\end{frame}

% ============================================

\begin{frame}{Aplicação: enlace DSB-SC com perda de canal}

\textbf{Enunciado.} Transmissor DSB-SC com $A_c = 10$ V, mensagem normalizada com $P_m = 0{,}5$, $f_c = 1$ MHz, banda $W = 4$ kHz. O enlace tem perda total $L = 60$ dB e o receptor opera com $N_0 = 10^{-15}$ W/Hz. Calcule $(S/N)_o$ em dB.

\begin{frame}{Resolução: potências}

\textbf{Passo 1.} Potência transmitida:
\[
P_t = \frac{A_c^2 P_m}{2} = \frac{100 \cdot 0{,}5}{2} = 25 \text{ W}
\]

\textbf{Passo 2.} Potência recebida após $L = 10^6$:
\[
P_r = \frac{P_t}{L} = \frac{25}{10^{6}} = 25 \text{ } \mu\text{W}
\]

\end{frame}

% ============================================

\begin{frame}{Resolução: SNR final}

\textbf{Passo 3.} SNR de referência:
\[
\gamma = \frac{P_r}{N_0 W} = \frac{25 \times 10^{-6}}{10^{-15} \times 4 \times 10^{3}} \approx 6{,}25 \times 10^{6} \approx 68 \text{ dB}
\]

\textbf{Passo 4.} SNR de saída em DSB-SC coerente:
\[
\left(\frac{S}{N}\right)_o = \gamma \approx 68 \text{ dB}
\]

\end{frame}

\section{Ruído em AM convencional}

\begin{frame}{Sinal AM e ruído na entrada}

\textbf{Sinal AM:}
\[
u(t) = A_c\bigl[1 + \mu\, m_n(t)\bigr]\cos(2\pi f_c t),
\]
com $m_n$ normalizada, $|m_n| \le 1$, e índice de modulação $\mu$.

\vspace{0.2cm}

\textbf{Sinal recebido}, com o ruído passa-banda em quadratura:
\[
r(t) = \bigl[A_c(1 + \mu m_n(t)) + n_c(t)\bigr]\cos(2\pi f_c t) - n_s(t)\sin(2\pi f_c t).
\]

\vspace{0.2cm}

Estrutura idêntica à do DSB-SC, trocando $m(t)$ por $1 + \mu m_n(t)$. Usaremos demodulação síncrona; o detector de envelope, em alta SNR, dá o mesmo resultado.

\vspace{0.2cm}

Banda passante do receptor: $B_T = 2W$, portanto $\sigma^2_{n_c} = \sigma^2_{n_s} = 2 N_0 W$.

\end{frame}

% ============================================

\begin{frame}{Demodulação síncrona}

Multiplicando $r(t)$ por $\cos(2\pi f_c t)$ e filtrando em $W$:
\[
y_1(t) = \tfrac{1}{2}\bigl[A_c\bigl(1 + \mu m_n(t)\bigr) + n_c(t)\bigr].
\]

A componente $n_s(t)$ é rejeitada, como no DSB-SC.

\vspace{0.2cm}

O bloco de DC remove o termo constante $A_c/2$:
\[
\boxed{y(t) = \tfrac{1}{2} A_c\, \mu\, m_n(t) + \tfrac{n_c(t)}{2}}
\]

\end{frame}

% ============================================

\begin{frame}{Potências e potência recebida}

Da saída $y(t) = \tfrac{1}{2} A_c \mu\, m_n(t) + \tfrac{n_c(t)}{2}$:
\begin{itemize}
\item sinal: $P_{\text{sig}} = \tfrac{1}{4} A_c^2 \mu^2 P_{m_n}$;
\item ruído: $P_{\text{noise}} = \tfrac{1}{4} P_{n_c} = \tfrac{1}{4}\cdot 2 N_0 W = \tfrac{N_0 W}{2}$.
\end{itemize}

\vspace{0.2cm}

A potência total recebida do sinal AM, supondo $m_n$ de média nula:
\[
P_r = \tfrac{A_c^2}{2}\bigl[1 + \mu^2 P_{m_n}\bigr].
\]

\end{frame}

% ============================================

\begin{frame}{Derivação da SNR de saída em AM}

\begin{align*}
\left(\frac{S}{N}\right)_{o,\text{AM}}
&= \frac{\tfrac{1}{4} A_c^2 \mu^2 P_{m_n}}{\tfrac{1}{4} P_{n_c}}
 = \frac{A_c^2 \mu^2 P_{m_n}}{2 N_0 W} \\[3pt]
&= \frac{\mu^2 P_{m_n}}{1+\mu^2 P_{m_n}} \cdot \frac{\tfrac{A_c^2}{2}\bigl[1+\mu^2 P_{m_n}\bigr]}{N_0 W} \\[3pt]
&= \frac{\mu^2 P_{m_n}}{1+\mu^2 P_{m_n}} \cdot \frac{P_r}{N_0 W} \\[3pt]
&= \eta\,\gamma
\end{align*}

com $\eta = \dfrac{\mu^2 P_{m_n}}{1+\mu^2 P_{m_n}}$, a eficiência de potência do AM.

\end{frame}

% ============================================

\begin{frame}{Limite superior da eficiência}

Para tom único $m_n(t) = \cos(2\pi f_m t)$ temos $P_{m_n} = 1/2$. O máximo de $\eta$ ocorre em $\mu = 1$:
\[
\eta_{\max} = \frac{1/2}{1 + 1/2} = \frac{1}{3}
\]

\begin{block}{Resultado}
$(S/N)_o \le \gamma/3$ no melhor caso. AM convencional é cerca de $4{,}77$ dB pior que banda base. O preço da simplicidade do detector de envelope é uma SNR pior.
\end{block}

\end{frame}

% ============================================

\begin{frame}{Detector de envelope com ruído}

A partir do sinal recebido em (6.1.29),
\[
r(t) = \bigl[A_c(1+\mu m_n(t)) + n_c(t)\bigr]\cos\omega_c t - n_s(t)\sin\omega_c t,
\]
o detector de envelope extrai
\[
V_r(t) = \sqrt{\bigl[A_c(1+\mu m_n(t)) + n_c(t)\bigr]^2 + n_s^2(t)}.
\]

\vspace{0.2cm}

Expressão não linear no ruído. Analisamos dois regimes: alta SNR na entrada e baixa SNR.

\end{frame}

% ============================================

\begin{frame}{Regime de alta SNR}

Quando $A_c\bigl(1 + \mu m_n(t)\bigr) \gg \sqrt{n_c^2(t) + n_s^2(t)}$ com alta probabilidade, vale
\[
V_r(t) \approx A_c\bigl(1 + \mu m_n(t)\bigr) + n_c(t).
\]
Após o bloqueio de DC,
\[
y(t) = A_c\,\mu\, m_n(t) + n_c(t),
\]
mesma estrutura do receptor síncrono, sem o coeficiente $1/2$. Esse coeficiente cancela na razão SNR.

\vspace{0.2cm}

\begin{block}{Conclusão}
No regime de alta SNR, detector de envelope e demodulador síncrono entregam a mesma $(S/N)_o = \eta\gamma$.
\end{block}

\end{frame}

% ============================================

\begin{frame}{Regime de baixa SNR: derivação}

Expandindo o quadrado dentro da raiz em $V_r(t)$:
\[
V_r = \sqrt{A_c^2(1+\mu m_n)^2 + n_c^2 + n_s^2 + 2 A_c n_c (1+\mu m_n)}.
\]

\textbf{Passo 1.} Hipótese de baixa SNR: $A_c^2(1+\mu m_n)^2 \ll n_c^2 + n_s^2$. Descartamos o termo quadrático em $A_c$:
\[
V_r \approx \sqrt{(n_c^2 + n_s^2)\left[1 + \frac{2 A_c\, n_c}{n_c^2 + n_s^2}(1+\mu m_n)\right]}.
\]

\textbf{Passo 2.} Definindo $V_n(t) = \sqrt{n_c^2 + n_s^2}$ e usando $\sqrt{1+\epsilon} \approx 1 + \epsilon/2$ para $\epsilon$ pequeno:
\[
V_r \approx V_n\left[1 + \frac{A_c\, n_c}{V_n^2}(1+\mu m_n)\right] = V_n(t) + \frac{A_c\, n_c(t)}{V_n(t)}\bigl(1 + \mu m_n(t)\bigr).
\]

\end{frame}

% ============================================

\begin{frame}{Efeito de limiar}

Da expressão derivada,
\[
V_r(t) \approx V_n(t) + \frac{A_c\, n_c(t)}{V_n(t)}\bigl(1 + \mu m_n(t)\bigr),
\]
o sinal $m_n(t)$ aparece multiplicado por $n_c(t)/V_n(t)$, não somado. Sinal e ruído deixam de ser aditivos e a SNR perde sentido.

\vspace{0.2cm}

\begin{block}{Limiar do detector de envelope}
Abaixo de uma SNR de entrada da ordem de $6$--$10$ dB, a aproximação linear quebra; a $(S/N)_o$ cai muito mais rápido do que $\eta\gamma$. Em enlaces fracos, prefere-se demodulação coerente mesmo no AM convencional.
\end{block}

\end{frame}

% ============================================

\begin{frame}{Aplicação: eficiência e SNR em AM tom único}

\textbf{Enunciado.} Um transmissor AM emite uma portadora modulada por um tom único com índice $\mu = 0{,}7$. O receptor com detector de envelope mede $\gamma = 40$ dB na entrada. Determine a eficiência $\eta$ e $(S/N)_o$ em dB.

\begin{frame}{Resolução: eficiência}

\textbf{Passo 1.} Para tom único, $P_{m_n} = 1/2$. Eficiência:
\[
\eta = \frac{\mu^2 P_{m_n}}{1 + \mu^2 P_{m_n}} = \frac{0{,}49 \cdot 0{,}5}{1 + 0{,}49 \cdot 0{,}5} = \frac{0{,}245}{1{,}245} \approx 0{,}197
\]

\textbf{Passo 2.} Eficiência em dB:
\[
10\log_{10}(0{,}197) \approx -7{,}06 \text{ dB}
\]

\end{frame}

% ============================================

\begin{frame}{Resolução: SNR de saída}

\textbf{Passo 3.} SNR de saída:
\[
\left(\frac{S}{N}\right)_o\Big|_{\text{dB}} = \eta\big|_{\text{dB}} + \gamma\big|_{\text{dB}} \approx -7{,}06 + 40 \approx 33 \text{ dB}
\]

\begin{block}{Interpretação}
Cerca de $80\%$ da potência transmitida fica na portadora e não contribui para a SNR útil. Por isso o AM convencional perde para o DSB-SC e o SSB.
\end{block}

\end{frame}

\section{Ruído em SSB AM}

\begin{frame}{Sinal recebido em SSB}

\textbf{Sinal SSB:}
\[
u(t) = A_c m(t)\cos(2\pi f_c t) \mp A_c \hat m(t)\sin(2\pi f_c t),
\]
com $\hat m(t)$ a transformada de Hilbert de $m(t)$ e largura $W$ em torno de $f_c$.

\vspace{0.2cm}

\textbf{Sinal recebido}, agrupando o ruído em quadratura:
\[
r(t) = \bigl(A_c m(t) + n_c(t)\bigr)\cos(2\pi f_c t) + \bigl(\mp A_c \hat m(t) - n_s(t)\bigr)\sin(2\pi f_c t).
\]

\vspace{0.2cm}

O filtro do receptor tem largura $W$. A potência total do ruído passa-banda é
\[
P_n = \int_{-\infty}^{\infty} S_n(f)\,df = \tfrac{N_0}{2}\cdot 2W = W N_0,
\]
com $\sigma^2_{n_c} = \sigma^2_{n_s} = P_n = W N_0$, metade do valor visto no DSB-SC.

\end{frame}

% ============================================

\begin{frame}{Demodulação coerente em SSB}

Multiplicando $r(t)$ por $\cos(2\pi f_c t)$ e filtrando em $W$, o termo em $\sin(2\pi f_c t)$ é rejeitado e os termos em $2f_c$ desaparecem. Sobra apenas a componente em fase com coeficiente $1/2$:
\[
\boxed{y(t) = \tfrac{A_c}{2} m(t) + \tfrac{1}{2} n_c(t)}
\]

\vspace{0.2cm}

A componente $n_s(t)$ é eliminada, mesma estrutura do DSB-SC.

\end{frame}

% ============================================

\begin{frame}{Potências na saída}

Da saída $y(t) = \tfrac{A_c}{2} m(t) + \tfrac{1}{2} n_c(t)$:
\begin{itemize}
\item sinal: $P_o = \dfrac{A_c^2}{4} P_M$;
\item ruído: $P_{n_o} = \dfrac{1}{4} P_{n_c} = \dfrac{W N_0}{4}$.
\end{itemize}

\vspace{0.2cm}

Razão direta:
\[
\left(\frac{S}{N}\right)_o = \frac{P_o}{P_{n_o}} = \frac{A_c^2 P_M / 4}{W N_0 / 4} = \frac{A_c^2 P_M}{W N_0}.
\]

\end{frame}

% ============================================

\begin{frame}{Potência recebida no SSB}

Da média temporal de $u^2(t)$, com $\langle\cos^2\rangle = \langle\sin^2\rangle = 1/2$, $\langle\sin\cos\rangle = 0$ e $P_{\hat m} = P_M$:
\[
P_R = \langle u^2\rangle = A_c^2\bigl[\tfrac{P_M}{2} + \tfrac{P_{\hat m}}{2}\bigr] = A_c^2 P_M.
\]

\vspace{0.2cm}

Diferentemente do DSB-SC, onde $P_R = A_c^2 P_M/2$, no SSB toda a potência fica em uma única banda lateral.

\end{frame}

% ============================================

\begin{frame}{Resultado em função de $\gamma$}

Substituindo $A_c^2 P_M = P_R$ na SNR de saída:

\begin{block}{Resultado}
\[
\left(\frac{S}{N}\right)_{o,\text{SSB}} = \frac{P_R}{W N_0} = \gamma
\]
SSB coerente alcança a mesma SNR do sistema banda base, usando metade da banda do DSB-SC.
\end{block}

\vspace{0.2cm}

Para um mesmo $P_R$, SSB e DSB-SC entregam exatamente a mesma SNR de saída; SSB economiza apenas no recurso de banda. O preço a pagar é a complexidade do receptor coerente.

\end{frame}

% ============================================

\begin{frame}{Aplicação: SSB versus DSB-SC}

\textbf{Enunciado.} Considere um enlace SSB com o mesmo $P_R = 25\,\mu$W do exemplo DSB-SC anterior, $W = 4$ kHz e $N_0 = 10^{-15}$ W/Hz. Compare a banda ocupada e a SNR de saída com o caso DSB-SC.

SSB ocupa metade da banda. Como a SNR se compara?

\end{frame}

\begin{frame}{Resolução passo a passo}

\textbf{Passo 1.} SNR de referência com $P_R = 25\,\mu$W:
\[

DSB-SC: $68$ dB com $8$ kHz. SSB: $68$ dB com $4$ kHz. Mesma SNR, metade da banda.
\end{block}

\end{frame}

\section{Ruído em modulação angular (FM)}

\begin{frame}{Sinal FM com ruído}

\textbf{Sinal FM transmitido:}
\[
s(t) = A_c \cos\!\left[2\pi f_c t + \phi(t)\right], \qquad \phi(t) = 2\pi k_f \int_{-\infty}^{t} m(\tau)\, d\tau
\]

\textbf{Sinal recebido:}
\[
r(t) = A_c \cos[2\pi f_c t + \phi(t)] + n(t)
\]

Decompondo o ruído em quadratura,
\[
n(t) = n_c(t)\cos(2\pi f_c t) - n_s(t)\sin(2\pi f_c t),
\]
e reescrevendo $r(t)$ em envelope e fase:
\[
r(t) = E(t)\cos\!\left[2\pi f_c t + \psi(t)\right].
\]

\end{frame}

% ============================================

\begin{frame}{Aproximação para alta SNR}

\begin{center}
\begin{tikzpicture}[font=\scriptsize, scale=0.9]
\draw[->] (-0.3,0) -- (4.5,0) node[right] {Re};
\draw[->] (0,-0.3) -- (0,2.4) node[above] {Im};
\draw[->, thick, blue] (0,0) -- (3.5,1.0) node[midway, below right] {$A_c$};
\draw[->, thick, red] (3.5,1.0) -- (3.9,1.9) node[midway, right] {ruído};
\draw[->, thick, black] (0,0) -- (3.9,1.9);
\node at (4.3,2.0) {$\vec r$};
\draw[dashed] (3.5,1.0) -- (3.5,1.9);
\draw[dashed] (3.5,1.0) -- (3.9,1.0);
\node[right] at (3.9,1.45) {$n_s$};
\node[below] at (3.7,1.0) {$n_c$};
% Ângulo phi: do eixo Re até A_c (fase do sinal)
\draw[->, blue!60!black] (0.8,0) arc (0:16:0.8);
\node[blue!60!black] at (0.95,0.1) {$\phi$};
% Ângulo Y_n: de A_c até r (perturbação de fase)
\draw[->, black] (2.31,0.66) arc (16:26:2.4);
\node at (2.55,0.95) {$Y_n$};
% Angulo Phi_n - phi: entre A_c e o vetor ruido (no ponto onde o ruido sai de A_c)
\draw[->, red!70!black] (4.08,1.17) arc (16:66:0.6);
\node[red!70!black] at (3.95,1.42) {$\Phi_n{-}\phi$};
\end{tikzpicture}
\end{center}

Escrevendo o ruído em envelope e fase, $n(t) = V_n(t)\cos[2\pi f_c t + \Phi_n(t)]$, com $V_n = \sqrt{n_c^2 + n_s^2}$. Sob $A_c \gg V_n$, o sinal recebido fica
\[
r(t) \approx [A_c + V_n\cos(\Phi_n - \phi)]\,\cos\!\bigl[2\pi f_c t + \phi(t) + Y_n(t)\bigr],
\]
ou seja, a fase é perturbada por $Y_n(t)$ apenas.

\end{frame}

% ============================================

\begin{frame}{De onde vem o $1/A_c$ na perturbação de fase}

\textbf{Soma fasorial} (envelope complexo):
\[
\tilde r(t) = A_c\, e^{j\phi(t)} + V_n\, e^{j\Phi_n(t)}
            = A_c\, e^{j\phi}\!\left[1 + \frac{V_n}{A_c}\, e^{j(\Phi_n - \phi)}\right].
\]

Separando o colchete em parte real e imaginária:
\[
1 + \tfrac{V_n}{A_c}\cos(\Phi_n - \phi) + j\,\tfrac{V_n}{A_c}\sin(\Phi_n - \phi).
\]

\textbf{Para $V_n \ll A_c$}, esse número complexo está perto de $1$. Sua fase é
\[
\arctan\!\left[\frac{\tfrac{V_n}{A_c}\sin(\Phi_n-\phi)}{1+\tfrac{V_n}{A_c}\cos(\Phi_n-\phi)}\right]
\;\approx\; \frac{V_n}{A_c}\sin(\Phi_n - \phi) \;\triangleq\; Y_n(t).
\]

O fator $1/A_c$ aparece naturalmente: a perturbação angular é a \emph{razão} entre a componente transversal do ruído e a amplitude do sinal — sinal mais forte, deflexão menor.

\end{frame}

% ============================================

\begin{frame}{Da fase do ruído até a saída do demodulador}

\textbf{Perturbação de fase} (expandindo $\sin(\Phi_n - \phi)$):
\[
Y_n(t) \triangleq \frac{V_n(t)}{A_c}\sin(\Phi_n - \phi) = \frac{1}{A_c}\bigl[n_s(t)\cos\phi(t) - n_c(t)\sin\phi(t)\bigr].
\]

\textbf{Saída do demodulador FM} (extrai $\tfrac{1}{2\pi}\,d\psi/dt$, com $\psi = \phi + Y_n$):
\[
y(t) = \frac{1}{2\pi}\frac{d}{dt}\bigl[\phi(t) + Y_n(t)\bigr]
     = \underbrace{k_f\, m(t)}_{\text{sinal}} + \underbrace{\frac{1}{2\pi}\frac{dY_n}{dt}}_{n_o(t)}.
\]

O sinal é \emph{aditivamente} corrompido pelo ruído $n_o(t)$, sem termos cruzados — graças à hipótese de alta SNR.

\end{frame}

% ============================================

\begin{frame}{PSD do ruído na saída do demodulador FM}

\textbf{PSD de $Y_n$.} Como $\phi(t)$ varia lentamente comparado a $n_c, n_s$, podemos tratar $\phi$ como constante dentro da janela de integração. Como $\cos^2\phi + \sin^2\phi = 1$:
\[
S_{Y_n}(f) = \frac{1}{A_c^2}\,S_{n_s}(f) = \frac{N_0}{A_c^2}, \qquad |f| \le W.
\]

\textbf{Derivação no tempo} equivale a multiplicar por $j2\pi f$ em frequência. Logo a PSD do ruído na saída é \textbf{parabólica}:
\[
S_{n_o}(f) = \frac{(2\pi f)^2}{(2\pi)^2}\,S_{Y_n}(f) = \frac{N_0\, f^2}{A_c^2}, \quad |f| \le W.
\]

\end{frame}

% ============================================

\begin{frame}{PSD parabólica: visualização}

\begin{center}
\begin{tikzpicture}[scale=0.9, font=\scriptsize]
\draw[->] (-2.5,0) -- (2.5,0) node[right] {$f$};
\draw[->] (0,-0.2) -- (0,2.0) node[above] {$S_{n_o}(f)$};
\draw[domain=-2:2, smooth, thick, blue] plot (\x, {0.4*\x*\x});
\draw[dashed] (-2,0) -- (-2,1.6);
\draw[dashed] (2,0) -- (2,1.6);
\node[below] at (2,0) {$W$};
\node[below] at (-2,0) {$-W$};
\end{tikzpicture}
\end{center}

\textbf{Consequência:} altas frequências da mensagem são mais afetadas pelo ruído que as baixas. Isso motiva a pré/pós-ênfase, vista mais adiante.

\end{frame}

% ============================================

\begin{frame}{Potência de ruído e de sinal na saída}

\textbf{Potência total do ruído} (após LPF com corte $W$):
\[
P_{n_o} = \int_{-W}^{W} S_{n_o}(f)\, df = \int_{-W}^{W} \frac{N_0 f^2}{A_c^2}\, df = \frac{N_0}{A_c^2}\cdot\frac{2 W^3}{3} = \frac{2 N_0 W^3}{3 A_c^2}.
\]

\textbf{Potência do sinal na saída} para mensagem qualquer:
\[
P_{s_o} = \overline{(k_f\,m(t))^2} = k_f^2\, P_M, \qquad P_M \triangleq \overline{m^2(t)}.
\]

Para tom $m(t) = A_m\cos(2\pi f_m t)$ com $P_M = A_m^2/2$ e $\Delta f = k_f A_m$:
\[
P_{s_o} = \frac{k_f^2 A_m^2}{2} = \frac{(\Delta f)^2}{2}.
\]

\end{frame}

% ============================================

\begin{frame}{SNR de saída em FM}

Dividindo $P_{s_o}$ por $P_{n_o}$:
\[
\left(\frac{S}{N}\right)_o
   = \frac{k_f^2\, P_M}{\,2 N_0 W^3 / (3 A_c^2)\,}
   = \frac{3\, A_c^2\, k_f^2\, P_M}{2\, N_0\, W^3}.
\]

Substituindo $\gamma = A_c^2/(2 N_0 W)$ e definindo $\beta = \Delta f/W$ com $\Delta f = k_f\max|m(t)|$:
\[
\left(\frac{S}{N}\right)_o
   = \frac{3\, k_f^2 P_M}{W^2}\cdot\frac{A_c^2}{2 N_0 W}
   = 3\,\beta^2\,\frac{P_M}{\max|m(t)|^2}\,\gamma.
\]

\textbf{Tom único} ($W = f_m$, $P_M/\max|m|^2 = 1/2$):
\[
\boxed{\left(\frac{S}{N}\right)_o = \tfrac{3}{2}\,\beta^2\, \gamma}, \qquad \beta = \frac{\Delta f}{f_m}.
\]

\end{frame}

% ============================================

\begin{frame}{O grande trade-off do FM}

\begin{itemize}
\item SNR de saída cresce com $\beta^2$.
\item Banda ocupada por Carson cresce com $\beta$: $B \approx 2(\beta + 1) f_m$.
\end{itemize}

\textbf{Logo:} dobrar o índice $\beta$ multiplica a SNR por 4 e a banda ocupada por aproximadamente 2.

\vspace{0.3cm}

\begin{block}{Princípio}
Em FM, trocamos \textbf{largura de banda} por \textbf{ganho de SNR}. Isso só faz sentido se há banda disponível no canal e se a SNR de entrada está acima do limiar.
\end{block}

\textbf{Comparação com AM:} para $\beta = 5$ e $\gamma$ fixo, $(S/N)_o = 37{,}5\,\gamma$ no FM, contra $\le \gamma/3$ no AM convencional. O FM pode estar \textbf{mais de 20 dB acima} do AM em SNR de saída.

\end{frame}

% ============================================

\]

\]

\left(\frac{S}{N}\right)_o\Big|_{\text{dB}} = 10\log_{10}(37{,}5) + 20 \approx 35{,}74 \text{ dB}

Ganho de $15{,}74$ dB sobre a referência $\gamma$, pagando $6$ vezes mais banda.
\end{block}

\end{frame}

\begin{frame}{Aplicação: FM com mensagem real (não-tonal)}

\textbf{Enunciado.} Um enlace FM transmite áudio com desvio máximo $\Delta f = 60$ kHz em banda de mensagem $W = 15$ kHz. Diferente de um tom puro, o áudio tem fator de potência normalizada
\[
P_{M_n} \triangleq \frac{P_M}{\max|m(t)|^2} = 0{,}2
\]
(valor típico para música, decorrente do fator de crista). O receptor mede $\gamma = 20$ dB.

\textbf{Pede-se:}
\begin{enumerate}
\item[(a)] o índice $\beta$ e a banda ocupada por Carson;
\item[(b)] $(S/N)_o$ usando a fórmula geral $(S/N)_o = 3\beta^2 P_{M_n}\, \gamma$;
\item[(c)] a comparação com a aproximação de tom único e a perda em dB.
\end{enumerate}

\end{frame}

% ============================================

\begin{frame}{Resolução: parâmetros do enlace}

\textbf{Passo 1.} Índice e banda de Carson:
\[
\beta = \frac{\Delta f}{W} = \frac{60}{15} = 4, \qquad B = 2(\Delta f + W) = 2(60 + 15) = 150 \text{ kHz}.
\]

\textbf{Passo 2.} SNR de canal em linear:
\[
\gamma = 10^{20/10} = 100.
\]

\end{frame}

% ============================================

\begin{frame}{Resolução: SNR com mensagem genérica}

\textbf{Passo 3.} Aplicando a fórmula geral:
\[
\left(\frac{S}{N}\right)_o = 3\,\beta^2\, P_{M_n}\, \gamma = 3\cdot 16 \cdot 0{,}2 \cdot 100 = 960,
\]
\[
\left(\frac{S}{N}\right)_o\bigg|_{\text{dB}} = 10\log_{10}(960) \approx 29{,}82 \text{ dB}.
\]

\textbf{Passo 4.} Caso fosse tom puro ($P_{M_n} = 1/2$):
\[
\left(\frac{S}{N}\right)_o^{\text{tom}} = \tfrac{3}{2}\beta^2\gamma = 1{,}5\cdot 16 \cdot 100 = 2400 \approx 33{,}80 \text{ dB}.
\]

\begin{block}{Conclusão}
A mensagem real custa $33{,}80 - 29{,}82 \approx 4$ dB em relação à hipótese de tom único, porque $P_{M_n} = 0{,}2$ contra $0{,}5$ do tom. A SNR efetiva escala linearmente com $P_{M_n}$.
\end{block}

\end{frame}

\section{Efeito de limiar em FM}

\begin{frame}{Quando a aproximação linear falha}

A análise anterior usou $E(t) \approx A_c$ e $\psi(t) \approx \phi + n_s/A_c$. Essas aproximações são válidas apenas enquanto a SNR de entrada é \textbf{alta}.

\vspace{0.3cm}

Quando a SNR de entrada cai, o vetor de ruído pode ter envelope comparável a $A_c$. Nesse caso:
\begin{itemize}
\item o envelope total $E(t)$ pode passar perto de zero;
\item o ponteiro de fase $\psi(t)$ pode dar uma \textbf{volta completa} em curto intervalo;
\item o discriminador interpreta essa volta como um pulso de frequência espúrio.
\end{itemize}

Esses pulsos rápidos são chamados de \textbf{cliques de ruído} (click noise) e geram picos audíveis na saída.

\end{frame}

% ============================================

\begin{frame}{Curva característica do limiar}

\begin{center}
\begin{tikzpicture}[font=\scriptsize, scale=0.9]
\draw[->] (0,0) -- (6.5,0) node[right] {$\gamma$ (dB)};
\draw[->] (0,0) -- (0,3.5) node[above] {$(S/N)_o$ (dB)};
\draw[thick, blue] (3,1.2) -- (6,3.3);
\draw[thick, blue] (1,0.2) .. controls (2,0.4) and (2.5,0.8) .. (3,1.2);
\draw[dashed] (3,1.2) -- (6,2.3) node[right] {banda base};
\draw[dashed, red] (3,0) -- (3,1.2);
\node[below] at (3,0) {limiar};
\node[right, blue] at (6,3.3) {$\tfrac{3}{2}\beta^2 \gamma$};
\end{tikzpicture}
\end{center}

\textbf{Acima do limiar:} comportamento linear $\tfrac{3}{2}\beta^2 \gamma$, ganho real sobre banda base. \textbf{Abaixo do limiar:} colapso da SNR, dominado por cliques.

\end{frame}

% ============================================

\begin{frame}{Fórmula aproximada do limiar}

A análise rigorosa do regime de cliques (fora do escopo) fornece uma relação aproximada entre a SNR de canal e a SNR de saída no joelho da curva. Para FM com índice $\beta$:
\[
\boxed{\left(\frac{S}{N}\right)_{b,\text{th}} = 20(\beta + 1)}
\qquad \text{(Proakis eq.~6.2.25)}
\]
onde $(S/N)_b = P_R/(N_0 W) = \gamma$ é a SNR de referência (banda base).

\textbf{Valores típicos:}
\begin{itemize}
\item $\beta = 2$: $\gamma_{\text{th}} = 60 \approx 17{,}8$ dB;
\item $\beta = 5$: $\gamma_{\text{th}} = 120 \approx 20{,}8$ dB;
\item $\beta = 7$: $\gamma_{\text{th}} = 160 \approx 22{,}0$ dB.
\end{itemize}

Aumentar $\beta$ amplia o ganho de SNR \emph{e} eleva o limiar -- mais banda exige mais potência.

\end{frame}

% ============================================

\begin{frame}{SNR de saída no joelho}

Substituindo $(S/N)_{b,\text{th}} = 20(\beta + 1)$ em $(S/N)_o = 3\beta^2 P_{M_n}\, (S/N)_b$ (eq.~6.2.22):
\[
\boxed{\left(\frac{S}{N}\right)_o\bigg|_{\text{th}} = 60\,\beta^2(\beta + 1)\, P_{M_n}}
\qquad \text{(eq.~6.2.33)}
\]

\textbf{Interpretação.} É o \emph{menor} $(S/N)_o$ obtido em operação linear. Para uma especificação $(S/N)_o$ desejada, essa equação dá o $\beta$ \emph{máximo} utilizável do ponto de vista de potência (acima dele, o sistema cai no regime de cliques).

\textbf{Tom único} ($P_{M_n} = 1/2$):
\[
\left(\frac{S}{N}\right)_o\bigg|_{\text{th}} = 30\,\beta^2(\beta + 1).
\]

\end{frame}

% ============================================

\begin{frame}{Critério prático de projeto}

\begin{block}{Regra}
Operar com $(S/N)_b$ pelo menos $10$ dB acima do joelho:
\[
\gamma_{\text{nom}} \ge \gamma_{\text{th}} + 10 \text{ dB}, \qquad
\gamma_{\text{th}} = 10\log_{10}\!\bigl[20(\beta+1)\bigr]\text{ dB}.
\]
\end{block}

\textbf{Passos:}
\begin{itemize}
\item calcular $\gamma_{\text{th}}$ pela fórmula acima a partir de $\beta$;
\item estimar $\gamma_{\text{nom}}$ pelo orçamento do enlace;
\item exigir $\gamma_{\text{nom}} - \gamma_{\text{th}} \ge 10$ dB;
\item incluir margem adicional para desvanecimentos.
\end{itemize}

\end{frame}

% ============================================

\begin{frame}{Aplicação: projeto de sistema FM (Proakis 6.2.2)}

\textbf{Enunciado.} Projetar um sistema FM com:
\begin{itemize}
\item SNR de saída requerida: $(S/N)_o = 40$ dB;
\item banda do canal: $B_c = 120$ kHz;
\item banda da mensagem: $W = 10$ kHz;
\item mensagem com $P_{M_n} = 1/2$;
\item PSD de ruído (unilateral): $N_0 = 10^{-8}$ W/Hz;
\item atenuação no canal: $40$ dB.
\end{itemize}

\textbf{Pede-se:} a potência mínima do transmissor $P_T$ que cumpre a especificação \emph{acima} do limiar de FM.

\end{frame}

% ============================================

\begin{frame}{Resolução: limites de $\beta$}

\textbf{Passo 1 -- Banda (Carson).}
\[
B_c = 2(\beta + 1)\, W \;\Rightarrow\; 120 = 2(\beta+1)\cdot 10
\;\Rightarrow\; \beta_{\max}^{\text{banda}} = 5.
\]

\textbf{Passo 2 -- Potência (limiar).} Da fórmula $(S/N)_o|_{\text{th}} = 60\beta^2(\beta+1) P_{M_n}$, igualando à especificação $(S/N)_o = 10^4$:
\[
60\,\beta^2(\beta+1)\cdot \tfrac{1}{2} = 10^4
\;\Rightarrow\; \beta^2(\beta+1) \approx 333{,}3
\;\Rightarrow\; \beta_{\max}^{\text{pot}} \approx 6{,}6.
\]

Como $\beta_{\max}^{\text{banda}} = 5 < \beta_{\max}^{\text{pot}} = 6{,}6$, o sistema está \textbf{limitado pela banda}. Adotamos $\beta = 5$.

\end{frame}

% ============================================

\begin{frame}{Cálculo analítico de $\beta_{\max}^{\text{pot}}$: setup}

A condição $\beta^2(\beta+1) = 333{,}3$ é uma cúbica:
\[
\beta^3 + \beta^2 - 333{,}3 = 0.
\]

\textbf{Aproximação inicial} (descartando $\beta^2$ frente a $\beta^3$):
\[
\beta^3 \approx 333{,}3 \;\Rightarrow\; \beta_0 \approx \sqrt[3]{333{,}3} \approx 6{,}933.
\]

\textbf{Substituição de Cardano.} $\beta = t - 1/3$ elimina o termo quadrático e produz a cúbica deprimida
\[
t^3 + p\,t + q = 0, \qquad p = -\tfrac{1}{3}, \quad q = \tfrac{2}{27} - 333{,}3 = -333{,}226.
\]

\end{frame}

% ============================================

\begin{frame}{Cálculo analítico de $\beta_{\max}^{\text{pot}}$: Cardano}

\textbf{Raiz real} da cúbica deprimida:
\[
t = \sqrt[3]{-q/2 + \sqrt{R}} + \sqrt[3]{-q/2 - \sqrt{R}}, \qquad
R \triangleq (q/2)^2 + (p/3)^3.
\]

Avaliando ($-q/2 = 166{,}613$, $(q/2)^2 = 27\,759{,}892$, $(p/3)^3 = -1/729 \approx -1{,}37\times 10^{-3}$):
\[
\sqrt{R} \approx 166{,}61300, \qquad
t \approx \sqrt[3]{333{,}226} + \sqrt[3]{4{,}1\times 10^{-6}}
   \approx 6{,}933 + 0{,}016 = 6{,}949.
\]

\[
\boxed{\beta_{\max}^{\text{pot}} = t - \tfrac{1}{3} \approx 6{,}616}
\]

\end{frame}

% ============================================

\begin{frame}{Resolução: potência do transmissor}

\textbf{Passo 3 -- SNR de canal exigida.} Com $\beta = 5$ na fórmula linear $(S/N)_o = \tfrac{3}{2}\beta^2 (S/N)_b$:
\[
\left(\frac{S}{N}\right)_b = \frac{10^4}{37{,}5} = \frac{800}{3} \approx 266{,}7 \;\;(24{,}26\text{ dB}).
\]

\textbf{Passo 4 -- Potência recebida.}
\[
P_R = \left(\frac{S}{N}\right)_b N_0\, W = 266{,}7 \cdot 10^{-8} \cdot 10^{4} \approx 2{,}67\times 10^{-2}\text{ W} = -15{,}74 \text{ dBW}.
\]

\textbf{Passo 5 -- Potência transmitida} (canal com $40$ dB de atenuação):
\[
\boxed{P_T = P_R + 40 \text{ dB} = -15{,}74 + 40 = 24{,}26 \text{ dBW} \approx 266{,}7 \text{ W}.}
\]

\end{frame}

% ============================================

\begin{frame}{Resolução: trade-off banda vs.\ potência}

\textbf{Cenário sem limite de banda.} Se a banda do canal não fosse restrita a $120$ kHz, poderíamos adotar $\beta \approx 6{,}6$ (o máximo permitido pelo limiar):
\[
B_c = 2(6{,}6 + 1)\cdot 10 = 152 \text{ kHz}, \qquad
P_R \approx 15{,}3 \text{ mW}, \qquad
P_T \approx 153 \text{ W}.
\]

\textbf{Comparação:}
\begin{center}
\begin{tabular}{lccc}
\toprule
Cenário & $\beta$ & $B_c$ (kHz) & $P_T$ (W) \\
\midrule
Limitado pela banda  & $5$    & $120$ & $266{,}7$ \\
Limitado pela potência & $6{,}6$ & $152$ & $153$    \\
\bottomrule
\end{tabular}
\end{center}

\begin{block}{Trade-off}
Mais banda $\Rightarrow$ menos potência transmitida. Aumentar $\beta$ de $5$ para $6{,}6$ custa $32$ kHz extras de banda mas economiza $114$ W ($\approx 2{,}4$ dB de potência).
\end{block}

\end{frame}

\section{Pré-ênfase e pós-ênfase em FM}

\begin{frame}{Motivação: PSD parabólica}

Vimos que o ruído na saída do demodulador FM tem PSD $\propto f^2$. Conclusão:

\vspace{0.3cm}

\begin{itemize}
\item baixas frequências da mensagem: pouco ruído;
\item altas frequências da mensagem: muito ruído;
\item em áudio musical, isso degrada agudos e atapetes a chocalhos.
\end{itemize}

\vspace{0.3cm}

\textbf{Ideia:} amplificar as altas frequências da mensagem \textbf{antes} de modular, e atenuar as altas frequências \textbf{depois} de demodular. O sinal volta plano; o ruído sofre apenas a atenuação final.

\end{frame}

% ============================================

\begin{frame}{Filtros de pré-ênfase e pós-ênfase}

\textbf{Pré-ênfase} no transmissor: filtro passa-altas suave,
\[
H_{pe}(f) = 1 + j\, \frac{f}{f_0}, \qquad |H_{pe}(f)|^2 = 1 + (f/f_0)^2
\]

\textbf{Pós-ênfase} no receptor: filtro inverso,
\[
H_{de}(f) = \frac{1}{1 + j f/f_0}
\]

\textbf{FM broadcast:} $\tau = 1/(2\pi f_0) = 75\,\mu$s, ou seja $f_0 \approx 2{,}1$ kHz.

\end{frame}

% ============================================

\begin{frame}{Resposta dos filtros}

\begin{center}
\begin{tikzpicture}[font=\footnotesize]
\draw[->] (0,0) -- (5,0) node[right] {$f$};
\draw[->] (0,0) -- (0,2.5) node[above] {$|H(f)|$};
\draw[thick, blue, domain=0:4.5, samples=50] plot (\x, {sqrt(1+(\x/2)^2)});
\draw[thick, red, domain=0:4.5, samples=50] plot (\x, {1/sqrt(1+(\x/2)^2)});
\node[blue, right] at (4.5, 2.3) {$|H_{pe}|$};
\node[red, right] at (4.5, 0.4) {$|H_{de}|$};
\draw[dashed] (0,1) -- (4.5,1);
\end{tikzpicture}
\end{center}

A pré-ênfase amplifica altas frequências antes da modulação. A pós-ênfase as atenua de volta no receptor, junto com o ruído acumulado.

\end{frame}

% ============================================

\begin{frame}{Ganho com pré/pós-ênfase}

Para mesma potência média transmitida, o ganho de SNR com pré/pós-ênfase é:
\[
G = \frac{(W/f_0)^3 / 3}{(W/f_0) - \arctan(W/f_0)}
\]

Para FM broadcast com $W = 15$ kHz e $f_0 \approx 2{,}1$ kHz:
\[
G \approx 13 \text{ dB}
\]

\vspace{0.3cm}

\begin{block}{Resultado prático}
A pré/pós-ênfase recupera mais de uma década de melhoria na SNR percebida em áudio FM, sem custo extra de banda.
\end{block}

\end{frame}

% ============================================

\begin{frame}{Aplicação: cálculo do ganho com $\tau = 75\,\mu$s}

\textbf{Enunciado.} FM broadcast usa constante de tempo de pré-ênfase $\tau = 75\,\mu$s. Para áudio com $W = 15$ kHz, calcule $f_0$ e o ganho $G$ proporcionado pela pré/pós-ênfase.

\begin{center}
\begin{tikzpicture}[font=\scriptsize]
\draw[->] (0,0) -- (5,0) node[right] {$f$ (kHz)};
\draw[->] (0,0) -- (0,2.5);
\draw[domain=0:4.5, smooth, samples=50, blue, thick] plot (\x, {0.1*\x*\x});
\node[blue] at (3.5,2.0) {$S_{n_o}\propto f^2$};
\draw[domain=0:4.5, smooth, samples=50, red, thick] plot (\x, {0.1*\x*\x/(1+(\x/2.1)^2)});
\node[red] at (4.5,0.7) {com pós-\^enfase};
\node[below] at (4.5,0) {15};
\node[below] at (0.6,0) {2{,}1};
\end{tikzpicture}
\end{center}

\end{frame}

% ============================================

\begin{frame}{Resolução: frequência de corte}

\textbf{Passo 1.} Frequência de corte da pré-ênfase:
\[
f_0 = \frac{1}{2\pi \tau} = \frac{1}{2\pi \cdot 75 \times 10^{-6}} \approx 2122 \text{ Hz}
\]

\textbf{Passo 2.} Razão entre a banda da mensagem e $f_0$:
\[
r = \frac{W}{f_0} = \frac{15\,000}{2122} \approx 7{,}07
\]

\end{frame}

% ============================================

\begin{frame}{Resolução: ganho final}

\textbf{Passo 3.} Aplicando a fórmula do ganho:
\[
G = \frac{r^3 / 3}{r - \arctan r} = \frac{117{,}8}{5{,}64} \approx 20{,}9
\]
\[
G\big|_{\text{dB}} = 10\log_{10}(20{,}9) \approx 13{,}2 \text{ dB}
\]

\begin{block}{Resultado}
O áudio FM ganha cerca de $13$ dB de SNR percebida graças à pré/pós-ênfase, sem aumentar potência nem banda.
\end{block}

\end{frame}

% ============================================

\begin{frame}{Aplicação: melhoria total de FM com PD (Proakis 6.2.3)}

\textbf{Enunciado.} Para FM broadcast comercial com:
\begin{itemize}
\item banda da mensagem $W = 15$ kHz;
\item frequência de corte da pré-ênfase $f_0 = 2100$ Hz;
\item índice de modulação $\beta = 5$;
\item potência média/de pico da mensagem $P_{M_n} = 0{,}5$.
\end{itemize}

\textbf{Pede-se} a melhoria \emph{total} em $(S/N)_o$ ao usar FM com pré/pós-ênfase (PD) em relação a um sistema banda base de mesma potência recebida.

\end{frame}

% ============================================

\begin{frame}{Resolução: FM puro vs banda base}

\textbf{Passo 1 -- FM sem PD.} Da eq.~6.2.22, $(S/N)_o = 3\beta^2 P_{M_n} (S/N)_b$:
\[
\left(\frac{S}{N}\right)_o = 3 \cdot 5^2 \cdot 0{,}5 \cdot \left(\frac{S}{N}\right)_b = 37{,}5\,\left(\frac{S}{N}\right)_b.
\]

Em dB:
\[
\left(\frac{S}{N}\right)_o\bigg|_{\text{dB}} = 10\log_{10}(37{,}5) + \left(\frac{S}{N}\right)_b\bigg|_{\text{dB}} \approx 15{,}7 + \left(\frac{S}{N}\right)_b\bigg|_{\text{dB}}.
\]

FM puro está $15{,}7$ dB acima de banda base.

\end{frame}

% ============================================

\begin{frame}{Resolução: contribuição da pré/pós-ênfase}

\textbf{Passo 2 -- Ganho de PD.} Aplicando $G = \dfrac{(W/f_0)^3 / 3}{(W/f_0) - \arctan(W/f_0)}$ com $W/f_0 = 15000/2100 \approx 7{,}14$:
\[
G = \frac{1}{3}\cdot\frac{(7{,}14)^3}{7{,}14 - \arctan(7{,}14)}
   = \frac{1}{3}\cdot\frac{364{,}4}{7{,}14 - 1{,}432}
   \approx 21{,}3.
\]

Em dB: $G\big|_{\text{dB}} = 10\log_{10}(21{,}3) \approx 13{,}3$ dB. Logo $(S/N)_{o,\text{PD}} = G \cdot (S/N)_o$:
\[
\left(\frac{S}{N}\right)_{o,\text{PD}}\bigg|_{\text{dB}} \approx 13{,}3 + \left(\frac{S}{N}\right)_o\bigg|_{\text{dB}}.
\]

\end{frame}

% ============================================

\begin{frame}{Resolução: melhoria total}

\textbf{Passo 3 -- Encadeamento.} Substituindo o resultado do Passo 1:
\[
\left(\frac{S}{N}\right)_{o,\text{PD}}\bigg|_{\text{dB}}
\approx 13{,}3 + 15{,}7 + \left(\frac{S}{N}\right)_b\bigg|_{\text{dB}}
= 29 + \left(\frac{S}{N}\right)_b\bigg|_{\text{dB}}.
\]

Ou em forma linear: $(S/N)_{o,\text{PD}} = 21{,}3 \cdot 37{,}5 \cdot (S/N)_b \approx 800 \cdot (S/N)_b$.

\begin{block}{Resultado final}
FM broadcast com pré/pós-ênfase entrega $29$ dB de SNR \emph{a mais} do que o equivalente banda base de mesma potência: $\boxed{15{,}7 \text{ dB (FM)} + 13{,}3 \text{ dB (PD)} = 29 \text{ dB}}$, sem custo adicional de banda ou potência média.
\end{block}

\end{frame}

\section{Ruído térmico}

\begin{frame}{Origem do ruído térmico}

Em qualquer resistor real, agitação térmica de portadores gera um ruído de tensão. Modelado como \textbf{ruído branco gaussiano} com PSD bilateral:
\[
S_n(f) = \frac{N_0}{2}, \qquad N_0 = kT \quad \text{em W/Hz}
\]
$k = 1{,}38\times 10^{-23}$ J/K é a constante de Boltzmann.

\textbf{Temperatura de referência $T_0 = 290$ K:}
\[
N_0 \approx 4 \times 10^{-21} \text{ W/Hz} = -174 \text{ dBm/Hz}
\]

\textbf{Potência em banda $B$ Hz:}
\[
P_{\text{noise}} = N_0 B
\]

\end{frame}

% ============================================

\begin{frame}{Aplicação: piso de ruído de receptores}

\textbf{Enunciado.} Calcule o piso de ruído térmico em dBm para três sistemas a $T = 290$ K:
\begin{enumerate}

\item LTE: $B = 10$ MHz;
\item Voz: $B = 4$ kHz.
\end{enumerate}

\]

\vspace{0.3cm}

\textbf{Caso 1: Wi-Fi, $B = 20$ MHz.}
\[
P_{\text{noise}} = -174 + 73 = -101 \text{ dBm}
\]

\textbf{Caso 2: LTE, $B = 10$ MHz.}
\[
P_{\text{noise}} = -174 + 70 = -104 \text{ dBm}
\]

\textbf{Caso 3: voz, $B = 4$ kHz.}
\[
P_{\text{noise}} = -174 + 36 = -138 \text{ dBm}
\]

\end{frame}

% ============================================

\begin{frame}{Conclusão: regra dos $10$ dB por década}

\begin{block}{Lição}
Cada década de banda custa $10$ dB de sensibilidade. Reduzir a banda do receptor é a forma mais barata de baixar o piso de ruído.
\end{block}

\vspace{0.3cm}

\textbf{Comparação dos três casos:}
\begin{itemize}
\item $20$ MHz $\to -101$ dBm;
\item $10$ MHz $\to -104$ dBm ($3$ dB melhor);
\item $4$ kHz $\to -138$ dBm ($37$ dB melhor que Wi-Fi).
\end{itemize}

Por isso receptores de voz podem operar com potências muito menores que receptores de banda larga.

\end{frame}

\section{Figura de ruído e temperatura equivalente}

\begin{frame}{Ruído próprio de um amplificador}

Um amplificador real \emph{adiciona} ruído. Sinal e ruído da fonte são amplificados pelo mesmo ganho $G$, mas o bloco soma seu próprio ruído $N_a$ na saída.

\begin{center}
\begin{tikzpicture}[font=\scriptsize]
\coordinate (in) at (-1.6,0);
\coordinate (al) at (0,0);
\coordinate (ar) at (1.6,0);
\coordinate (out) at (3.6,0);
\draw[fill=blue!10] (al) ++(0,-0.6) -- ++(0,1.2) -- (ar) -- cycle;
\node at ($(al)!0.45!(ar)$) {$G$};
\draw[->, thick] (in) -- node[above, font=\tiny] {$S_i$} node[below, font=\tiny] {$N_i = k T_0 B$} (al);
\draw[->, thick] (ar) -- node[above, font=\tiny] {$S_o = G S_i$} node[below, font=\tiny] {$N_o = G N_i + N_a$} (out);
\end{tikzpicture}
\end{center}

Comparando as SNRs:
\[
\left(\tfrac{S}{N}\right)_i = \frac{S_i}{N_i},
\qquad
\left(\tfrac{S}{N}\right)_o = \frac{G S_i}{G N_i + N_a} < \left(\tfrac{S}{N}\right)_i.
\]

A SNR \textbf{sempre piora} ao atravessar o bloco -- esse é o efeito que queremos quantificar.

\end{frame}

% ============================================

\begin{frame}{Figura de ruído $F$}

A \textbf{figura de ruído} quantifica essa degradação.

\textbf{Definição} (com fonte na entrada à temperatura $T_0 = 290$ K):
\[
F = \frac{(S/N)_{\text{entrada}}}{(S/N)_{\text{saída}}} \ge 1
\]

Em dB: $\mathrm{NF} = 10\log_{10} F$.

\vspace{0.3cm}

\begin{block}{Interpretação}
$F$ é o fator pelo qual a SNR se degrada no bloco. $F = 1$ corresponde a um bloco ideal sem ruído próprio.
\end{block}

\end{frame}

% ============================================

\begin{frame}{De onde vem $T_e$ -- modelo do amplificador}

Amplificador com ganho $G$, banda $B$ e fonte na entrada à temperatura $T_0$.

\textbf{Entrada:} sinal $S_i$ e ruído $N_i = k T_0 B$.

\textbf{Saída:} $S_o = G S_i$ e
\[
N_o = G N_i + N_a,
\]
onde $N_a$ é o ruído próprio do bloco, medido na saída.

Pela definição de $F$:
\[
F = \frac{(S/N)_i}{(S/N)_o} = \frac{S_i/N_i}{G S_i / N_o} = \frac{N_o}{G N_i} = 1 + \frac{N_a}{G\, k T_0 B}.
\]

\end{frame}

% ============================================

\begin{frame}{De onde vem $T_e$ -- equivalência térmica}

Define-se $T_e$ como a temperatura de uma fonte fictícia que, posta na entrada de um bloco \emph{ideal}, produz o mesmo ruído próprio:
\[
N_a = G\, k T_e B \;\Longrightarrow\; \frac{N_a}{G\, k T_0 B} = \frac{T_e}{T_0}.
\]

Substituindo na expressão de $F$ do slide anterior:
\[
F = 1 + \frac{T_e}{T_0} \quad\Longleftrightarrow\quad \boxed{\,T_e = (F-1)\, T_0\,}
\]

\end{frame}

% ============================================

\begin{frame}{Temperatura equivalente de ruído}

Alternativa a $F$, especialmente útil em sistemas de baixo ruído como antenas e LNAs:
\[
T_e = (F - 1)\, T_0
\]

\textbf{Significado:} o ruído interno do bloco equivale a colocar uma fonte térmica à temperatura $T_e$ na entrada, em série com a temperatura do ambiente $T_0$.

\vspace{0.3cm}

\textbf{Equivalências úteis:}
\begin{center}
\small
\begin{tabular}{|c|c|c|}
\hline
$F$ (linear) & $\mathrm{NF}$ (dB) & $T_e$ (K) \\
\hline
1{,}1 & 0{,}4 & 29 \\
1{,}5 & 1{,}8 & 145 \\
2 & 3 & 290 \\
4 & 6 & 870 \\
10 & 10 & 2610 \\
\hline
\end{tabular}
\end{center}

\end{frame}

% ============================================

\begin{frame}{Demonstração: setup do problema}

Dois blocos em cascata: $(G_1, F_1, T_1)$ seguido de $(G_2, F_2, T_2)$. Cada bloco adiciona ruído próprio, modelado por uma temperatura equivalente $T_i = (F_i - 1) T_0$ na entrada.

\begin{center}
\begin{tikzpicture}[font=\scriptsize, node distance=0.6cm,
  block/.style={rectangle, draw, rounded corners, minimum height=0.9cm, minimum width=1.5cm, align=center}]
\node (in) {$T_0$};
\node[block, right=of in, fill=green!10] (b1) {$G_1, F_1$\\\tiny $T_1$};
\node[block, right=of b1, fill=yellow!10] (b2) {$G_2, F_2$\\\tiny $T_2$};
\node[right=of b2] (out) {saída};
\draw[->] (in) -- (b1);
\draw[->] (b1) -- (b2);
\draw[->] (b2) -- (out);
\end{tikzpicture}
\end{center}

\textbf{Estratégia:} referir todo o ruído à \textbf{entrada} do bloco 1, somar as contribuições e comparar com o ruído da fonte $kT_0$.

\end{frame}

% ============================================

\begin{frame}{Demonstração: ruído na saída}

\textbf{Passo 1.} Ruído na saída devido à fonte $T_0$:
\[
N_{0,\text{out}} = k T_0\, G_1 G_2 B
\]

\textbf{Passo 2.} Ruído na saída devido ao bloco 1 (gerado em sua entrada com temperatura $T_1$, amplificado por $G_1 G_2$):
\[
N_{1,\text{out}} = k T_1\, G_1 G_2 B
\]

\textbf{Passo 3.} Ruído do bloco 2 (gerado em sua entrada com temperatura $T_2$, amplificado só por $G_2$):
\[
N_{2,\text{out}} = k T_2\, G_2 B
\]

\textbf{Passo 4.} Ruído total na saída:
\[
N_{\text{out}} = k B G_1 G_2 \left( T_0 + T_1 + \tfrac{T_2}{G_1} \right)
\]

\end{frame}

% ============================================

\begin{frame}{Demonstração: temperatura equivalente da cascata}

\textbf{Passo 5.} Reescrevendo $N_{\text{out}} = k\, G_1 G_2 B\, (T_0 + T_{e,\text{tot}})$, identificamos:
\[
\boxed{T_{e,\text{tot}} = T_1 + \frac{T_2}{G_1}}
\]
O ruído do bloco 2, referido à entrada, é \textbf{dividido} por $G_1$.

\textbf{Passo 6.} Convertendo para figura de ruído via $F = 1 + T_e/T_0$:
\[
F_{\text{tot}} = 1 + \frac{T_1}{T_0} + \frac{T_2}{T_0 G_1} = 1 + (F_1 - 1) + \frac{F_2 - 1}{G_1}
\]

\begin{block}{Friis para dois estágios}
\[
F_{\text{tot}} = F_1 + \frac{F_2 - 1}{G_1}
\]
\end{block}

\end{frame}

% ============================================

\begin{frame}{Generalização para $n$ estágios}

Aplicando o mesmo raciocínio recursivamente, o ruído do estágio $k$ é referido à entrada dividindo por $G_1 G_2 \cdots G_{k-1}$:
\[
\boxed{F_{\text{tot}} = F_1 + \frac{F_2 - 1}{G_1} + \frac{F_3 - 1}{G_1 G_2} + \cdots + \frac{F_n - 1}{G_1 G_2 \cdots G_{n-1}}}
\]

\textbf{Implicações:}
\begin{itemize}
\item se $G_1$ é grande, o primeiro estágio \textbf{domina} a figura total;
\item por isso o primeiro estágio do receptor é um LNA com $F_1$ baixo e $G_1$ alto;
\item estágios ruidosos posteriores são tolerados, desde que o ganho prévio os mascare.
\end{itemize}

\end{frame}

% ============================================

\textbf{Enunciado.} Um front-end de receptor tem três blocos em cascata:
\begin{itemize}

\item Misturador: $\mathrm{NF}_2 = 8$ dB, $G_2 = 10$ dB;

\end{itemize}
Calcule a figura de ruído total da cadeia.

\begin{frame}{Como medir $F$ na bancada: método Y}

Em laboratório não se mede $F$ pela definição direta. Usa-se uma fonte de ruído calibrada (diodo de ruído) que comuta entre dois estados:
\begin{itemize}
\item \emph{hot}: temperatura equivalente $T_H \gg T_0$ (diodo ligado);
\item \emph{cold}: temperatura equivalente $T_C \approx T_0$ (diodo desligado).
\end{itemize}

\begin{center}
\begin{tikzpicture}[font=\scriptsize]
\coordinate (al) at (0,0);
\coordinate (ar) at (1.6,0);
\node[draw, rounded corners, fill=orange!10, minimum height=0.7cm, minimum width=1.4cm] (src) at (-2.6,0) {fonte $T_H/T_C$};
\draw[fill=blue!10] (al) ++(0,-0.6) -- ++(0,1.2) -- (ar) -- cycle;
\node at ($(al)!0.45!(ar)$) {DUT};
\node[draw, rounded corners, fill=green!10, minimum height=0.7cm, minimum width=1.4cm] (pm) at (3.6,0) {medidor $P$};
\draw[->, thick] (src) -- (al);
\draw[->, thick] (ar) -- (pm);
\end{tikzpicture}
\end{center}

A potência de ruído na saída do DUT, em cada estado, vale:
\[
N_H = G k B \,(T_H + T_e), \qquad N_C = G k B \,(T_C + T_e).
\]

\end{frame}

% ============================================

\begin{frame}{Método Y: extração de $T_e$ e $F$}

Define-se a razão entre as duas medidas (ganho $G$ e banda $B$ cancelam):
\[
Y = \frac{N_H}{N_C} = \frac{T_H + T_e}{T_C + T_e}
\]

Isolando $T_e$:
\[
T_e = \frac{T_H - Y\, T_C}{Y - 1}, \qquad F = 1 + \frac{T_e}{T_0}.
\]

\textbf{Forma com ENR.} O catálogo do diodo de ruído fornece o \emph{Excess Noise Ratio}
\[
\text{ENR} = \frac{T_H - T_0}{T_0} \quad (\text{tipicamente } 5\text{--}15 \text{ dB}).
\]
Quando $T_C = T_0$, a expressão acima reduz a
\[
\boxed{\, F = \frac{\text{ENR}}{Y - 1} \,}
\]

\textbf{Vantagem:} mede-se apenas a razão $Y$ -- nenhum ganho, banda ou potência absoluta precisa ser calibrado.

\end{frame}

% ============================================

\begin{frame}{Demonstração: $F = \mathrm{ENR}/(Y-1)$}

\footnotesize
Partimos das três relações estabelecidas:
\[
T_e = \frac{T_H - Y\,T_C}{Y-1}, \quad F = 1 + \frac{T_e}{T_0}, \quad \mathrm{ENR} = \frac{T_H - T_0}{T_0}.
\]

Com $T_C = T_0$ (estado COLD na temperatura de referência):
\[
T_e = \frac{T_H - Y\,T_0}{Y-1}
\;\Longrightarrow\;
F = 1 + \frac{T_H - Y\,T_0}{T_0\,(Y-1)} = \frac{T_0\,(Y-1) + T_H - Y\,T_0}{T_0\,(Y-1)}.
\]

No numerador, $Y\,T_0$ se cancela: $\;T_0 Y - T_0 + T_H - Y T_0 = T_H - T_0$.

Logo:
\[
F = \frac{T_H - T_0}{T_0\,(Y-1)} = \frac{1}{Y-1}\cdot\underbrace{\frac{T_H - T_0}{T_0}}_{\mathrm{ENR}} \;\;\Longrightarrow\;\; \boxed{\,F = \frac{\mathrm{ENR}}{Y-1}\,}
\]

\end{frame}

% ============================================

\begin{frame}{O que a ENR significa fisicamente}

\small A \textbf{ENR} mede o ruído \emph{excedente} ao piso térmico. No estado HOT a fonte emite $k T_H B$; no COLD, $k T_0 B$. O excesso normalizado por $T_0$:
\[
\text{ENR} = \frac{T_H - T_0}{T_0} \quad\Longleftrightarrow\quad T_H = T_0\,(1 + \text{ENR}).
\]

\begin{frame}{Exemplo de bancada: medindo a NF de um LNA}

\textbf{Enunciado.} Caracterizar a NF de um LNA em $f_0 = 2{,}4$ GHz pelo método Y.

\textbf{Equipamento e dados:}
\begin{itemize}
\item fonte de ruído com $\text{ENR} = 15$ dB em $2{,}4$ GHz (catálogo);
\item analisador de espectro com banda de resolução $B = 100$ kHz;
\item temperatura ambiente $T_C = T_0 = 290$ K.
\end{itemize}

\textbf{Procedimento:}
\begin{enumerate}
\item Conectar: fonte de ruído $\to$ DUT $\to$ analisador.
\item Sintonizar o analisador em $2{,}4$ GHz, fixar $B = 100$ kHz e ativar média.
\item Ligar o diodo (estado \emph{hot}) e registrar a potência $N_H$.
\item Desligar o diodo (estado \emph{cold}) e registrar $N_C$.
\item Calcular $Y_{\text{dB}} = N_H - N_C$ e em seguida $F$, $T_e$ e NF.
\end{enumerate}

\textbf{Medidas obtidas:} $N_H = -50{,}0$ dBm e $N_C = -60{,}0$ dBm.

\end{frame}

\begin{frame}{Precisamos calibrar?}

Sim -- em duas frentes distintas.

\textbf{Fonte de ruído: já vem calibrada de fábrica.} O catálogo fornece a curva $\text{ENR}(f)$ ponto a ponto, com rastreabilidade metrológica (NIST/NPL/INMETRO). Cabe ao usuário ler o valor da ENR na frequência exata do teste.

\textbf{Receptor: cabe ao usuário corrigir.} O próprio analisador (e qualquer pré-amplificador antes dele) adiciona ruído. A medida bruta não dá $F_{\text{DUT}}$ diretamente, mas sim $F_{\text{med}}$ contaminado, ligado ao receptor por Friis:
\[
F_{\text{med}} = F_{\text{DUT}} + \frac{F_{\text{rec}} - 1}{G_{\text{DUT}}}.
\]

Sem isolar a parcela do receptor, a NF do DUT sai \emph{superestimada}.

\end{frame}

% ============================================

\begin{frame}{Calibração de duas etapas: procedimento}

\textbf{Etapa CAL (sem DUT).} Conectar a fonte de ruído \emph{diretamente} ao analisador e aplicar o método Y para obter a NF do receptor:
\[
F_{\text{rec}} = \frac{\text{ENR}}{Y_{\text{cal}} - 1}.
\]

\textbf{Etapa MEAS (com DUT).} Inserir o DUT no caminho e repetir o método Y:
\[
F_{\text{med}} = \frac{\text{ENR}}{Y_{\text{med}} - 1}.
\]

\textbf{Correção de Friis (isola o DUT).}
\[
\boxed{\;F_{\text{DUT}} = F_{\text{med}} - \frac{F_{\text{rec}} - 1}{G_{\text{DUT}}}\;}
\]

\end{frame}

% ============================================

\begin{frame}{Calibração de duas etapas: aplicação prática}

\textbf{No exemplo anterior.} Com $G_{\text{DUT}} = 100$ (20 dB) e $F_{\text{rec}} \approx 10$ (10 dB típico de analisador):
\[
\frac{F_{\text{rec}} - 1}{G_{\text{DUT}}} = \frac{9}{100} = 0{,}09.
\]
A correção subtrai $\sim 0{,}09$ de $F$ -- pequena no nosso caso porque o LNA tem ganho alto que mascara o receptor. Em DUTs de ganho baixo, ou com $F_{\text{rec}}$ alta, a correção passa a dominar a medida.

\vspace{0.3cm}

\textbf{Na prática.} Analisadores dedicados a figura de ruído (Keysight N8975B, R\&S FSWP) executam as duas etapas automaticamente: gravam $F_{\text{rec}}(f)$ num passe de calibração e aplicam a correção de Friis ponto a ponto durante a medida do DUT.

\end{frame}

% ============================================

\begin{frame}{Fonte de ruído calibrada: visão geral}

Uma fonte de ruído de bancada gera \textbf{ruído branco gaussiano} com densidade espectral conhecida, em uma saída coaxial de $50\,\Omega$.

\textbf{Dois estados controlados por chave DC:}
\begin{itemize}
\item \emph{cold:} alimentação desligada -- a saída emite apenas o ruído térmico do hardware passivo, $T_C \approx T_0 = 290$ K;
\item \emph{hot:} alimentação ligada -- a saída emite ruído equivalente a $T_H \gg T_0$.
\end{itemize}

\textbf{Parâmetros de catálogo:}
\begin{itemize}
\item curva de ENR versus frequência (tabela ou polinômio do fabricante);
\item faixa de frequência (DC--18 GHz típico; até 110 GHz em modelos mmWave);
\item VSWR de saída nos dois estados (idealmente $< 1{,}3$).
\end{itemize}

\end{frame}

% ============================================

\begin{frame}{Como o ruído é gerado fisicamente}

Três mecanismos físicos principais:

\begin{itemize}
\item \textbf{Ruído térmico (Johnson-Nyquist).} Carga resistiva à temperatura $T$ entrega $kTB$. Excelente como padrão primário de calibração, mas atingir ENR alta exigiria milhares de K -- inviável fora de cargas criogênicas/aquecidas dedicadas.
\item \textbf{Ruído \emph{shot} de diodo de avalanche.} Diodo reverso-polarizado em \emph{breakdown}; cada avalanche é um evento de Poisson cuja flutuação domina o ruído. Banda larga, ENR de 5--25 dB, baixo custo -- é a base da quase totalidade das fontes comerciais.
\item \textbf{Tubo de gás ionizado (legacy).} Plasma de neônio com $T \sim 10^4$ K em guia de onda. Boa estabilidade mas frágil e volumoso; quase aposentado pelo diodo de avalanche.
\end{itemize}

\end{frame}

\textbf{Função de cada bloco:}
\begin{itemize}

\item \emph{Diodo de avalanche:} converte a corrente DC em ruído \emph{shot} de banda larga;
\item \emph{Atenuador resistivo:} desempenha dois papéis críticos -- achata a curva de ENR vs frequência \emph{e} mascara a variação de impedância entre os dois estados, mantendo VSWR baixo em hot e em cold;
\item \emph{Saída casada:} conector $50\,\Omega$ rastreável.
\end{itemize}

\end{frame}

% ============================================

\begin{frame}{O que torna a fonte cara: faixa de frequência}

A faixa de frequência domina o preço de catálogo:
\begin{itemize}
\item DC--1 GHz: dezenas de dólares (kits de bancada e didáticos);
\item DC--18 GHz: alguns milhares de dólares (Keysight 346B, NoiseCom NC346);
\item até 50 GHz: 10--30 k USD;
\item até 110 GHz: 60 k USD ou mais.
\end{itemize}

Conector também escala com a frequência: SMA é barato; $2{,}92$ mm e $1{,}85$ mm encarecem bastante o conjunto.

\end{frame}

% ============================================

\begin{frame}{Outros fatores e versão didática}

\textbf{Demais fatores de custo:}
\begin{itemize}
\item planura da ENR: variação $< 0{,}2$ dB ao longo da faixa exige ajuste fino do atenuador e do casamento;
\item VSWR baixo \emph{simultaneamente} em hot e em cold é difícil de garantir;
\item estabilidade térmica e compensação de drift do diodo;
\item rastreabilidade metrológica contra padrão primário (NIST, NPL, INMETRO).
\end{itemize}

\textbf{Versão didática.} Um diodo Zener BZX79 reverso-polarizado seguido de um amplificador discreto entrega ENR utilizável até $\sim 500$ MHz por menos de R\$~50 -- o suficiente para um experimento de método Y em sala de aula.

\end{frame}

\section{Perdas de transmissão}

\begin{frame}{Modelo de perda de canal}

Um enlace com perda $L > 1$ (adimensional) entrega:
\[
P_r = \frac{P_t}{L}
\]
ao receptor. Em dB, $L_{\text{dB}} = 10\log_{10} L$ se subtrai diretamente.

\vspace{0.3cm}

\textbf{Hipótese:} ruído é adicionado no receptor; a potência de ruído na entrada do receptor não depende de $L$.

\vspace{0.3cm}

\textbf{Efeito na SNR de referência:}
\[
\gamma = \frac{P_r}{N_0 W} = \frac{P_t / L}{N_0 W}
\]
Cada decibel a mais de perda significa um decibel a menos de $\gamma$.

\end{frame}

% ============================================

\begin{frame}[allowframebreaks]{Modelo de perda no espaço livre}

Em visada direta entre duas antenas, a equação de Friis dá a potência recebida:
\[
P_r = P_t \, G_t \, G_r \left( \frac{\lambda}{4 \pi d} \right)^2
\]
com $\lambda = c/f$ o comprimento de onda e $d$ a distância entre as antenas.

\vspace{0.2cm}

\textbf{Perda isotrópica de percurso:} é a perda definida com antenas isotrópicas, $G_t = G_r = 1$,
\[
L_{\text{fs}} = \left( \frac{4 \pi d}{\lambda} \right)^2 = \left( \frac{4 \pi d f}{c} \right)^2
\]

\textbf{Forma prática em dB:}
\[
L_{\text{fs,dB}} = 92{,}45 + 20\log_{10} d_{\text{km}} + 20\log_{10} f_{\text{GHz}}
\]

\vspace{0.2cm}

Cada duplicação da distância acrescenta $6$ dB de perda; o mesmo vale para a frequência. A perda cresce com $f^2$ porque a abertura efetiva de uma antena isotrópica diminui com $\lambda^2$.

\end{frame}

% ============================================

\begin{frame}[allowframebreaks]{Atenuador passivo: derivação de $G$, $F$ e $T_e$}

\textbf{Cenário.} Atenuador passivo, casado dos dois lados, em equilíbrio térmico a $T_0$. A perda é $L > 1$ e a fonte na entrada também está a $T_0$.

\textbf{Ganho.} O atenuador divide a potência do sinal por $L$, logo $G = 1/L$.

\textbf{Ruído interno.} As perdas vêm de resistores dissipativos que geram ruído de Johnson. Pelo teorema de Nyquist, o ruído disponível na saída de qualquer rede passiva casada a $T_0$ vale $kT_0 B$. Decompondo em fonte atenuada mais ruído interno:
\[
\underbrace{\frac{kT_0 B}{L}}_{\text{fonte atenuada}} \;+\; \underbrace{N_{\text{int}}}_{\text{ruído interno}} \;=\; kT_0 B
\quad\Longrightarrow\quad
N_{\text{int}} = kT_0 B \, \frac{L-1}{L}
\]

\textbf{Figura de ruído.} Com fonte a $T_0$ na entrada,
\[
F = \frac{(S/N)_{\text{in}}}{(S/N)_{\text{out}}} = \frac{S/(kT_0 B)}{(S/L)/(kT_0 B)} = L
\]

\textbf{Temperatura de ruído equivalente.} $T_e$ é a temperatura fictícia na entrada que, amplificada por $G$, reproduz o ruído interno na saída:
\[
G \, k T_e B = N_{\text{int}}
\;\Longrightarrow\;
\frac{k T_e B}{L} = kT_0 B \, \frac{L-1}{L}
\;\Longrightarrow\;
T_e = (L-1)\, T_0
\]

A relação geral $F = 1 + T_e/T_0$ devolve $F = 1 + (L-1) = L$, confirmando a consistência. As três fórmulas vêm da mesma ideia: o equilíbrio térmico a $T_0$ obriga o ruído de saída a valer $kT_0 B$.

\end{frame}

% ============================================

\begin{frame}{Atenuador como bloco da cascata Friis}

Um atenuador passivo à temperatura $T_0$ se comporta como um bloco com:
\[
G = \frac{1}{L}, \qquad F = L, \qquad T_e = (L - 1) T_0
\]

\textbf{Consequência grave:} se a perda do trecho entre antena e LNA é $L = 3$ dB, a figura efetiva já parte de $F = 2$, antes mesmo do LNA.

\vspace{0.3cm}

\begin{block}{Regra prática}
Colocar o LNA \textbf{antes} de qualquer atenuação significativa: junto à antena, cabos curtos, conectores limpos.
\end{block}

\end{frame}

% ============================================

\begin{frame}{Aplicação: orçamento de enlace via satélite}

\textbf{Enunciado.} Enlace descendente GEO:
\begin{itemize}
\item $f_c = 4$ GHz, distância $d = 36\,000$ km;
\item $P_t = 100$ W, antenas $G_t = G_r = 30$ dBi;
\item receptor com $T_{\text{sys}} = 100$ K, banda $B = 1$ MHz.
\end{itemize}
Calcule $P_r$ e $\gamma$.

% ============================================

\begin{frame}{Resolução: potência recebida}

\textbf{Passo 1.} Comprimento de onda:
\[
\lambda = \frac{c}{f} = \frac{3\times 10^{8}}{4\times 10^{9}} = 0{,}075 \text{ m}
\]

\textbf{Passo 2.} Perda de espaço livre:
\[
L_{\text{FS}}|_{\text{dB}} = 20\log_{10}\!\left(\frac{4\pi d}{\lambda}\right) \approx 195{,}6 \text{ dB}
\]

\textbf{Passo 3.} Potência recebida em dBm:
\[
P_r = P_t + G_t + G_r - L_{\text{FS}} = 50 + 30 + 30 - 195{,}6 \approx -85{,}6 \text{ dBm}
\]

\end{frame}

% ============================================

\begin{frame}{Resolução: piso de ruído e SNR}

\textbf{Passo 4.} Piso de ruído com $T_{\text{sys}} = 100$ K, $B = 1$ MHz:
\[
P_{\text{noise}} = k T_{\text{sys}} B = 1{,}38\times 10^{-23} \cdot 100 \cdot 10^{6}
\]
\[
\approx 1{,}38\times 10^{-15} \text{ W} \approx -118{,}6 \text{ dBm}
\]

\textbf{Passo 5.} SNR:
\[
\gamma = P_r - P_{\text{noise}} \approx -85{,}6 - (-118{,}6) = 33 \text{ dB}
\]

\begin{block}{Observação}
Margem confortável para sistemas digitais; para FM analógico com $\beta$ alto, ainda é necessário verificar o limiar.
\end{block}

\end{frame}

\section{Repetidores}

\begin{frame}{Repetidor analógico amplificador}

Para enlaces longos, a perda total inviabiliza uma única travessia. Solução: \textbf{repetidores}.

\vspace{0.3cm}

\textbf{Repetidor analógico} amplifica sinal e ruído juntos. Em $n$ saltos com $L$ e $F$ iguais:
\[
F_{\text{tot}} \approx n \cdot F
\]

A SNR final degrada \textbf{aproximadamente $10\log_{10} n$ dB} pela acumulação de ruído.

\vspace{0.3cm}

\textbf{Limitação fundamental:} o número de saltos é limitado pela SNR mínima aceitável no destino.

\end{frame}

% ============================================

\begin{frame}{Repetidor regenerativo}

Em sistemas digitais, o repetidor \textbf{detecta} os símbolos, \textbf{regenera} pulsos limpos e retransmite.

\vspace{0.3cm}

\begin{itemize}
\item o ruído \textbf{não se acumula} entre saltos, desde que cada detecção esteja com baixa probabilidade de erro;
\item a degradação fica limitada pelas (poucas) decisões erradas ao longo da cadeia.
\end{itemize}

\vspace{0.3cm}

\begin{block}{Por que digital ganhou}
O repetidor regenerativo é uma das principais razões pelas quais as redes de longa distância migraram de analógico para digital ao longo do século XX.
\end{block}

\end{frame}

% ============================================

\begin{frame}{Aplicação: enlace com 5 saltos}

\textbf{Enunciado.} Compare a SNR final em um enlace de longa distância com $n = 5$ saltos iguais, cada um com perda $L = 30$ dB e repetidor com $\mathrm{NF} = 4$ dB e $G = 30$ dB:
\begin{enumerate}
\item repetidor analógico amplificador;
\item repetidor digital regenerativo com BER por salto de $10^{-9}$.
\end{enumerate}

\begin{frame}{Resolução: caso analógico}

\textbf{Passo A1.} Cada repetidor compensa exatamente a perda do trecho ($G = L$), mas o ruído de cada estágio é amplificado pelos seguintes.

\textbf{Passo A2.} A figura total da cadeia cresce aproximadamente linear em $n$:
\[
F_{\text{tot}} \approx n \cdot F = 5 \cdot 10^{0{,}4} \approx 12{,}6 \Rightarrow \approx 11 \text{ dB}
\]

A SNR final degrada cerca de $7$ dB em relação a um único salto.

\end{frame}

% ============================================

\begin{frame}{Resolução: caso digital regenerativo}

\textbf{Passo D1.} A cada salto, o detector regenera o símbolo. O ruído \emph{não acumula em amplitude}.

\textbf{Passo D2.} Apenas erros somam:
\[
\text{BER}_{\text{tot}} \le n \cdot \text{BER} = 5 \times 10^{-9}
\]
Degradação desprezível.

\begin{block}{Conclusão}
$5$ saltos analógicos perdem cerca de $7$ dB de SNR; o equivalente digital regenerativo permanece praticamente intacto.
\end{block}

\end{frame}

\section{Resumo do capítulo}

\begin{frame}{Síntese: SNR de cada sistema}

Para $\gamma = P_r/(N_0 W)$ fixo:

\begin{itemize}
\item Banda base: $(S/N)_o = \gamma$.
\item DSB-SC coerente: $(S/N)_o = \gamma$.
\item SSB coerente: $(S/N)_o = \gamma$.
\item AM convencional, envelope: $(S/N)_o \le \gamma/3$, com efeito de limiar.
\item FM, $\beta$ (tom): $(S/N)_o = \tfrac{3}{2}\beta^2 \gamma$, com efeito de limiar.
\item Pré/pós-ênfase em FM acrescenta $\sim 13$ dB para áudio.
\end{itemize}

\end{frame}

% ============================================

\begin{frame}{Síntese: ruído de hardware e enlace}

\begin{itemize}
\item Ruído térmico: $N_0 = kT$; piso de $-174$ dBm/Hz a $290$ K.
\item Figura de ruído $F$ e temperatura $T_e = (F-1)T_0$.
\item Friis: $F_{\text{tot}} = F_1 + (F_2-1)/G_1 + \cdots$; primeiro estágio domina.
\item Perdas reduzem $\gamma$ na mesma quantidade em dB.
\item Atenuador passivo equivale a $F = L$, $T_e = (L-1)T_0$.
\item Repetidor analógico acumula ruído; regenerativo digital não.
\end{itemize}

\vspace{0.3cm}

\textbf{Mensagem final:} projetar um receptor analógico significa equilibrar três coisas: \textbf{potência}, \textbf{banda} e \textbf{ruído}. Cada esquema de modulação escolhe um ponto diferente nesse triângulo.

\end{frame}
