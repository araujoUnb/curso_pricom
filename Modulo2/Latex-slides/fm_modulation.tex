% MODULAÇÃO ANGULAR: FM E PM
% ============================================

\section{Introdução}

\begin{frame}{Modulação Angular vs. Modulação em Amplitude}

\textbf{AM:} amplitude varia, frequência e fase fixas
\[
s_{\AM}(t) = A(t) \cos(2\pi f_c t)
\]

\vspace{0.2cm}
\textbf{Modulação angular:} amplitude fixa, ângulo varia
\[
s(t) = A_c \cos[\theta(t)] = A_c \cos[2\pi f_c t + \phi(t)]
\]

onde $\phi(t)$ é a fase modulada.

\vspace{0.3cm}
\textbf{Duas variantes:}
\begin{itemize}
\item \textbf{FM} -- frequência instantânea varia com $m(t)$.
\item \textbf{PM} -- fase instantânea varia com $m(t)$.
\end{itemize}

\end{frame}

% ============================================

\begin{frame}{Fase e Frequência Instantâneas}

Para sinal modulado $s(t) = A_c \cos[\theta(t)]$:

\begin{block}{Definições}
\textbf{Fase instantânea:}
\[
\theta(t) = 2\pi f_c t + \phi(t)
\]
\textbf{Frequência instantânea:}
\[
f_i(t) = \frac{1}{2\pi} \frac{d\theta(t)}{dt} = f_c + \frac{1}{2\pi}\frac{d\phi(t)}{dt}
\]
\end{block}

\small
\textbf{Interpretação:} $\theta(t)$ é o argumento do cosseno; $f_i(t)$ é a taxa instantânea de oscilação; $f_c$ é a portadora não modulada; $\phi(t)$ é o desvio de fase causado pela mensagem.

\end{frame}

% ============================================

\section{Modulação em Frequência (FM)}

\begin{frame}{Definição de FM}

\begin{block}{Modulação em Frequência (FM)}
A frequência instantânea é proporcional à mensagem:
\[
f_i(t) = f_c + k_f m(t)
\]
onde $k_f$ é a constante de sensibilidade de frequência, em Hz/V.
\end{block}

\textbf{Desvio de frequência instantâneo:}
\[
\Delta f(t) = k_f m(t)
\]

\textbf{Desvio máximo de frequência:}
\[
\Delta f = k_f \max|m(t)|
\]

\end{frame}

% ============================================

\begin{frame}{Da frequência à fase: integração}

A partir da definição de frequência instantânea,
\[
\frac{d\theta(t)}{dt} = 2\pi f_i(t) = 2\pi[f_c + k_f m(t)].
\]

\pause
Integrando dos dois lados, e identificando $\phi(t)$:
\[
\frac{d\phi(t)}{dt} = 2\pi k_f m(t)
\]
\[
\Rightarrow \quad \phi(t) = 2\pi k_f \int_{-\infty}^{t} m(\tau)\,d\tau
\]

\pause
\textbf{Conclusão:} a fase do FM é a integral da mensagem, escalada por $2\pi k_f$.

\end{frame}

% ============================================

\begin{frame}{Sinal FM no Domínio do Tempo}

\begin{block}{Sinal FM}
\[
s_{\FM}(t) = A_c \cos\left[2\pi f_c t + 2\pi k_f \int_{-\infty}^{t} m(\tau)\,d\tau\right]
\]
\end{block}

Definindo o \textbf{índice de fase} $\beta(t) = 2\pi k_f \int_{-\infty}^{t} m(\tau)\,d\tau$:
\[
s_{\FM}(t) = A_c \cos[2\pi f_c t + \beta(t)]
\]

\pause
\small
\textbf{Características:} amplitude constante $|s_{\FM}(t)| = A_c$, toda a informação na fase, potência transmitida fixa $P_s = A_c^2/2$, imunidade a variações de amplitude por ruído ou desvanecimento.

\end{frame}

% ============================================

\begin{frame}{Modulação em Fase (PM)}

\begin{block}{Modulação em Fase (PM)}
A fase instantânea é diretamente proporcional à mensagem:
\[
\phi(t) = k_p m(t)
\]
onde $k_p$ é a constante de sensibilidade de fase, em rad/V.
\end{block}

\textbf{Sinal PM:}
\[
s_{\PM}(t) = A_c \cos[2\pi f_c t + k_p m(t)]
\]

\pause
\textbf{Frequência instantânea:}
\[
f_i(t) = f_c + \frac{1}{2\pi}\frac{d\phi(t)}{dt} = f_c + \frac{k_p}{2\pi} \frac{dm(t)}{dt}
\]

\end{frame}

% ============================================

\begin{frame}{FM vs.\ PM: comparação direta}

\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|l|c|c|}
\hline
                        & \textbf{FM}                       & \textbf{PM} \\
\hline
Fase $\phi(t)$          & $\propto \int m(t)\,dt$           & $\propto m(t)$ \\
\hline
Frequência $f_i(t)$     & $\propto m(t)$                    & $\propto dm(t)/dt$ \\
\hline
\end{tabular}
\end{center}

\vspace{0.4cm}
\textbf{Resumo:}
\begin{itemize}
\item Em FM, a frequência é a cópia direta da mensagem.
\item Em PM, é a fase que copia diretamente a mensagem.
\item As duas são equivalentes a menos de uma integração ou diferenciação.
\end{itemize}

\end{frame}

% ============================================

\begin{frame}{Relação entre FM e PM}

\textbf{FM pode ser vista como PM do integral da mensagem:}
\[
\text{FM de } m(t) \;\equiv\; \text{PM de } \int m(t)\,dt
\]

\pause
\textbf{PM pode ser vista como FM da derivada da mensagem:}
\[
\text{PM de } m(t) \;\equiv\; \text{FM de } \frac{dm(t)}{dt}
\]

\vspace{0.4cm}
\pause
Na prática, FM é mais usada do que PM por ter melhor desempenho em ruído.

\end{frame}

\section{Análise Espectral de FM}

\begin{frame}{FM com Tom Único: substituição}

\textbf{Sinal modulante:} $m(t) = A_m \cos(2\pi f_m t)$.

\textbf{Sinal FM:}
\[
s(t) = A_c \cos\!\left[2\pi f_c t + 2\pi k_f A_m\!\!\int_{-\infty}^{t}\!\! \cos(2\pi f_m \tau)\,d\tau\right]
\]

\pause
\textbf{Integral elementar:}
\[
\int \cos(2\pi f_m \tau)\,d\tau = \frac{\sin(2\pi f_m t)}{2\pi f_m}
\]

\pause
\textbf{Substituindo:}
\[
s(t) = A_c \cos\!\left[2\pi f_c t + \frac{k_f A_m}{f_m}\sin(2\pi f_m t)\right]
\]

\end{frame}

% ============================================

\begin{frame}{Índice de modulação $\beta$}

\textbf{Definição:}
\[
\boxed{\;\beta \;=\; \frac{k_f A_m}{f_m} \;=\; \frac{\Delta f}{f_m}\;}
\]

\pause
\textbf{Forma canônica do FM com tom único:}
\[
\boxed{\;s(t) = A_c \cos[2\pi f_c t + \beta \sin(2\pi f_m t)]\;}
\]

\vspace{0.3cm}
\pause
\textbf{Duas leituras de $\beta$:}
\begin{itemize}
\item Pico do desvio de fase, em radianos.
\item Razão entre desvio máximo de frequência e frequência da mensagem.
\end{itemize}

\end{frame}

% ============================================

\begin{frame}{Expansão de Bessel: passo 1 -- série de Fourier}

\textbf{Ideia:} a exponencial $g(t) = e^{j\beta\sin(\omega_m t)}$ é periódica de período $T_m = 2\pi/\omega_m$.

\pause
Toda função periódica admite série de Fourier:
\[
e^{j\beta\sin(\omega_m t)} = \sum_{n=-\infty}^{\infty} c_n\, e^{jn\omega_m t}
\]

\pause
\textbf{Coeficientes:}
\[
c_n = \frac{1}{T_m}\int_{0}^{T_m}\! e^{j[\beta\sin(\omega_m t) - n\omega_m t]}\,dt
\]

\pause
\textbf{Substituindo} $\theta = \omega_m t$, com $d\theta = \omega_m\,dt$:
\[
c_n = \frac{1}{2\pi}\int_{0}^{2\pi}\! e^{j[\beta\sin\theta - n\theta]}\,d\theta
\]

\end{frame}

% ============================================

\begin{frame}{Expansão de Bessel: passo 2 -- $c_n = J_n(\beta)$}

\textbf{Definição integral da função de Bessel de primeira espécie:}
\[
\boxed{\,J_n(\beta) \;\triangleq\; \frac{1}{2\pi}\int_{-\pi}^{\pi}\! e^{j[\beta\sin\theta - n\theta]}\,d\theta\,}
\]

\pause
\textbf{Logo:} $c_n = J_n(\beta)$, e
\[
e^{j\beta\sin(\omega_m t)} = \sum_{n=-\infty}^{\infty} J_n(\beta)\, e^{jn\omega_m t}
\]

\pause
\textbf{Tomando partes real e imaginária} via $e^{jx} = \cos x + j\sin x$:
\begin{align*}
\cos[\beta\sin(\omega_m t)] &= \sum_{n} J_n(\beta)\cos(n\omega_m t)\\
\sin[\beta\sin(\omega_m t)] &= \sum_{n} J_n(\beta)\sin(n\omega_m t)
\end{align*}

\end{frame}

% ============================================

\begin{frame}{Expansão de Bessel: passo 3 -- simetria}

\textbf{Propriedade:} $J_{-n}(\beta) = (-1)^n J_n(\beta)$.

\vspace{0.2cm}
Combinando os termos $n$ e $-n$ no cosseno: sobrevivem só os \emph{pares}.

\[
\boxed{\,\cos[\beta\sin(\omega_m t)] = J_0(\beta) + 2\!\sum_{n=1}^{\infty}\! J_{2n}(\beta)\cos(2n\omega_m t)\,}
\]

\pause
Combinando os termos $n$ e $-n$ no seno: sobrevivem só os \emph{ímpares}.

\[
\boxed{\,\sin[\beta\sin(\omega_m t)] = 2\!\sum_{n=0}^{\infty}\! J_{2n+1}(\beta)\sin[(2n+1)\omega_m t]\,}
\]

\pause
Estas são as identidades de Jacobi--Anger, obtidas aqui como consequência direta da série de Fourier.

\end{frame}

% ============================================

\begin{frame}{Expansão de Bessel: passo 4 -- aplicação ao FM}

Aplicando $\cos(A+B) = \cos A\cos B - \sin A\sin B$ no sinal FM:
\[
s(t)/A_c = \cos(2\pi f_c t)\cos[\beta\sin(2\pi f_m t)] - \sin(2\pi f_c t)\sin[\beta\sin(2\pi f_m t)]
\]

\pause
Substituindo as duas expansões e usando produtos trigonométricos
\[
\cos a\cos b = \tfrac{1}{2}[\cos(a-b)+\cos(a+b)],\quad
\sin a\sin b = \tfrac{1}{2}[\cos(a-b)-\cos(a+b)],
\]
todos os termos viram cossenos em $f_c \pm k f_m$.

\pause
\begin{block}{Resultado}
\[
s(t) = A_c\!\sum_{n=-\infty}^{\infty}\! J_n(\beta)\,\cos[2\pi(f_c + n f_m)t]
\]
\end{block}

\end{frame}

% ============================================

\begin{frame}{Espectro FM: Resultado Final}

\[
s(t) = A_c \sum_{n=-\infty}^{\infty} J_n(\beta)\,\cos[2\pi(f_c + nf_m)t]
\]

\textbf{Interpretação:}
\begin{itemize}\itemsep0pt
\item Portadora em $f_c$ com amplitude $A_c J_0(\beta)$.
\item Bandas laterais em $f_c \pm nf_m$ com amplitude $A_c J_n(\beta)$.
\item Infinitas componentes na teoria; na prática, $J_n(\beta)\approx 0$ para $n > \beta + 1$.
\end{itemize}

\vspace{0.2cm}
\textbf{Diferença fundamental para AM:} AM tem apenas duas bandas laterais USB e LSB; FM tem infinitas, mas só algumas relevantes.

\end{frame}

% ============================================

\begin{frame}{Funções de Bessel}

\textbf{Propriedades importantes:}

\begin{itemize}
\item $J_{-n}(\beta) = (-1)^n J_n(\beta)$ (simetria)
\item $J_n(-\beta) = (-1)^n J_n(\beta)$
\item $\sum_{n=-\infty}^{\infty} J_n^2(\beta) = 1$ (conservação de potência!)
\item Para $\beta \ll 1$: $J_0(\beta) \approx 1$, $J_1(\beta) \approx \beta/2$, $J_n(\beta) \approx 0$ para $n \geq 2$
\end{itemize}

\vspace{0.3cm}

\textbf{Valores típicos:}

\begin{center}
\footnotesize

\hline

\hline
0.5 & 0.94 & 0.24 & 0.03 & 0.00 & - & - \\
\hline
1.0 & 0.77 & 0.44 & 0.11 & 0.02 & 0.00 & - \\
\hline
2.0 & 0.22 & 0.58 & 0.35 & 0.13 & 0.03 & 0.01 \\
\hline
5.0 & -0.18 & -0.33 & 0.05 & 0.36 & 0.39 & 0.26 \\
\hline
\end{tabular}
\end{center}

\end{frame}

\begin{frame}{Exercício: Espectro FM com Mensagem de Dois Tons}

\footnotesize
\textbf{Dados:} $m(t) = A_1\cos(2\pi f_1 t) + A_2\cos(2\pi f_2 t)$, com $f_1 = 1$\,kHz, $f_2 = 10$\,kHz, $\beta_1 = 1$, $\beta_2 = 0.5$, $A_c = 1$\,V, $f_c = 100$\,kHz.

\textbf{Sinal FM:}
\[
s(t) = A_c \cos[2\pi f_c t + \beta_1\sin(2\pi f_1 t) + \beta_2\sin(2\pi f_2 t)]
\]

\pause
\textbf{Passo 1.} Forma exponencial, fatorando os três termos da fase:
\[
s(t) = A_c\,\Re\!\left\{ e^{j2\pi f_c t}\, e^{j\beta_1\sin(2\pi f_1 t)}\, e^{j\beta_2\sin(2\pi f_2 t)} \right\}
\]

\pause
\textbf{Passo 2.} Identidade de Jacobi--Anger em cada exponencial periódica, $k = 1, 2$:
\[
e^{j\beta_k\sin(2\pi f_k t)} = \sum_{n=-\infty}^{\infty} J_n(\beta_k)\,e^{j2\pi n f_k t}
\]

\end{frame}

% ============================================

\begin{frame}{Dois Tons: Derivação (continuação)}

\footnotesize
\textbf{Passo 3.} Substituir as duas séries e agrupar exponenciais em uma única soma dupla:
\[
s(t) = A_c\,\Re\!\left\{ \sum_{m,n} J_m(\beta_1)\, J_n(\beta_2)\, e^{j2\pi(f_c + m f_1 + n f_2)t} \right\}
\]

\pause
\textbf{Passo 4.} Tomar a parte real do somatório:
\[
\boxed{\;s(t) = A_c\!\!\sum_{m,n}\!\! J_m(\beta_1)\, J_n(\beta_2)\, \cos[2\pi(f_c + m f_1 + n f_2)t]\;}
\]

\pause
\vspace{0.2cm}
\textbf{Leitura do resultado:}
\begin{itemize}\itemsep0pt
\item Cada par de inteiros $(m, n)$ contribui com uma componente em $f_c + m f_1 + n f_2$.
\item A amplitude é o produto $A_c\, J_m(\beta_1)\, J_n(\beta_2)$, fatorada por modulação.
\item Surgem produtos de intermodulação $f_c \pm m f_1 \pm n f_2$ ausentes em sistemas lineares como AM.
\end{itemize}

\end{frame}

% ============================================

\footnotesize

Com $J_m(1) = 0.77;\ 0.44;\ 0.11$ e $J_n(0.5) = 0.94;\ 0.24$, $A_c = 1$\,V:

\end{frame}

\begin{frame}{Largura de Banda de FM}

\textbf{Problema:} Teoricamente, espectro FM tem largura infinita!

Na prática: Bandas com $|J_n(\beta)| < 0.01$ são desprezíveis.

\vspace{0.3cm}

\textbf{Número significativo de bandas laterais:}

Aproximadamente $n_{\max} \approx \beta + 1$

\textbf{Largura de banda aproximada:}
\[
B \approx 2n_{\max} f_m = 2(\beta + 1)f_m
\]

\begin{block}{Regra de Carson}
\[
\boxed{B_{\FM} \approx 2(\Delta f + f_m) = 2f_m(\beta + 1)}
\]
\end{block}

onde $\Delta f = k_f A_m$ é o desvio de frequência.

\textbf{Precisão:} Contém $\approx 98\%$ da potência do sinal FM.

\end{frame}

\end{frame}

\begin{frame}{NBFM vs. WBFM}

\textbf{Dois regimes de operação:}

\begin{enumerate}
\item \textbf{NBFM (Narrowband FM):} $\beta \ll 1$
   \begin{itemize}
   \item $B \approx 2f_m$ (similar a AM)
   \item Poucas bandas laterais significativas
   \item Comportamento quase linear
   \end{itemize}

\item \textbf{WBFM (Wideband FM):} $\beta \gg 1$
   \begin{itemize}
   \item $B \approx 2\Delta f$ (dominado pelo desvio)
   \item Muitas bandas laterais
   \item Melhor desempenho em ruído
   \end{itemize}
\end{enumerate}

\vspace{0.3cm}

\textbf{Comparação com AM:}

\begin{itemize}
\item AM: $B = 2W$ (independente de $A_c$ ou $\mu$)
\item FM: $B = 2\Delta f(1 + 1/\beta)$ (depende de $\Delta f$!)
%\item FM troca largura de banda por melhor SNR
\end{itemize}

\end{frame}

\begin{frame}{Visualização: NBFM vs. WBFM}

\end{frame}

\begin{frame}{Exemplo 1: FM com Tom Único}

\textbf{Dados:} $m(t) = \cos(2\pi\cdot 5000\,t)$, $\Delta f = 75$\,kHz, $f_c = 100$\,MHz.

\pause
\textbf{Índice de modulação:}
\[
\beta = \frac{\Delta f}{f_m} = \frac{75\,\text{kHz}}{5\,\text{kHz}} = 15
\]

\pause
\textbf{Largura de banda por Carson:}
\[
B \approx 2(\Delta f + f_m) = 2(75 + 5) = 160\,\text{kHz}
\]

\pause
\textbf{Bandas significativas:} $n_{\max} \approx \beta + 1 = 16$, ou seja, 16 pares de bandas laterais.

\textbf{Classificação:} $\beta = 15 \gg 1$, WBFM.

\end{frame}

\begin{frame}{Exemplo 2: Rádio FM Comercial}

\textbf{Padrão FCC para FM broadcast:}

\begin{itemize}
\item Desvio máximo: $\Delta f = 75$ kHz
\item Frequência máxima de áudio: $f_m = 15$ kHz
\item Alocação de canal: 200 kHz
\end{itemize}

\textbf{Índice de modulação:}
\[
\beta = \frac{75 \text{ kHz}}{15 \text{ kHz}} = 5
\]

\textbf{Largura de banda necessária:}
\[
B = 2(\Delta f + f_m) = 2(75 + 15) = 180 \text{ kHz}
\]

\textbf{Observação:} Canal de 200 kHz acomoda 180 kHz + bandas de guarda

\vspace{0.3cm}

\textbf{Banda FM:} 88-108 MHz, espaçamento de 200 kHz

Número de canais: $(108-88)/0.2 = 100$ canais

\end{frame}

\section{Demodulação FM}

\begin{frame}{Princípios de Demodulação FM}

\textbf{Objetivo:} Recuperar $m(t)$ de $s_{\FM}(t) = A_c \cos[2\pi f_c t + 2\pi k_f \int m(\tau)d\tau]$

\vspace{0.3cm}

\textbf{Métodos principais:}

\begin{enumerate}
\item \textbf{Discriminador de frequência (slope detector)}
   \begin{itemize}
   \item Converte FM em AM, depois detecta envelope
   \end{itemize}

\item \textbf{Detector PLL (Phase-Locked Loop)}
   \begin{itemize}
   \item VCO "segue" a frequência de entrada
   \item Saída do VCO é proporcional a $m(t)$
   \end{itemize}

\item \textbf{Detector de Foster-Seeley}
   \begin{itemize}
   \item Usa transformador balanceado
   \end{itemize}

\item \textbf{Detector de razão (ratio detector)}
   \begin{itemize}
   \item Variante do Foster-Seeley
   \item Menos sensível a variações de amplitude
   \end{itemize}
\end{enumerate}

\end{frame}

% ============================================

\begin{frame}{Discriminador de Frequência: princípio}

\textbf{Princípio:} converter variação de frequência em variação de amplitude e depois usar detector de envelope.

\begin{center}
\begin{tikzpicture}[>=latex, scale=0.95]
\node[draw, rectangle, minimum width=2cm, minimum height=0.8cm] (der) at (0,0) {Derivador};
\node[draw, rectangle, minimum width=2.4cm, minimum height=0.8cm] (env) at (3.4,0) {Det.\ envelope};

\draw[->] (-2,0) -- node[above, font=\small] {$s_{\FM}(t)$} (der);
\draw[->] (der) -- node[above, font=\small] {AM} (env);
\draw[->] (env) -- node[above, font=\small] {$\hat{m}(t)$} (5.8,0);
\end{tikzpicture}
\end{center}

\pause
\textbf{Aplicando $d/dt$:}
\[
\frac{ds_{\FM}(t)}{dt} = -A_c \cdot 2\pi f_i(t)\,\sin[2\pi f_c t + \phi(t)]
\]

\pause
\textbf{Amplitude do sinal derivado:}
\[
A(t) = 2\pi A_c f_i(t) = 2\pi A_c [f_c + k_f m(t)]
\]

\end{frame}

% ============================================

\begin{frame}{Discriminador: saída e limitação}

\textbf{Após detector de envelope e bloqueio DC:}
\[
\hat{m}(t) = K\cdot k_f\, m(t)
\]

\vspace{0.3cm}
\pause
\textbf{Problema:} o derivador também responde a variações de amplitude. Ruído de amplitude do canal contamina a saída do detector de envelope.

\vspace{0.3cm}
\pause
\textbf{Solução:} colocar um limitador de amplitude antes do discriminador. Como em FM toda informação está na frequência, recortar amplitude não destrói a mensagem.

\end{frame}

% ============================================

\end{frame}

\begin{frame}{Derivador como filtro linear em frequência}

\small
\textbf{Operação no tempo:} $y(t) = \dfrac{ds(t)}{dt}$.

\pause
\textbf{No domínio da frequência:}
\[
Y(f) = j2\pi f\, S(f) \;\Longrightarrow\; \boxed{\,H_d(f) = j2\pi f\,} \;\Longrightarrow\; |H_d(f)| = 2\pi|f|.
\]

\pause
\begin{center}
\begin{tikzpicture}[>=latex, scale=0.55]
\draw[->] (-3.2,0) -- (3.4,0) node[right, font=\scriptsize] {$f$};
\draw[->] (0,-0.2) -- (0,2.5) node[above, font=\scriptsize] {$|H_d(f)|$};
\draw[thick, blue] (-2.5,2.2) -- (0,0) -- (2.5,2.2);
\node[font=\tiny, blue] at (2.5,2.5) {$2\pi|f|$};
\end{tikzpicture}
\end{center}

A linearidade em $f$ produz, na saída do envelope, tensão proporcional a $f_i(t) = f_c + k_f m(t)$.

\end{frame}

% ============================================

\begin{frame}{Aproximação na banda do FM}

\textbf{Banda ocupada pelo FM:} $B_T = 2(\Delta f + f_m)$ em torno de $f_c$. Basta o filtro ser linear nessa janela.

\pause
\vspace{0.15cm}
\textbf{Linearização local em $f = f_c + \delta$:}
\[
|H_d(f_c + \delta)| = 2\pi f_c + 2\pi \delta
\]

A parcela $2\pi f_c$ é DC após o envelope; $2\pi\delta = 2\pi k_f m(t)$ é a mensagem.

\pause
\vspace{0.15cm}
\textbf{Por que não usar $H_d(f) = j2\pi f$ exato?}
\begin{itemize}\itemsep0pt
\item Ganho cresce sem limite e amplifica ruído fora da banda.
\item Não é realizável com componentes passivos.
\end{itemize}

\pause
\textbf{Solução prática:} aproximar $H_d(f)$ apenas dentro de $B_T$, com um passa-faixas cujo flanco seja amplitude-linear em torno de $f_c$.

\end{frame}

\begin{frame}{Slope detector com circuito ressonante}

\small
\textbf{Tank LC com $Q$ e ressonância em $f_0$:}
\[
|H(f)| = \dfrac{1}{\sqrt{1 + Q^2\!\left(\dfrac{f}{f_0} - \dfrac{f_0}{f}\right)^{\!2}}}
\]

\pause
\textbf{Ideia:} sintonizar $f_0$ ao lado de $f_c$, deixando $f_c$ no flanco do filtro.

\begin{center}
\begin{tikzpicture}[>=latex, scale=0.62]
\draw[->] (-0.2,0) -- (5.8,0) node[right, font=\scriptsize] {$f$};
\draw[->] (0,-0.2) -- (0,2.7) node[above, font=\scriptsize] {$|H(f)|$};
\draw[thick, blue] plot[domain=0.5:5.2, samples=120] (\x, {2.3/sqrt(1+30*(\x/3 - 3/\x)^2)});
\draw[dashed] (1.7,0) node[below, font=\tiny] {$f_c$} -- (1.7,1.05);
\draw[dashed] (3.0,0) node[below, font=\tiny] {$f_0$} -- (3.0,2.3);
\draw[red, very thick] (1.2,0.43) -- (2.3,1.85);
\node[red, font=\tiny, anchor=west] at (3.6,0.8) {flanco linear};
\end{tikzpicture}
\end{center}

Em $f \approx f_c$, o flanco aproxima a rampa $2\pi(f-f_c)$ do derivador ideal.

\end{frame}

% ============================================

\begin{frame}{Slope detector: análise da linearidade}

\textbf{Taylor de $|H(f)|$ em torno de $f_c$:}
\[
|H(f_c + \delta)| = a_0 + a_1 \delta + a_2 \delta^2 + a_3 \delta^3 + \cdots
\]

\pause
\vspace{0.1cm}
\textbf{Saída do envelope com $\delta = k_f m(t)$, após bloqueio DC:}
\[
\hat m(t) \propto a_1 k_f m(t) + a_2 k_f^2 m^2(t) + a_3 k_f^3 m^3(t) + \cdots
\]

\pause
\vspace{0.1cm}
\textbf{Termo desejado:} $a_1 k_f m(t)$.

\textbf{Distorção dominante:} $a_2 k_f^2 m^2(t)$, harmônico par da mensagem.

\pause
\vspace{0.1cm}
\textbf{Limitação do filtro único:} $a_2 \neq 0$. A saída só é razoavelmente fiel se $|m(t)|$ for pequeno o bastante para tornar o termo quadrático desprezível.

\end{frame}

% ============================================

\begin{frame}{Slope detector balanceado}

\small
\textbf{Estrutura:} dois filtros ressonantes em $f_0 \pm \Delta$, com envelopes subtraídos.

\begin{center}
\begin{tikzpicture}[>=latex, scale=0.62]
\draw[->] (-0.2,0) -- (6.4,0) node[right, font=\scriptsize] {$f$};
\draw[->] (0,-2.3) -- (0,2.7);
\draw[thick, blue] plot[domain=0.4:6.0, samples=160] (\x, {2.1/(1+((\x-2)/0.7)^2)});
\draw[thick, red]  plot[domain=0.4:6.0, samples=160] (\x, {2.1/(1+((\x-4)/0.7)^2)});
\draw[very thick, violet] plot[domain=0.4:6.0, samples=200] (\x, {2.1/(1+((\x-2)/0.7)^2) - 2.1/(1+((\x-4)/0.7)^2)});
\draw[gray, dashed] (-0.2,0) -- (6.0,0);
\draw[dashed] (3.0,-2.1) -- (3.0,2.2) node[above, font=\tiny] {$f_c$};
\draw[dashed] (2.0,0) node[below, font=\tiny] {$f_0\!-\!\Delta$} -- (2.0,2.1);
\draw[dashed] (4.0,0) node[below, font=\tiny] {$f_0\!+\!\Delta$} -- (4.0,2.1);
\node[blue,    font=\scriptsize] at (1.3,2.4) {$|H_1|$};
\node[red,     font=\scriptsize] at (4.7,2.4) {$|H_2|$};
\node[violet,  font=\scriptsize] at (5.6,-1.6) {$H_{\rm bal}$};
\end{tikzpicture}
\end{center}

\pause
\textbf{Resposta combinada:} $H_{\rm bal}(f) = |H_1(f)| - |H_2(f)|$, função ímpar em torno de $f_c$, cruzando zero exatamente em $f = f_c$ e linearizando o flanco.

\pause
\textbf{Consequência:} cancelam-se os termos pares de Taylor, $H_{\rm bal}(f_c + \delta) = 2 a_1 \delta + 2 a_3 \delta^3 + \cdots$, eliminando a distorção quadrática $a_2\delta^2$.

\end{frame}

% ============================================

\begin{frame}{Por que só os termos ímpares sobrevivem -- I}

\textbf{Simetria especular dos dois filtros:} $|H_2|$ é a imagem refletida de $|H_1|$ em torno de $f_c$. Em fórmula,
\[
|H_2(f_c + \delta)| = |H_1(f_c - \delta)|.
\]

\pause
\vspace{0.3cm}
\textbf{Expansão de Taylor de $|H_1|$ em $f_c$, escrita em variável genérica $u$:}
\[
|H_1(f_c + u)| = a_0 + a_1 u + a_2 u^2 + a_3 u^3 + a_4 u^4 + \cdots
\]

\pause
\vspace{0.3cm}
A próxima frame avalia esta série no ponto $u = -\delta$.

\end{frame}

% ============================================

\begin{frame}{Por que só os termos ímpares sobrevivem -- II}

\textbf{Substituindo $u = -\delta$,} cada potência fica
\[
(-\delta)^n = (-1)^n\,\delta^n.
\]

\pause
\vspace{0.2cm}
Termos pares conservam o sinal, $(-1)^{2k} = +1$; ímpares trocam, $(-1)^{2k+1} = -1$:
\[
|H_1(f_c - \delta)| = a_0 - a_1 \delta + a_2 \delta^2 - a_3 \delta^3 + a_4 \delta^4 - \cdots
\]

\pause
\vspace{0.2cm}
\textbf{Pela simetria especular,} $|H_2(f_c + \delta)| = |H_1(f_c - \delta)|$:
\[
|H_2(f_c + \delta)| = a_0 - a_1 \delta + a_2 \delta^2 - a_3 \delta^3 + a_4 \delta^4 - \cdots
\]

\end{frame}

% ============================================

\begin{frame}{Por que só os termos ímpares sobrevivem -- III}

\textbf{Subtraindo termo a termo} as duas séries:
\[
H_{\rm bal}(f_c + \delta) = \sum_{n \geq 0} a_n\,\bigl[1 - (-1)^n\bigr]\,\delta^n.
\]

\pause
\vspace{0.2cm}
\textbf{Fator $1 - (-1)^n$:}
\begin{itemize}\itemsep0pt
\item $n$ par: $1 - 1 = 0$, o termo cancela.
\item $n$ ímpar: $1 - (-1) = 2$, o termo dobra.
\end{itemize}

\pause
\vspace{0.3cm}
\[
\boxed{\;H_{\rm bal}(f_c + \delta) = 2 a_1 \delta + 2 a_3 \delta^3 + 2 a_5 \delta^5 + \cdots\;}
\]

A não-linearidade dominante passa a ser cúbica, $2 a_3 \delta^3$, muito menor que $a_2 \delta^2$ do filtro único.

\end{frame}

% ============================================

\begin{frame}{Exemplo numérico: definindo o filtro}

\footnotesize
\textbf{Passo 1 -- modelo do flanco:}
\[
|H(f)| = \frac{1}{\sqrt{1 + ((f-f_0)/W)^2}}, \quad W = 250\,\text{kHz}.
\]
$f_0$ é o pico, $W$ a meia-largura: em $f = f_0 \pm W$, $|H| = 1/\sqrt{2}$.

\pause
\textbf{Passo 2 -- adimensionalizar:} definindo $x \equiv (f - f_0)/W$, o filtro vira $|H| = 1/\sqrt{1 + x^2}$. A variável $x$ mede o afastamento do pico em unidades de $W$.

\begin{center}
\begin{tikzpicture}[>=latex, scale=0.45]
\draw[->] (-0.2,0) -- (5.6,0) node[right, font=\tiny] {$f$};
\draw[->] (0,-0.2) -- (0,2.4);
\draw[thick, blue] plot[domain=0.5:5.2, samples=120] (\x, {2.1/(1+((\x-3.0)/0.8)^2)});
\draw[dashed] (3.0,0) node[below, font=\tiny] {$f_0$} -- (3.0,2.1);
\draw[dashed] (3.8,0) node[below, font=\tiny] {$f_0\!+\!W$} -- (3.8,1.05);
\draw[dashed] (2.2,0) node[below, font=\tiny] {$f_0\!-\!W$} -- (2.2,1.05);
\end{tikzpicture}
\end{center}

\end{frame}

% ============================================

\begin{frame}{Exemplo numérico: porque $x_0 = -1$}

\small
\textbf{Passo 3 -- escolher onde colocar $f_c$.} Para que o filtro forneça uma resposta amplitude-linear, $f_c$ precisa cair no flanco, e não no pico. Escolhemos $f_c$ deslocado uma largura $W$ para a esquerda do pico:
\[
f_c \;=\; f_0 - W.
\]

\pause
\vspace{0.15cm}
\textbf{Passo 4 -- avaliar $x$ no ponto $f = f_c$.} Substituindo na definição $x = (f - f_0)/W$:
\[
x_0 \;=\; \left.x\right|_{f = f_c} \;=\; \frac{f_c - f_0}{W}.
\]

\pause
\textbf{Passo 5 -- usar a escolha do passo 3.} Como $f_c - f_0 = (f_0 - W) - f_0 = -W$:
\[
x_0 \;=\; \frac{-W}{W} \;=\; -1.
\]

\pause
\vspace{0.1cm}
\textbf{Leitura:} $x_0 = -1$ significa que $f_c$ está exatamente uma meia-largura abaixo do pico, no ponto onde o flanco do filtro tem boa linearidade local. É nesse ponto que vamos expandir $|H|$ em série de Taylor.

\end{frame}

% ============================================

\begin{frame}{Exemplo numérico: $a_3 \delta^3$ versus $a_2 \delta^2$}

\small
\textbf{Coeficientes de Taylor de $|H|$ em $f_c$,} obtidos derivando em $x_0 = -1$:
\[
a_1 = \frac{1}{2\sqrt{2}\,W}, \qquad
a_2 = \frac{1}{8\sqrt{2}\,W^2}, \qquad
a_3 = -\frac{1}{16\sqrt{2}\,W^3}.
\]

\pause
\textbf{Substituindo} $W = 250$\,kHz e o desvio máximo $\delta = 75$\,kHz:
\begin{center}
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{|l|c|c|}
\hline
Termo & Valor & Razão ao linear \\
\hline
$a_1\,\delta$        & $1{,}06 \times 10^{-1}$  & $100\%$ \\
$a_2\,\delta^2$      & $7{,}95 \times 10^{-3}$  & $7{,}5\%$ \\
$a_3\,\delta^3$      & $-1{,}19 \times 10^{-3}$ & $1{,}1\%$ \\
\hline
\end{tabular}
\end{center}

\end{frame}

% ============================================

\begin{frame}{Exemplo numérico: leitura do resultado}

\footnotesize
\textbf{Razão entre cúbico e quadrático:}
\[
\left|\frac{a_3 \delta^3}{a_2 \delta^2}\right| = \frac{\delta}{2W} = \frac{75}{500} \approx 0{,}15.
\]
O termo cúbico vale cerca de um sétimo do quadrático.

\pause
\textbf{Distorção do filtro único:} $|a_2 \delta^2|/|a_1 \delta| \approx 7{,}5\%$. A parcela $a_2 \delta^2$ domina e gera segundo harmônico da mensagem.

\pause
\textbf{Distorção do detector balanceado:} $a_2 \delta^2$ se cancela; resta apenas $2 a_3 \delta^3$, com
\[
\frac{|a_3 \delta^3|}{|a_1 \delta|} \approx 1{,}1\%.
\]

\pause
\textbf{Ganho prático:} o balanceamento reduz a distorção total em fator $\approx 7$, sem mudar a sensibilidade $a_1$.

\end{frame}

\begin{frame}{Exemplo de projeto: discriminador para FM broadcast}

\footnotesize
\textbf{Especificação:} receptor de FM broadcast operando na frequência intermediária.
\begin{itemize}\itemsep0pt
\item Portadora IF: $f_c = 10{,}7$\,MHz.
\item Desvio máximo: $\Delta f = 85$\,kHz; áudio máximo: $f_m = 15$\,kHz.
\item Largura de Carson: $B_T = 2(\Delta f + f_m) = 200$\,kHz.
\end{itemize}

\pause
\vspace{0.15cm}
\textbf{Passo 1 -- afastamento entre os filtros $\Delta$:} adotamos $\Delta = 1000$\,kHz, deixando $f_c$ bem dentro do flanco quase-linear de cada filtro:
\[
f_1 = f_c - \Delta = 9{,}7\,\text{MHz}, \qquad f_2 = f_c + \Delta = 11{,}7\,\text{MHz}.
\]

\pause
\vspace{0.1cm}
\textbf{Passo 2 -- fator de qualidade $Q$:} a banda 3\,dB de cada filtro precisa cobrir $\pm(\Delta + \Delta f)$ em torno de seu pico,
\[
\frac{f_0}{Q} \approx 2(\Delta + \Delta f) = 2{,}17\,\text{MHz}
\;\Longrightarrow\; Q \approx \frac{10{,}7\,\text{MHz}}{2{,}17\,\text{MHz}} \approx 5.
\]

\end{frame}

% ============================================

\begin{frame}{Exemplo: dimensionamento de $L, C$ e verificação}

\footnotesize
\textbf{Passo 3 -- $LC$ de cada filtro:} de $\omega_0 = 1/\sqrt{LC}$, adotando $L = 1{,}0\,\mu\text{H}$,
\[
C = \frac{1}{(2\pi f_0)^2 L}.
\]
Substituindo: $C_1 \approx 269$\,pF para $f_1 = 9{,}7$\,MHz e $C_2 \approx 185$\,pF para $f_2 = 11{,}7$\,MHz.

\pause
\vspace{0.15cm}
\textbf{Passo 4 -- verificação:}
\begin{itemize}\itemsep0pt
\item Excursão da frequência instantânea: $\pm 85$\,kHz em torno de $f_c$.
\item Cai dentro do trecho quase-linear de cada flanco, longe da extremidade onde $a_2$ cresce.
\item A subtração balanceada cancela $a_2\delta^2$; resta como distorção residual apenas $a_3\delta^3$.
\end{itemize}

\pause
\vspace{0.1cm}
\textbf{Resultado:} discriminador linear sobre $\pm 85$\,kHz, sintonia ajustável trocando os capacitores. A constante de demodulação é $K \propto a_1 k_f$, com $a_1$ fixado pela escolha de $Q$ e $\Delta$.

\end{frame}

\begin{frame}{Demodulação de NBFM: decomposição em banda básica}

Partindo do sinal FM genérico:
\[
s_{\FM}(t) = A_c \cos[2\pi f_c t + \phi(t)],
\qquad \phi(t) = 2\pi k_f \int_{-\infty}^{t} m(\tau)\,d\tau
\]

\pause
\textbf{Expandir o cosseno da soma:}
\[
s_{\FM}(t) = A_c \cos(2\pi f_c t)\cos\phi(t) - A_c \sin(2\pi f_c t)\sin\phi(t)
\]

\pause
\textbf{Regime NBFM} ($\beta \ll 1 \Rightarrow |\phi(t)| \ll 1$):
\[
\cos\phi(t) \approx 1, \qquad \sin\phi(t) \approx \phi(t)
\]

\pause
\begin{block}{Forma de banda básica}
\[
\boxed{\,s_{\NBFM}(t) \approx A_c \cos(2\pi f_c t) \;-\; A_c\,\phi(t)\,\sin(2\pi f_c t)\,}
\]
\end{block}

\end{frame}

% ============================================

\begin{frame}{NBFM tem a estrutura de AM convencional}

\textbf{Lado a lado:}
\[
s_{\AM}(t) = A_c \cos(2\pi f_c t) \;+\; A_c\, \mu\, m(t)\, \cos(2\pi f_c t)
\]
\[
s_{\NBFM}(t) \approx A_c \cos(2\pi f_c t) \;-\; A_c\, \phi(t)\, \sin(2\pi f_c t)
\]

\pause
\textbf{Mesma anatomia:} portadora pura mais um termo DSB-SC modulado por um sinal de banda básica.

\vspace{0.2cm}
\textbf{Únicas diferenças:}
\begin{itemize}\itemsep0pt
\item AM modula em fase com a portadora; NBFM modula em quadratura.
\item Em AM o sinal de banda básica é $m(t)$; em NBFM é $\phi(t) = 2\pi k_f \int m(\tau)d\tau$, a integral da mensagem.
\end{itemize}

\pause
\vspace{0.2cm}
\textbf{Consequência espectral:} as bandas laterais ocupam a mesma faixa que em AM, $B_{\NBFM} \approx 2W$, justificando a regra de Carson para $\beta \ll 1$.

\end{frame}

% ============================================

\begin{frame}{Demodulador coerente para NBFM}

\textbf{Ideia:} extrair o termo em quadratura por mistura coerente.

\pause
\textbf{Multiplicação por $-\sin(2\pi f_c t)$:}
\begin{align*}
&-A_c \cos(2\pi f_c t)\sin(2\pi f_c t) + A_c\,\phi(t)\sin^2(2\pi f_c t) \\
&= -\tfrac{A_c}{2}\sin(4\pi f_c t) + \tfrac{A_c}{2}\phi(t)\bigl[1 - \cos(4\pi f_c t)\bigr]
\end{align*}

\pause
\textbf{Após filtro passa-baixa} (remove $2f_c$): $\;\tfrac{A_c}{2}\phi(t)$.

\pause
\textbf{Derivando:} $\dfrac{d}{dt}\!\left[\tfrac{A_c}{2}\phi(t)\right] = \pi A_c k_f\, m(t)$.

\end{frame}

% ============================================

\begin{frame}{PLL: sinais de entrada e VCO}

\textbf{Entrada FM} (forma usada em análise PLL, equivalente a $\cos$ com 90° de defasagem):
\[
r(t) = A_c \sin[2\pi f_c t + \phi(t)], \quad \phi(t) = 2\pi k_f\!\int m(\tau)\,d\tau.
\]

\pause
\textbf{Saída do VCO} (em quadratura com a entrada para que o detector entregue $\sin$ do erro):
\[
y_v(t) = A_v \cos[2\pi f_c t + \phi_v(t)],
\]
\[
\boxed{\;\phi_v(t) = 2\pi k_v \int_0^t v(\tau)\,d\tau\;} \qquad (\text{Proakis 4.3.17})
\]
onde $k_v$ é a sensibilidade do VCO em Hz/V e $v(t)$ é a tensão de controle.

\pause
\textbf{Comparador de fase:} multiplicador analógico seguido de filtro passa-baixa que rejeita o termo em $2 f_c$.

\end{frame}

% ============================================

\begin{frame}{Detector de fase: derivação}

\textbf{Produto} $r(t)\,y_v(t)$ com $\omega_c \equiv 2\pi f_c$:
\[
A_c A_v \sin[\omega_c t + \phi]\,\cos[\omega_c t + \phi_v]
\]

\pause
\textbf{Identidade:} $\sin A\,\cos B = \tfrac{1}{2}[\sin(A+B) + \sin(A-B)]$. Com $A = \omega_c t + \phi$, $B = \omega_c t + \phi_v$:
\[
r\,y_v = \tfrac{A_c A_v}{2}\Bigl\{\underbrace{\sin[2\omega_c t + \phi + \phi_v]}_{2f_c\;\to\;\text{LPF rejeita}} + \sin[\phi - \phi_v]\Bigr\}
\]

\pause
\textbf{Após o LPF do comparador:}
\[
\boxed{\;e(t) = \tfrac{1}{2}A_v A_c\,\sin[\phi(t) - \phi_v(t)]\;} \qquad (\text{Proakis 4.3.18})
\]

\pause
\textbf{Observação:} o seno aparece naturalmente da identidade $\sin\!\cdot\!\cos$. Garante $e = 0$ exatamente em $\phi = \phi_v$ -- condição de \emph{lock}.

\end{frame}

% ============================================

\begin{frame}{Linearização e modelo do PLL}

\textbf{Erro de fase:} $\phi_e(t) \equiv \phi(t) - \phi_v(t)$.

\pause
\textbf{Hipótese de PLL travado} ($|\phi_e| \ll 1$): pela expansão $\sin x \approx x$,
\[
\sin[\phi(t) - \phi_v(t)] \approx \phi(t) - \phi_v(t) = \phi_e(t) \qquad (\text{Proakis 4.3.19})
\]
e o detector vira linear: $e(t) \approx \tfrac{1}{2}A_v A_c\,\phi_e(t)$.

\pause
\textbf{Modelo linearizado do PLL} (Proakis Fig 4.15):

\node[draw, rectangle, minimum width=2cm, minimum height=0.7cm] (vco) at (3,-1.6) {$2\pi k_v\!\int_0^t v(\tau)d\tau$};

\end{frame}

% ============================================

\begin{frame}{Equação integro-diferencial do erro de fase}

Substituindo $\phi_v(t) = 2\pi k_v \!\int_0^t v(\tau)d\tau$ em $\phi_e = \phi - \phi_v$:
\[
\phi_e(t) = \phi(t) - 2\pi k_v \int_0^t v(\tau)\,d\tau \qquad (\text{Proakis 4.3.20})
\]

\pause
\textbf{Diferenciando:}
\[
\frac{d\phi_e(t)}{dt} + 2\pi k_v\,v(t) = \frac{d\phi(t)}{dt} \qquad (\text{4.3.21})
\]

\pause
\textbf{Filtro de loop:} $v(t) = \int_0^\infty \phi_e(\tau)\,g(t-\tau)\,d\tau$ -- convolução com $g(t)$.

\pause
\textbf{Substituindo:}
\[
\frac{d\phi_e}{dt} + 2\pi k_v\!\int_0^\infty\!\phi_e(\tau)\,g(t-\tau)\,d\tau = \frac{d\phi}{dt} \qquad (\text{4.3.22})
\]

\end{frame}

% ============================================

\begin{frame}{PLL no domínio da frequência}

\textbf{Transformada de Fourier} da equação integro-diferencial 4.3.22:
\[
(j2\pi f)\,\Phi_e(f) + 2\pi k_v\,\Phi_e(f)\,G(f) = (j2\pi f)\,\Phi(f) \qquad (\text{4.3.23})
\]

\pause
\textbf{Resolvendo para $\Phi_e(f)$:}
\[
\boxed{\;\Phi_e(f) = \frac{1}{1 + \dfrac{k_v}{jf}\,G(f)}\,\Phi(f)\;} \qquad (\text{4.3.24})
\]

\pause
\textbf{Tensão de controle $V(f) = \Phi_e(f)\,G(f)$:}
\[
V(f) = \frac{G(f)}{1 + \dfrac{k_v}{jf}\,G(f)}\,\Phi(f) \qquad (\text{4.3.25})
\]

\end{frame}

% ============================================

\begin{frame}{Demodulação FM via PLL}

\textbf{Critério de projeto} (Proakis 4.3.26): para $|f| < W$ (banda da mensagem),
\[
\left|\dfrac{k_v\,G(f)}{jf}\right| \gg 1.
\]

\pause
Nessa faixa o denominador de 4.3.25 fica $\approx (k_v/jf)\,G(f)$:
\[
V(f) \approx \dfrac{j2\pi f}{2\pi k_v}\,\Phi(f) \qquad (\text{4.3.27})
\]

\pause
\textbf{No tempo:} $v(t) = \dfrac{1}{2\pi k_v}\,\dfrac{d\phi}{dt}$. Como $\dot\phi = 2\pi k_f\,m(t)$:
\[
\boxed{\;v(t) = \dfrac{k_f}{k_v}\,m(t)\;} \qquad (\text{4.3.28})
\]

A tensão de controle do VCO é proporcional à mensagem.

\end{frame}

% ============================================

\begin{frame}{Filtro de loop $G(s)$ vs.\ transferência em malha fechada $H(s)$}

\textbf{$G(s)$} é o \emph{filtro de loop} -- mesma função da derivação Proakis, com $s = j2\pi f$. É o que você \textbf{escolhe} no projeto.

\pause
\vspace{0.3cm}
\textbf{$H(s) = \Phi_v(s)/\Phi(s)$} é a \emph{função de transferência em malha fechada}, calculada a partir de $G$ e do ganho do VCO:
\[
\boxed{\;H(s) = \frac{2\pi k_v\,G(s)}{s + 2\pi k_v\,G(s)}\;}
\]

\pause
\vspace{0.3cm}
$H$ é o que \textbf{resulta} -- a relação entre o sinal de entrada e o sinal travado pelo VCO. Determina a dinâmica do laço (rapidez, amortecimento, banda).

\vspace{0.3cm}
Próximos slides: dois casos comuns de $G$ e o $H$ correspondente.

\end{frame}

% ============================================

\begin{frame}{Caso 1: filtro proporcional $G(s) = 1$}

\textbf{Escolha mais simples:} $G(s) = 1$ (sem dinâmica adicional, só ganho unitário).

\pause
\textbf{Substituindo na fórmula de $H$:}
\[
H(s) = \frac{2\pi k_v}{s + 2\pi k_v}
\]

\textbf{Sistema de 1ª ordem.} Polo único em $s = -2\pi k_v$, constante de tempo $\tau = 1/(2\pi k_v)$.

\pause
\vspace{0.3cm}
\textbf{Exemplo numérico} ($k_v = 50$\,kHz/V):
\[
2\pi k_v = 2\pi\cdot 50\,\text{kHz} = 3{,}14\!\times\!10^5\,\text{rad/s}
\]
\[
\tau = \frac{1}{3{,}14\!\times\!10^5} = 3{,}2\,\mu\text{s}, \qquad B_L = \frac{2\pi k_v}{4\pi} = 25\,\text{kHz}
\]

\pause
\textbf{Limitação:} 1ª ordem não permite ajustar amortecimento independentemente. Aumentar $k_v$ acelera resposta mas degrada margem de estabilidade.

\end{frame}

% ============================================

\begin{frame}{Caso 2: filtro lag-lead $G(s) = (1 + s\tau_2)/(1 + s\tau_1)$}

\textbf{Topologia mais usada em receptores reais:} permite ajustar $\zeta$ e $\omega_n$ separadamente.

\pause
\vspace{0.2cm}
Substituindo na fórmula de $H$, obtém-se sistema de 2ª ordem padrão:
\[
\omega_n = \sqrt{\dfrac{2\pi k_v}{\tau_1}}, \qquad
\zeta = \dfrac{\omega_n}{2}\!\left(\tau_2 + \dfrac{1}{2\pi k_v}\right)
\]

\pause
\vspace{0.3cm}
\textbf{Projeto:} dado $\zeta$ e $f_n$ desejados, resolve-se para $\tau_1$ e $\tau_2$.

\pause
\vspace{0.2cm}
\textbf{Exemplo} ($k_v = 50$\,kHz/V, alvo $\zeta = 0{,}7$, $f_n = 30$\,kHz):
\[
\omega_n = 2\pi\cdot 30\,\text{kHz} = 1{,}88\!\times\!10^5\,\text{rad/s}
\]
\[
\tau_1 = \frac{2\pi k_v}{\omega_n^2} \approx 88\,\mu\text{s}, \qquad
\tau_2 = \frac{2\zeta}{\omega_n} - \frac{1}{2\pi k_v} \approx 7{,}3\,\mu\text{s}.
\]

\end{frame}

\begin{frame}{Parâmetros do PLL: capture e lock}

\textbf{Lock range} -- faixa onde o PLL \emph{mantém} o travamento já adquirido. Vale para qualquer ordem:
\[
\Delta f_{lock} \approx \frac{K_d K_v}{2\pi} \quad\text{(com $K_v$ em rad/s/V)}
\]

\pause
\textbf{Capture range} -- faixa onde o PLL \emph{adquire} o travamento partindo do destravado. Aqui a ordem importa:
\begin{itemize}\itemsep0pt
\item \textbf{1ª ordem} ($G=1$): não há transitório de aquisição -- $\Delta f_{capture} = \Delta f_{lock}$.
\item \textbf{2ª ordem} (lag-lead): há \emph{pull-in} com dinâmica de $\zeta, \omega_n$, e
\[
\Delta f_{capture} \approx 2\zeta f_n < \Delta f_{lock}
\]
\end{itemize}

\pause
\textbf{Exemplo} ($K_d = 1$\,V/rad, $k_v = 50$\,kHz/V): $\;\Delta f_{lock} = 50$\,kHz.

Com lag-lead de $\zeta = 0{,}7$, $f_n = 30$\,kHz: $\Delta f_{capture} = 42$\,kHz $< 50$\,kHz.

\pause
\textbf{Para FM broadcast} ($\Delta f = 75$\,kHz): aumentar $k_v$ para $\sim$100\,kHz/V garante $\Delta f_{lock} > 75$\,kHz.

\end{frame}

% ============================================

\begin{frame}{Convenção dos ganhos do PLL}

As três fórmulas de parâmetros ($\Delta f_{lock}$, $\Delta f_{capture}$, $B_L$) usam a mesma combinação $K_d K_v$. Antes dos exemplos numéricos, fixar unidades:

\pause
\vspace{0.3cm}
\begin{itemize}\itemsep0.3em
\item \textbf{$K_d$} -- ganho do detector de fase, em \textbf{V/rad}.

Vem da derivação: $K_d = \frac{1}{2}A_c A_v$.

\pause
\item \textbf{$K_v$} -- ganho do VCO, em \textbf{rad/s/V}.

Relação com $k_v$ em Hz/V: $\;K_v = 2\pi k_v$.

\pause
\item \textbf{$K_d K_v$} -- \emph{ganho de loop em malha aberta}, em \textbf{rad/s}.

Aparece em todas as fórmulas dinâmicas (lock range, banda, polos).
\end{itemize}

\end{frame}

% ============================================

\begin{frame}{Largura de banda do loop $B_L$}

\textbf{$B_L$} determina resposta dinâmica e rejeição de ruído do PLL.

\vspace{0.3cm}
\textbf{Para 1ª ordem} (filtro proporcional):
\[
\boxed{\;B_L = \frac{K_d K_v}{4\pi}\;}
\]

\pause
\vspace{0.3cm}
\textbf{Exemplo numérico} ($K_d = 1$\,V/rad, $k_v = 50$\,kHz/V):
\[
K_d K_v = 1 \cdot 2\pi \cdot 50\,\text{kHz} = 3{,}14\!\times\!10^5\,\text{rad/s}
\]
\[
B_L = \frac{3{,}14\!\times\!10^5}{4\pi} = 25\,\text{kHz}
\]

\end{frame}

% ============================================

\begin{frame}{Trade-off de banda do loop}

\textbf{$B_L$ largo:}
\begin{itemize}\itemsep0pt
\item Resposta rápida -- PLL acompanha variações rápidas de fase.
\item \textbf{Mais ruído} passa pela saída.
\end{itemize}

\pause
\vspace{0.3cm}
\textbf{$B_L$ estreito:}
\begin{itemize}\itemsep0pt
\item Saída mais limpa (ruído rejeitado).
\item PLL \textbf{lento} -- pode não acompanhar variações rápidas da mensagem.
\end{itemize}

\pause
\vspace{0.3cm}
\textbf{Critério prático:} $B_L \gtrsim f_{m,\max}$ da mensagem para garantir tracking.

\textbf{Exemplo:} áudio FM tem $f_{m,\max} = 15$\,kHz. O $B_L = 25$\,kHz do exemplo anterior acompanha confortavelmente, com margem para o ruído.

\end{frame}

\pause
\vspace{0.2cm}

\begin{frame}{Método de Armstrong: ideia geral}

\textbf{Ideia central:} gerar NBFM estável a partir de cristal e depois multiplicar a frequência até obter WBFM.

\vspace{0.4cm}
\pause
\textbf{Processo:}

\begin{enumerate}
\item Gerar NBFM com $\beta$ pequeno e $f_c$ estável de cristal.
\item Multiplicar frequência por $n$, o que multiplica também $f_c$, $\Delta f$ e $\beta$ pelo mesmo fator.
\item Converter para a banda desejada via mixer.
\item Repetir multiplicação se necessário.
\end{enumerate}

\pause
\vspace{0.3cm}
\textbf{Efeito da multiplicação por $n$:}
\[
f_c \to n f_c, \qquad \beta \to n\beta, \qquad \Delta f \to n\Delta f.
\]

\end{frame}

% ============================================

\begin{frame}{Modulador NBFM: núcleo do Armstrong}

\textbf{Sintetizar} $s_{\NBFM}(t) = A_c\cos(2\pi f_1 t) - A_c\,\phi(t)\sin(2\pi f_1 t)$ \textbf{com} $\phi(t) = 2\pi k_f\!\int m(\tau)d\tau$:

\begin{center}
\begin{tikzpicture}[>=latex, every node/.style={font=\footnotesize}]

\node[draw, circle, minimum size=0.75cm] (mul) at (5.8,1.7) {$\times$};
\node[draw, circle, minimum size=0.75cm] (sum) at (8.2,0.85) {$+$};
\node[draw, rectangle, minimum width=2cm, minimum height=0.7cm] (osc) at (0,0) {Cristal $f_1$};
\node[draw, rectangle, minimum width=1.4cm, minimum height=0.7cm] (sh) at (5.8,0) {$-90^\circ$};

\draw[->] (-1.4,1.7) -- node[above] {$m(t)$} (int);
\draw[->] (int) -- node[above] {$\phi(t)$} (mul);
\draw[->] (mul) -- (sum);
\draw[->] (osc) -- node[above] {$\cos(2\pi f_1 t)$} (sh);
\draw[->] (sh) -- node[right] {$-\sin(2\pi f_1 t)$} (mul);
\fill (3.2,0) circle (1.5pt);

\textbf{Cristal} fornece $f_1$ estável; \textbf{integrador} converte $m(t)$ em $\phi(t)$; \textbf{defasador, multiplicador e somador} montam a soma em quadratura. Como $f_1$ não passa pelo modulador, sua estabilidade é preservada.

\end{frame}

\begin{frame}{Cadeia Armstrong completa}

\begin{center}
\begin{tikzpicture}[>=latex, every node/.style={font=\footnotesize}]
\node[draw, rectangle, minimum width=2cm, minimum height=0.9cm] (nbfm) at (0,0) {Mod.\ NBFM};
\node[draw, rectangle, minimum width=1cm, minimum height=0.9cm] (n1) at (2.7,0) {$\times n_1$};
\node[draw, circle, minimum size=0.8cm] (mix) at (4.6,0) {$\times$};
\node[draw, rectangle, minimum width=1cm, minimum height=0.9cm] (bpf) at (6.4,0) {BPF};

\begin{frame}[shrink=10]{Expansão de banda na cadeia Armstrong}

\begin{center}

\toprule

\midrule

NBFM             & 200\,kHz  & 25\,Hz     & 0{,}0017 & 30\,kHz  \\
$\times 60$      & 12\,MHz   & 1{,}5\,kHz & 0{,}1    & 33\,kHz  \\
mixer            & 2\,MHz    & 1{,}5\,kHz & 0{,}1    & 33\,kHz  \\
$\times 50$      & 100\,MHz  & 75\,kHz    & 5        & \textbf{180\,kHz} \\
\midrule
\multicolumn{5}{l}{\emph{Cenário B} ($f_m = 3$\,kHz)} \\
NBFM             & 200\,kHz   & 25\,Hz   & 0{,}0083 & 6\,kHz   \\
$\times 8$       & 1{,}6\,MHz & 200\,Hz  & 0{,}067  & 6{,}4\,kHz \\
mixer            & 4\,MHz     & 200\,Hz  & 0{,}067  & 6{,}4\,kHz \\

\bottomrule
\end{tabular}

\end{frame}

\section{Receptor Superheterodino}

\begin{frame}{O problema: várias estações compartilham a antena}

A antena recebe \textbf{toda} a faixa FM (88--108\,MHz), com canais espaçados de 200\,kHz. Para ouvir uma estação só, é preciso \emph{filtrar} apenas o canal desejado e rejeitar os adjacentes.

% ============================================

\begin{frame}[shrink=8]{Por que filtrar direto em $f_c$ é difícil}

\textbf{Seletividade exigida:} $Q = \dfrac{f_c}{B_{\text{canal}}} = \dfrac{100\,\text{MHz}}{200\,\text{kHz}} = 500$.

\pause
\textbf{Conflito $Q$ alto vs.\ sintonia.} Olhando as tecnologias de filtro:

\begin{center}
\footnotesize
\begin{tabular}{lccc}
\toprule

\midrule

Cerâmico               & 200--1000 & não         & freq.\ fixa \\
Cristal de quartzo     & $10^3$--$10^5$ & não    & freq.\ fixa \\
SAW                    & 5000--20000 & não        & freq.\ fixa \\
\bottomrule
\end{tabular}
\end{center}

\pause
\textbf{Quem tem $Q \geq 500$ não é tunável.}\quad \textbf{Quem é tunável (LC) não atinge $Q = 500$.}

\pause

% ============================================

\begin{frame}{Relação fundamental do superheterodino}

A mistura entre o sinal recebido e o oscilador local produz duas componentes; o filtro IF seleciona uma delas:
\[
\boxed{\;f_{IF} = |f_{RF} - f_{LO}|\;}
\]

\pause
\textbf{Para sintonizar $f_{RF}$:} basta ajustar $f_{LO}$ tal que $f_{LO} = f_{RF} + f_{IF}$ (ou $f_{RF} - f_{IF}$).

\pause
\textbf{Exemplo: rádio FM com $f_{IF} = 10{,}7$\,MHz.}
\begin{center}

\midrule

\textbf{$f_{LO}$ tunável de 98,7 a 118,7\,MHz} (oscilador é fácil de variar); \textbf{filtro IF fixo em 10,7\,MHz} (filtro de cristal/cerâmico de alto $Q$, projetado uma vez).

\end{frame}

\begin{frame}{Vantagens do Superheterodino}

\textbf{Vantagens:}

\begin{enumerate}
\item \textbf{Seletividade:}
   \begin{itemize}
   \item Filtro IF fixo pode ser muito seletivo
   \item Não precisa ajustar filtro ao trocar estação
   \end{itemize}

\item \textbf{Sensibilidade:}
   \begin{itemize}
   \item Amplificação concentrada em frequência fixa
   \item Melhor controle de ganho (AGC)
   \end{itemize}

\item \textbf{Estabilidade do caminho IF:}
   \begin{itemize}
   \item Filtro IF, amp IF e demodulador operam em frequência fixa $\to$ deriva térmica e tolerâncias previsíveis e calibráveis
   \item O LO continua variável; sua estabilidade é tratada à parte (cristal, PLL, AFC)
   \end{itemize}
\end{enumerate}

\textbf{Próximo problema:} a mistura cria uma \textbf{frequência imagem} que precisa ser rejeitada antes do mixer.

\end{frame}

% ============================================

\begin{frame}{Exemplo: Receptor FM Superheterodino}

\textbf{Rádio FM comercial:}

\begin{itemize}
\item Banda FM: 88-108 MHz
\item IF padrão: $f_{IF} = 10.7$ MHz
\end{itemize}

\textbf{Para receber} $f_{RF} = 100.0$ MHz:

\[
f_{LO} = f_{RF} + f_{IF} = 100.0 + 10.7 = 110.7 \text{ MHz}
\]

\textbf{Frequência imagem:}
\[
f_{im} = f_{LO} + f_{IF} = 110.7 + 10.7 = 121.4 \text{ MHz}
\]

Fora da banda FM (88-108 MHz) $\rightarrow$ filtro RF facilmente rejeita

\end{frame}

\begin{frame}{Resumo: FM vs.\ PM e Espectro}

\textbf{FM vs. PM:}
\begin{itemize}\itemsep0pt
\item FM -- frequência $\propto m(t)$, fase $\propto \int m(t)$. Mais comum, melhor em ruído.
\item PM -- fase $\propto m(t)$, frequência $\propto dm(t)/dt$.
\end{itemize}

\vspace{0.3cm}
\textbf{Espectro FM:}
\begin{itemize}\itemsep0pt
\item Infinitas bandas laterais via Bessel.
\item Carson: $B \approx 2(\Delta f + f_m)$.
\item NBFM se $\beta \ll 1$; WBFM se $\beta \gg 1$.
\end{itemize}

\end{frame}

% ============================================

\begin{frame}{Resumo: Demodulação e Geração}

\textbf{Demodulação:}
\begin{itemize}\itemsep0pt
\item Discriminador de frequência -- precisa de limitador antes para rejeitar ruído de amplitude.
\item PLL -- mais usado atualmente, intrinsecamente imune a variações de amplitude.
\end{itemize}

\vspace{0.3cm}
\textbf{Geração:}
\begin{itemize}\itemsep0pt
\item Direta -- VCO, simples mas instável em $f_c$.
\item Indireta -- método de Armstrong, NBFM com cristal seguido de multiplicação.
\end{itemize}

\vspace{0.3cm}
\textbf{Receptor:}
\begin{itemize}\itemsep0pt
\item Arquitetura superheterodina, IF fixa de 10,7\,MHz no caso comercial.
\end{itemize}

\end{frame}
