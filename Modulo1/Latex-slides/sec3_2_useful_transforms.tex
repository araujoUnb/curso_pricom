\begin{frame}{As três propriedades que definem $\delta(t)$}

\vspace{0.1cm}
\[
\tikzmarknode{dpA}{\delta(t) = 0 \;\;\text{para } t \neq 0}
\qquad
\tikzmarknode{dpB}{\int_{-\infty}^{\infty}\delta(t)\,dt = 1}
\]

\anotacoes{%
  \only<2->{
    \node[nota=cBlue, text width=2.2cm] (dpNA) at ($(dpA)+(-0.6,-1.3)$)
      {fora da origem não existe nada};
    \draw[setinha, cBlue] (dpNA.north) -- (dpA.south);}
  \only<3->{
    \node[nota=cGreen, text width=2.2cm] (dpNB) at ($(dpB)+(0.6,-1.3)$)
      {toda a massa está concentrada em $t=0$};
    \draw[setinha, cGreen] (dpNB.north) -- (dpB.south);}
}%

\vspace{1.9cm}

\onslide<4->{Das duas sai a terceira, que é a que realmente usamos:}

\vspace{-0.1cm}
\onslide<5->{
\[
\boxed{\;\int_{-\infty}^{\infty} f(t)\,\delta(t-t_0)\,dt \;=\; f(t_0)\;}
\]}

\onslide<6->{\alert{A delta não integra o sinal: ela colhe um único valor.}}

\end{frame}

\begin{frame}{Transformada da delta}

Aplicamos a definição:

\vspace{-0.1cm}
\[
\FT{\delta(t)} = \int_{-\infty}^{\infty}
\tikzmarknode{tdA}{\delta(t)}\;
\tikzmarknode{tdB}{e^{-j\omega t}}\, dt
\]

\anotacoes{%
  \only<2->{
    \node[nota=cRed, text width=2.0cm] (tdNA) at ($(tdA)+(-2.3,-1.3)$)
      {a amostragem em $t_0=0$};
    \draw[setinha, cRed] (tdNA.east) -- (tdA.south);}
  \only<3->{
    \node[nota=cGreen, text width=2.2cm] (tdNB) at ($(tdB)+(2.5,-1.3)$)
      {este é o $f(t)$ que será colhido};
    \draw[setinha, cGreen] (tdNB.west) -- (tdB.south);}
}%

\vspace{1.25cm}

\onslide<4->{
\[
= \; e^{-j\omega \cdot 0} \; = \; 1
\]}

\vspace{0.1cm}
\onslide<5->{
\[
\boxed{\;\delta(t) \xleftrightarrow{\ft} 1\;}
\]}

\end{frame}

$\delta(t-t_0) \xleftrightarrow{\ft} e^{-j\omega t_0}$.}}

\end{frame}

\begin{frame}{Onde isso aparece: canal com dois caminhos}

Um sinal chega por dois percursos, com atrasos e ganhos diferentes:

\vspace{-0.1cm}
\[
h(t) = \alpha_1\,\delta(t-\tau_1) + \alpha_2\,\delta(t-\tau_2)
\]

\onslide<2->{Aplicando o par recém-deduzido a cada impulso:

\vspace{-0.1cm}
\[
H(\omega) = \alpha_1 e^{-j\omega \tikzmarknode{mcA}{\tau_1}}
          + \alpha_2 e^{-j\omega \tikzmarknode{mcB}{\tau_2}}
\]}

\anotacoes{%
  \only<3->{
    \node[nota=cOrange, text width=3.8cm] (mcN) at ($(mcA)+(1.0,-1.35)$)
      {dois fasores girando em velocidades diferentes com $\omega$};
    \draw[setinha, cOrange] (mcN.north west) to[bend left=18] (mcA.south);
    \draw[setinha, cOrange] (mcN.north east) to[bend right=18] (mcB.south);}
}%

\vspace{1.25cm}

\onslide<4->{Em certas frequências eles se somam, em outras se cancelam.}

\onslide<5->{\alert{É o desvanecimento seletivo em frequência, o modelo básico
de multipercurso.}}

\end{frame}

\subsection{Constante e senoides}

\begin{frame}{Transformada de uma constante}

Queremos $\mathcal{F}\{A\}$, com $A$ valendo o mesmo para todo $t$.

\vspace{0.1cm}
\onslide<2->{
\[
\tikzmarknode{ctA}{\int_{-\infty}^{\infty} A\, e^{-j\omega t}\, dt}
\]}

\anotacoes{%
  \only<3->{
    \node[nota=cRed, text width=3.4cm] (ctN) at ($(ctA)+(0,-1.35)$)
      {não converge: a área de $|A|$ é infinita};
    \draw[setinha, cRed] (ctN.north) -- (ctA.south);}
}%

\vspace{1.2cm}

\onslide<4->{\alert{Vamos pelo caminho inverso: propor o espectro e verificar
que ele reconstrói a constante.}}

\end{frame}

% --------------------------------------------

\begin{frame}{Constante: verificando pela transformada inversa}

Proponha $F(\omega) = 2\pi A\,\delta(\omega)$ e aplique a transformada inversa:

\vspace{-0.1cm}
\[
f(t) = \frac{1}{2\pi}\int_{-\infty}^{\infty}
2\pi A\, \tikzmarknode{ciA}{\delta(\omega)}\, e^{j\omega t}\, d\omega
\]

\anotacoes{%
  \only<2->{
    \node[nota=cGreen, text width=3.2cm] (ciN) at ($(ciA)+(0.3,-1.35)$)
      {a amostragem de novo, agora colhendo em $\omega = 0$};
    \draw[setinha, cGreen] (ciN.north) -- (ciA.south);}
}%

\vspace{1.3cm}

\onslide<3->{
\[
f(t) = A\, e^{\,j\cdot 0\cdot t} = A
\]}

\onslide<4->{
\[
\boxed{\;A \xleftrightarrow{\ft} 2\pi A\,\delta(\omega)\;}
\]}

\end{frame}

% --------------------------------------------

\begin{frame}{Exponencial complexa}

\[
f(t) = e^{j\omega_0 t}
\]

\onslide<2->{Juntando as exponenciais:

\vspace{-0.1cm}
\[
F(\omega) = \int_{-\infty}^{\infty} e^{j\omega_0 t}\, e^{-j\omega t}\, dt
= \int_{-\infty}^{\infty} \tikzmarknode{ecA}{e^{-j(\omega-\omega_0)t}}\, dt
\]}

\anotacoes{%
  \only<3->{
    \node[nota=cGreen, text width=3.8cm] (ecN) at ($(ecA)+(0.2,-1.35)$)
      {é a integral da constante, com $\omega$ deslocado de $\omega_0$};
    \draw[setinha, cGreen] (ecN.north) -- (ecA.south);}
}%

\vspace{0.85cm}

\onslide<4->{
\[
\boxed{\;e^{j\omega_0 t} \xleftrightarrow{\ft} 2\pi\,\delta(\omega-\omega_0)\;}
\]}

\onslide<5->{\alert{Multiplicar por $e^{j\omega_0 t}$ desloca o espectro.}}

\end{frame}

\begin{frame}{Cosseno: Euler mais linearidade}

Escrevemos o cosseno como par de exponenciais:

\vspace{-0.1cm}
\[
\cos(\omega_0 t) = \tfrac{1}{2}\,\tikzmarknode{csA}{e^{j\omega_0 t}}
                 + \tfrac{1}{2}\,\tikzmarknode{csB}{e^{-j\omega_0 t}}
\]

\anotacoes{%
  \only<2->{
    \node[nota=cGreen, text width=2.0cm] (csNA) at ($(csA)+(-2.4,-1.25)$)
      {vira $2\pi\delta(\omega-\omega_0)$};
    \draw[setinha, cGreen] (csNA.east) -- (csA.south);}
  \only<3->{
    \node[nota=cPurple, text width=2.0cm] (csNB) at ($(csB)+(2.4,-1.25)$)
      {vira $2\pi\delta(\omega+\omega_0)$};
    \draw[setinha, cPurple] (csNB.west) -- (csB.south);}
}%

\vspace{1.6cm}

\onslide<4->{Somando com os pesos $1/2$:

\vspace{-0.15cm}
\[
\boxed{\;\cos(\omega_0 t) \xleftrightarrow{\ft}
\pi\big[\delta(\omega-\omega_0)+\delta(\omega+\omega_0)\big]\;}
\]}

\end{frame}

% --------------------------------------------

\begin{frame}{Seno e a assinatura da fase}

Com $\sin(\omega_0 t) = \dfrac{e^{j\omega_0 t}-e^{-j\omega_0 t}}{2j}$, o mesmo
caminho leva a:

\vspace{-0.1cm}
\[
\boxed{\;\sin(\omega_0 t) \xleftrightarrow{\ft}
\tikzmarknode{snA}{j}\,\pi\big[\delta(\omega+\omega_0)-\delta(\omega-\omega_0)\big]\;}
\]

\anotacoes{%
  \only<2->{
    \node[nota=cRed, text width=3.4cm] (snN) at ($(snA)+(0.6,-1.35)$)
      {o $j$ e o sinal trocado vêm do $1/2j$: é a fase que separa seno de cosseno};
    \draw[setinha, cRed] (snN.north west) to[bend right=15] (snA.south);}
}%

\vspace{2.0cm}

\onslide<3->{Cada senoide real gera duas raias, em $+\omega_0$ e $-\omega_0$.}

\onslide<4->{\alert{Magnitude idêntica nas duas; o que distingue seno de
cosseno é só a fase.}}

\end{frame}

\begin{frame}{Por que não dá para integrar direto}

\[
\FT{\sgn(t)} = \int_{-\infty}^{\infty}

\anotacoes{%
  \only<2->{

apertar a aproximação:}

\vspace{-0.1cm}
\onslide<4->{
\[
\sgn(t) = \lim_{a\to 0^{+}}\Big[e^{-at}u(t) - e^{at}u(-t)\Big]
\]}

\end{frame}

\subsection{Gaussiana}

\begin{frame}{Pulso gaussiano: o expoente}

\[
f(t) = e^{-\alpha t^2}, \qquad \alpha > 0
\]

\onslide<2->{
\[
F(\omega) = \int_{-\infty}^{\infty} e^{\tikzmarknode{gaA}{-\alpha t^2 - j\omega t}}\, dt
\]}

\anotacoes{%
  \only<3->{
    \node[nota=cOrange, text width=3.6cm] (gaN) at ($(gaA)+(0.3,-1.35)$)
      {um polinômio de segundo grau em $t$: dá para completar o quadrado};
    \draw[setinha, cOrange] (gaN.north) -- (gaA.south);}
}%

\vspace{1.6cm}

\onslide<4->{
\[
-\alpha t^2 - j\omega t
= -\alpha\left(t + \frac{j\omega}{2\alpha}\right)^{2} - \frac{\omega^2}{4\alpha}
\]}

\end{frame}

% --------------------------------------------

\begin{frame}{Pulso gaussiano: separando os fatores}

O termo sem $t$ sai da integral:

\vspace{0.1cm}
\[
F(\omega) = \tikzmarknode{gbA}{e^{-\omega^2/(4\alpha)}}
\int_{-\infty}^{\infty}
\tikzmarknode{gbB}{e^{-\alpha\left(t+\frac{j\omega}{2\alpha}\right)^{2}}}\, dt
\]

\anotacoes{%
  \only<2->{
    \node[nota=cGreen, text width=2.2cm] (gbNA) at ($(gbA)+(-2.4,-1.35)$)
      {não depende de $t$};
    \draw[setinha, cGreen] (gbNA.east) -- (gbA.south);}
  \only<3->{
    \node[nota=cBlue, text width=2.6cm] (gbNB) at ($(gbB)+(1.4,-1.35)$)
      {integral gaussiana: vale $\sqrt{\pi/\alpha}$};
    \draw[setinha, cBlue] (gbNB.north) -- (gbB.south);}
}%

\vspace{1.9cm}

\onslide<4->{
\[
F(\omega) = \sqrt{\frac{\pi}{\alpha}}\; e^{-\omega^2/(4\alpha)}
\]}

\end{frame}

% --------------------------------------------

\begin{frame}{Gaussiana vira gaussiana}

\[
\boxed{\;e^{-\alpha t^2} \xleftrightarrow{\ft}
\sqrt{\frac{\pi}{\alpha}}\; e^{-\omega^2/(4\tikzmarknode{grA}{\alpha})}\;}
\]

\anotacoes{%
  \only<2->{
    \node[nota=cRed, text width=3.8cm] (grN) at ($(grA)+(0.2,-1.4)$)
      {$\alpha$ aparece invertido: pulso estreito no tempo, espectro largo};
    \draw[setinha, cRed] (grN.north) -- (grA.south);}
}%

\vspace{2.1cm}

\onslide<3->{É a única forma que se reproduz nos dois domínios.}

\onslide<4->{\alert{E é também a que atinge o mínimo do princípio da incerteza,
com $\Delta t\,\Delta\omega = 1/2$.}}

\end{frame}

% --------------------------------------------

\begin{frame}{Por que fibra óptica usa pulsos gaussianos}

Um pulso de luz $p(t) = e^{-t^2/(2\sigma^2)}$ atravessa a fibra e sofre
dispersão cromática, que o alarga no tempo.

\vspace{0.2cm}

\onslide<2->{Como a transformada de uma gaussiana é outra gaussiana, a forma do
pulso se preserva ao longo da propagação; só a largura muda.}

\onslide<3->{Sendo o mínimo do produto $\Delta t\,\Delta\omega$, nenhum outro
formato ocupa menos tempo e banda ao mesmo tempo.}

\onslide<4->{\alert{Se o alargamento fizer pulsos vizinhos se tocarem, aparece
interferência intersimbólica, e é isso que limita a taxa da fibra.}}

\end{frame}

\subsection{Trem de impulsos}

\begin{frame}{Trem de impulsos}

\vspace{-0.1cm}
\begin{center}
\begin{tikzpicture}[scale=0.9, every node/.style={font=\scriptsize}]
  \draw[->, gray!70] (-3.0,0) -- (3.2,0) node[right, black, font=\tiny] {$t$};
  \draw[->, gray!70] (0,-0.25) -- (0,1.9);
  \foreach \n in {-2,-1,0,1,2}
    {\draw[cBlue, very thick, -{Stealth[length=2mm]}] (\n,0) -- (\n,1.5);}
  \only<2->{\draw[<->, cOrange, thick] (0,1.75) -- (1,1.75)
     node[midway, above, cOrange] {$T$};}
  \node[below, font=\tiny] at (1,-0.15) {$T$};
  \node[below, font=\tiny] at (-1,-0.15) {$-T$};
\end{tikzpicture}
\end{center}

\vspace{0.1cm}

\[
f(t) = \sum_{n=-\infty}^{\infty} \delta(t-nT)
\]

\onslide<3->{É periódico, então tem série de Fourier, com
$\omega_0 = 2\pi/T$.}

\onslide<4->{\alert{Vamos pela série, porque a transformada direta não sai.}}

\end{frame}

% --------------------------------------------

\begin{frame}{Os coeficientes são todos iguais}

Dentro de um período só existe o impulso em $t=0$:

\vspace{0.1cm}
\[
c_n = \frac{1}{T}\int_{-T/2}^{T/2}
\tikzmarknode{tiA}{\delta(t)}\, e^{-jn\omega_0 t}\, dt
\]

\anotacoes{%
  \only<2->{
    \node[nota=cRed, text width=3.2cm] (tiN) at ($(tiA)+(0.4,-1.35)$)
      {a peneira de novo, agora colhendo em $t=0$};
    \draw[setinha, cRed] (tiN.north) -- (tiA.south);}
}%

\vspace{1.8cm}

\onslide<3->{
\[
c_n = \frac{1}{T}\, e^{0} = \frac{1}{T}
\qquad \text{para todo } n
\]}

\onslide<4->{\alert{Todos os harmônicos entram com o mesmo peso.}}

\end{frame}

% --------------------------------------------

\begin{frame}{Transformando termo a termo}

Com $c_n = 1/T$, a série fica:

\vspace{-0.1cm}
\[
f(t) = \frac{1}{T}\sum_{n=-\infty}^{\infty}
