\subsection{Introdução}

\begin{frame}{Funções Fundamentais em Sistemas de Comunicação}

Certas funções aparecem frequentemente na análise de sinais e sistemas:

\begin{itemize}
\item \textbf{Função delta:} impulsos, amostragem
\item \textbf{Constantes e senoidais:} portadoras, osciladores
\item \textbf{Função degrau:} ligar/desligar, causalidade
\item \textbf{Pulsos:} transmissão de dados digitais
\item \textbf{Gaussiana:} ruído, pulsos em fibras ópticas
\end{itemize}

\vspace{0.2cm}

Conhecer as transformadas dessas funções permite:
\begin{itemize}
\item Análise rápida de sistemas complexos
\item Uso de propriedades para derivar novas transformadas
\item Compreensão do comportamento espectral
\end{itemize}

\end{frame}

% ============================================

\subsection{Função Delta de Dirac}

\begin{frame}{Função Delta de Dirac: Definição}

A \textbf{função delta de Dirac} $\delta(t)$ não é uma função no sentido clássico, mas uma \textit{distribuição} definida por suas propriedades:

\begin{enumerate}
\item $\delta(t) = 0$ para todo $t \neq 0$
\item $\int_{-\infty}^{\infty} \delta(t) dt = 1$
\item \textbf{Propriedade de amostragem:}
\[
\int_{-\infty}^{\infty} f(t) \delta(t - t_0) dt = f(t_0)
\]
\end{enumerate}

\vspace{0.3cm}

\textbf{Interpretação física:} Impulso infinitamente estreito com área unitária.


\end{frame}

% ============================================

\begin{frame}{Transformada da Delta de Dirac}

\textbf{Cálculo:}
\vspace{-2mm}
\[
\FT{\delta(t)} = \int_{-\infty}^{\infty} \delta(t) e^{-j\omega t} dt
\]
\vspace{-4mm}

Usando a propriedade de amostragem com $f(t) = e^{-j\omega t}$ e $t_0 = 0$:
\vspace{-2mm}
\[
\FT{\delta(t)} = e^{-j\omega \cdot 0} = 1
\]
\vspace{-4mm}

\begin{block}{Par de Transformadas}
\vspace{-2mm}
\[
\boxed{\delta(t) \xleftrightarrow{\ft} 1}
\]
\vspace{-2mm}
\end{block}

\textbf{Interpretação:} Um impulso no tempo contém \textbf{todas as frequências} com amplitude igual.

\textbf{Delta deslocada:} $\delta(t - t_0) \xleftrightarrow{\ft} e^{-j\omega t_0}$

\end{frame}

% ============================================

\begin{frame}{Exemplo: Resposta ao Impulso de um Canal}

\textbf{Cenário:} Um canal de comunicação tem resposta ao impulso
\[
h(t) = \alpha_1\,\delta(t - \tau_1) + \alpha_2\,\delta(t - \tau_2)
\]
(dois caminhos com atrasos $\tau_1$, $\tau_2$ e ganhos $\alpha_1$, $\alpha_2$)

\vspace{0.2cm}

\textbf{Resposta em frequência:}
\[
H(\omega) = \alpha_1 e^{-j\omega\tau_1} + \alpha_2 e^{-j\omega\tau_2}
\]

\textbf{Consequência prática:}
\begin{itemize}
\item $|H(\omega)|$ varia com $\omega$ $\Rightarrow$ \textbf{desvanecimento seletivo em frequência}
\item Certas frequências são atenuadas (interferência destrutiva entre caminhos)
\item Este é o modelo básico de \textbf{multipercurso} em canais sem fio
\end{itemize}

\end{frame}

\subsection{Constante}

\begin{frame}{Transformada de uma Constante}

Queremos a transformada de $f(t) = A$ (constante para todo $t$).

\textbf{Problema:} A integral $\int_{-\infty}^{\infty} A e^{-j\omega t}\,dt$ não converge classicamente.

\textbf{Solução via dualidade:} Como $\delta(t) \xleftrightarrow{\ft} 1$, a propriedade de
dualidade ($F(t) \xleftrightarrow{\ft} 2\pi f(-\omega)$) implica diretamente:
\[
1 \xleftrightarrow{\ft} 2\pi\,\delta(-\omega) = 2\pi\,\delta(\omega)
\]

\begin{block}{Par de Transformadas}
\[
\boxed{A \xleftrightarrow{\ft} 2\pi A\,\delta(\omega)}
\]
\end{block}

\textbf{Interpretação:} Um sinal constante tem toda a sua energia em $\omega = 0$ (componente DC puro).

\end{frame}

% ============================================

\subsection{Exponencial Complexa}

\begin{frame}{Transformada da Exponencial Complexa}

Considere $f(t) = e^{j\omega_0 t}$ (senoidal complexa na frequência $\omega_0$).

\[
F(\omega) = \int_{-\infty}^{\infty} e^{j\omega_0 t} e^{-j\omega t} dt = \int_{-\infty}^{\infty} e^{-j(\omega - \omega_0) t} dt
\]

Esta integral é $2\pi\delta(\omega - \omega_0)$:

\begin{block}{Par de Transformadas}
\[
\boxed{e^{j\omega_0 t} \xleftrightarrow{\ft} 2\pi \delta(\omega - \omega_0)}
\]
\end{block}

\end{frame}

% ============================================

\begin{frame}{Interpretação: Exponencial Complexa no Espectro}

\textbf{Significado:} Uma senoidal complexa de frequência $\omega_0$ tem espectro com um \textbf{único impulso} em $\omega = \omega_0$. É um sinal \textbf{monocromático} puro.

\begin{frame}{Transformadas de Cosseno e Seno}

Usando Euler ($\cos\omega_0 t = \tfrac{e^{j\omega_0 t}+e^{-j\omega_0 t}}{2}$) e a linearidade:

\[
\FT{\cos(\omega_0 t)}
  = \tfrac{1}{2}\cdot 2\pi\delta(\omega-\omega_0) + \tfrac{1}{2}\cdot 2\pi\delta(\omega+\omega_0)
  = \pi\bigl[\delta(\omega-\omega_0)+\delta(\omega+\omega_0)\bigr]
\]

Analogamente, com $\sin\omega_0 t = \tfrac{e^{j\omega_0 t}-e^{-j\omega_0 t}}{2j}$:

\begin{block}{Pares de Transformadas}
\[
\cos(\omega_0 t) \xleftrightarrow{\ft} \pi\bigl[\delta(\omega - \omega_0) + \delta(\omega + \omega_0)\bigr]
\]
\[
\sin(\omega_0 t) \xleftrightarrow{\ft} j\pi\bigl[\delta(\omega + \omega_0) - \delta(\omega - \omega_0)\bigr]
\]
\end{block}

\textbf{Interpretação:} Cada sinal gera \textbf{dois impulsos} espectrais: em $+\omega_0$ e $-\omega_0$.

\end{frame}

\subsection{Função Degrau}

\begin{frame}{Função Degrau Unitário}

A \textbf{função degrau unitário} (Heaviside):

\[
u(t) = \begin{cases}
1 & t > 0 \\
1/2 & t = 0 \\
0 & t < 0
\end{cases}
\]

\textbf{Relação com a função sinal:} $\;\sgn(t) = 2u(t) - 1$, ou seja $u(t) = \frac{1 + \sgn(t)}{2}$

\end{frame}

% ============================================

\begin{frame}{Função Sinal (sgn)}

\begin{columns}
\begin{column}{0.45\textwidth}
\[
\sgn(t) = \begin{cases}
1 & t > 0 \\
0 & t = 0 \\
-1 & t < 0
\end{cases}
\]

\vspace{0.3cm}

Usaremos $\sgn(t)$ como passo intermediário para obter $\mathcal{F}\{u(t)\}$.
\end{column}

\begin{frame}{Transformada da Função Sinal: Método}

Usamos um \textbf{limite regulador}. Sabemos que:

\[
e^{-at}u(t)\xleftrightarrow{\ft}\frac{1}{a+j\omega} \qquad e^{at}u(-t)\xleftrightarrow{\ft}\frac{1}{a-j\omega}
\]

Escrevemos $\sgn(t)$ como limite:

\[
\sgn(t) = \lim_{a\to 0^+}\!\bigl[e^{-at}u(t)-e^{at}u(-t)\bigr]
\]

\end{frame}

% ============================================

\begin{frame}{Transformada da Função Sinal: Resultado}

\[
\FT{\sgn(t)}
  = \lim_{a\to 0^+}\!\left[\frac{1}{a+j\omega}-\frac{1}{a-j\omega}\right]
   = \lim_{a\to 0^+}\frac{-2j\omega}{a^2+\omega^2}
  = \frac{2}{j\omega} \quad (\omega\neq 0)
\]

\begin{block}{Transformada da Função Sinal}
\[
\boxed{\sgn(t) \xleftrightarrow{\ft} \frac{2}{j\omega}}
\]
\end{block}

\textbf{Observação:} $\frac{2}{j\omega} = \frac{-2j}{\omega}$ — espectro puramente imaginário (sinal ímpar $\Rightarrow$ transformada imaginária).

\end{frame}

% ============================================

\begin{frame}{Transformada do Degrau Unitário}

Usando $u(t) = \frac{1 + \sgn(t)}{2}$ com os pares conhecidos
($1 \xleftrightarrow{\ft} 2\pi\delta(\omega)$ e $\sgn(t) \xleftrightarrow{\ft} \frac{2}{j\omega}$):

\[
\FT{u(t)}
  = \tfrac{1}{2}\cdot 2\pi\delta(\omega) + \tfrac{1}{2}\cdot\frac{2}{j\omega}
  = \pi\delta(\omega) + \frac{1}{j\omega}
\]

\begin{block}{Transformada do Degrau}
\[
u(t) \xleftrightarrow{\ft} \pi\delta(\omega) + \frac{1}{j\omega}
\]
\end{block}

\textbf{Interpretação:}
\begin{itemize}
\item Componente DC: $\pi\delta(\omega)$ — valor médio de $1/2$
\item Componente variável: $1/(j\omega)$ — conteúdo espectral contínuo
\end{itemize}

\end{frame}

\subsection{Funções de Pulso}

\begin{frame}{Pulso Retangular}

\[
\rect(t/\tau) = \begin{cases}
1 & |t| \leq \tau/2 \\
0 & |t| > \tau/2
\end{cases}
\]

\begin{block}{Transformada}
\[
\boxed{\rect(t/\tau) \xleftrightarrow{\ft} \tau \sinc\left(\frac{\omega\tau}{2\pi}\right)}
\]
\end{block}

onde $\sinc(x) = \frac{\sin(\pi x)}{\pi x}$.

\end{frame}

% ============================================

\begin{frame}{Propriedades do Espectro do Pulso Retangular}

\begin{itemize}
\item Lóbulo principal: $-2\pi/\tau < \omega < 2\pi/\tau$ (largura $= 4\pi/\tau$)
\item Zeros em $\omega = \pm 2\pi n/\tau$, $n = 1, 2, 3, \ldots$
\item Amplitude decresce como $1/\omega$
\item Pulso mais estreito ($\tau$ menor) $\Rightarrow$ espectro mais largo
\end{itemize}

\vspace{0.3cm}

\end{tikzpicture}
\end{center}

\end{frame}

% ============================================

\begin{frame}{Exemplo: Transmissão de Bits com Pulsos Retangulares}

\textbf{Cenário:} Em transmissão digital NRZ (Non-Return-to-Zero), cada bit é representado por um pulso retangular de duração $T_b$:

\[
s(t) = \sum_k a_k \, \rect\!\left(\frac{t - kT_b}{T_b}\right), \quad a_k \in \{+1, -1\}
\]

\textbf{Espectro de cada pulso:}
$\rect(t/T_b) \xleftrightarrow{\ft} T_b \sinc(\omega T_b / 2\pi)$

\vspace{0.2cm}

\textbf{Consequências:}
\begin{itemize}
\item A largura de banda ocupada é $\approx 1/T_b$ Hz (lóbulo principal da sinc)
\item Taxa de bits mais alta ($T_b$ menor) $\Rightarrow$ maior largura de banda
\item Lóbulos laterais causam \textbf{interferência em canais adjacentes}
\item Por isso, na prática usa-se formatação de pulso (ex: cosseno levantado)
\end{itemize}

\end{frame}

\begin{frame}{Função Triangular}

A \textbf{função triangular}:

\[
\tri(t/\tau) = \begin{cases}
1 - |t|/\tau & |t| \leq \tau \\
0 & |t| > \tau
\end{cases}
\]

\textbf{Observação:} $\tri(t/\tau) = \rect(t/\tau) \conv \rect(t/\tau)$ (convolução de retângulo consigo mesmo).

\begin{block}{Transformada}
\[
\boxed{\tri(t/\tau) \xleftrightarrow{\ft} \tau \sinc^2\!\left(\frac{\omega\tau}{2\pi}\right)}
\]
\end{block}

\end{frame}

% ============================================

\begin{frame}{Função Triangular: Justificativa}

Pela propriedade de convolução (que veremos na próxima seção):

\[
\FT{\tri(t/\tau)} = \FT{\rect(t/\tau)} \cdot \FT{\rect(t/\tau)} = \left[\tau \sinc\!\left(\frac{\omega\tau}{2\pi}\right)\right]^2
\]

\vspace{0.3cm}

\textbf{Comparação com o pulso retangular:}
\begin{itemize}
\item Retângulo: espectro $\propto \sinc$ --- decai como $1/\omega$
\item Triângulo: espectro $\propto \sinc^2$ --- decai como $1/\omega^2$ (mais rápido)
\item Sinal mais ``suave'' no tempo $\Rightarrow$ espectro decai mais rápido
\end{itemize}

\end{frame}

% ============================================

\begin{frame}{Exemplo: Rect vs.\ Tri na Formatação de Pulso}

\textbf{Cenário:} Comparando pulsos retangulares e triangulares para transmissão digital.

\vspace{0.2cm}

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{Pulso retangular:}
\begin{itemize}
\item Espectro: $\sinc(\omega)$
\item Decai como $1/\omega$
\item Lóbulos laterais significativos
\item Mais interferência entre canais
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{Pulso triangular:}
\begin{itemize}
\item Espectro: $\sinc^2(\omega)$
\item Decai como $1/\omega^2$
\item Lóbulos laterais menores
\item Menos interferência entre canais
\end{itemize}
\end{column}
\end{columns}

\vspace{0.3cm}

\textbf{Trade-off:} O pulso triangular tem lóbulos laterais menores (menos interferência fora da banda), porém seu lóbulo principal é mais largo --- ou seja, ocupa mais largura de banda para transmitir na mesma taxa.

\end{frame}

\subsection{Função Gaussiana}

\begin{frame}{Pulso Gaussiano}

A \textbf{função Gaussiana}:

\[
f(t) = e^{-\alpha t^2}, \quad \alpha > 0
\]

\textbf{Cálculo da transformada:}

\[
F(\omega) = \int_{-\infty}^{\infty} e^{-\alpha t^2} e^{-j\omega t} dt
\]

Completando o quadrado no expoente:
\[
-\alpha t^2 - j\omega t = -\alpha\left(t + \frac{j\omega}{2\alpha}\right)^2 - \frac{\omega^2}{4\alpha}
\]

\end{frame}

% ============================================

\begin{frame}{Pulso Gaussiano (continuação)}

Fatorando a exponencial:
\[
F(\omega) = e^{-\omega^2/(4\alpha)} \int_{-\infty}^{\infty} e^{-\alpha\left(t + \frac{j\omega}{2\alpha}\right)^2} dt
\]

Com $u = t + j\omega/(2\alpha)$ e a integral gaussiana $\int e^{-\alpha u^2}du = \sqrt{\pi/\alpha}$:

\[
F(\omega) = e^{-\omega^2/(4\alpha)}\cdot\sqrt{\tfrac{\pi}{\alpha}}
\]

\end{frame}

% ============================================

\begin{frame}{Pulso Gaussiano: Resultado}

\begin{block}{Transformada da Gaussiana}
\[
e^{-\alpha t^2} \xleftrightarrow{\ft} \sqrt{\frac{\pi}{\alpha}}\,e^{-\omega^2/(4\alpha)}
\]
\end{block}

\vspace{0.3cm}

\begin{itemize}
\item \textbf{Propriedade notável:} Uma Gaussiana transforma-se em outra Gaussiana!
\item \textbf{Princípio de incerteza:} A Gaussiana é o sinal com mínimo produto $\Delta t \cdot \Delta\omega = 1/2$.
\item $\alpha$ grande $\Rightarrow$ pulso estreito no tempo, espectro largo na frequência
\end{itemize}

\end{frame}

% ============================================

\begin{frame}{Exemplo: Pulsos Gaussianos em Fibra Óptica}

\textbf{Cenário:} Em sistemas de comunicação por fibra óptica, os pulsos de luz são bem aproximados por gaussianas.

\vspace{0.2cm}

Um pulso óptico $p(t) = e^{-t^2/(2\sigma^2)}$ ao se propagar sofre \textbf{dispersão cromática}, que alarga o pulso no tempo.

\vspace{0.2cm}

\textbf{Por que gaussianas?}
\begin{itemize}
\item Transformada de uma gaussiana é outra gaussiana --- a \textbf{forma se preserva} durante a propagação
\item O princípio de incerteza garante: $\Delta t \cdot \Delta \omega \geq 1/2$, com igualdade para gaussianas
\item Pulsos gaussianos são os mais ``eficientes'' em ocupação tempo-frequência
\end{itemize}

\vspace{0.2cm}

\textbf{Na prática:} A taxa máxima de transmissão em fibra depende diretamente da largura $\sigma$ dos pulsos --- pulsos que se alargam demais causam \textbf{interferência intersimbólica} (ISI).

\end{frame}

\subsection{Trem de Impulsos}

\begin{frame}{Trem de Impulsos Periódico}

Um \textbf{trem de impulsos} (impulse train) com período $T$:

\[
f(t) = \sum_{n=-\infty}^{\infty} \delta(t - nT)
\]

Este sinal é periódico ($\omega_0 = 2\pi/T$). Os coeficientes de Fourier valem:
\[
c_n = \frac{1}{T}\!\int_{-T/2}^{T/2}\!\delta(t)\,e^{-jn\omega_0 t}\,dt = \frac{1}{T}
\]

\textbf{Interpretação:} Todos os harmônicos têm a mesma amplitude $1/T$ — espectro plano!

\end{frame}

% ============================================

\begin{frame}{Trem de Impulsos: Expansão em Série de Fourier}

Como $c_n = 1/T$ para todo $n$, a \textbf{série de Fourier} do trem de impulsos é:

\[
f(t) = \sum_{n=-\infty}^{\infty} c_n\, e^{jn\omega_0 t}
     = \frac{1}{T}\sum_{n=-\infty}^{\infty} e^{jn\omega_0 t}
\]

Ou seja, o trem de impulsos é uma soma de exponenciais complexas de todas as frequências harmônicas $n\omega_0$, cada uma com peso $1/T$.

\vspace{0.3cm}

Agora basta aplicar a transformada de Fourier a cada exponencial separadamente (linearidade).

\end{frame}

% ============================================

\begin{frame}{Transformada do Trem de Impulsos}

Aplicando $\mathcal{F}$ termo a termo na série, e usando
$e^{jn\omega_0 t} \xleftrightarrow{\ft} 2\pi\delta(\omega - n\omega_0)$:

\[
F(\omega)
  = \FT{\frac{1}{T}\sum_{n=-\infty}^{\infty} e^{jn\omega_0 t}}
  = \frac{1}{T}\sum_{n=-\infty}^{\infty} 2\pi\delta(\omega - n\omega_0)
\]

\[
F(\omega) = \frac{2\pi}{T}\sum_{n=-\infty}^{\infty}\delta(\omega - n\omega_0)
\]

\begin{block}{Par de Transformadas: Trem de Impulsos}
\[
\sum_{n=-\infty}^{\infty} \delta(t - nT)
\;\xleftrightarrow{\ft}\;
\frac{2\pi}{T} \sum_{n=-\infty}^{\infty} \delta\!\left(\omega - \frac{2\pi n}{T}\right)
\]
\end{block}

\end{frame}

% ============================================

\begin{frame}{Trem de Impulsos: Interpretação}

\begin{itemize}
\item \textbf{Simetria perfeita:} impulsos periódicos no tempo $\leftrightarrow$ impulsos periódicos na frequência
\item \textbf{Separação espectral:} $\Delta\omega = 2\pi/T$ (inversamente proporcional a $T$)
\item \textbf{Aplicação fundamental:} base matemática do teorema de amostragem
\end{itemize}

\begin{frame}{Observações e Padrões}

\textbf{Padrões observados:}

\begin{enumerate}
\item \textbf{Dualidade tempo-frequência:}
   \begin{itemize}
   \item Impulso no tempo $\leftrightarrow$ constante na frequência
   \item Constante no tempo $\leftrightarrow$ impulso na frequência
   \item Impulsos periódicos $\leftrightarrow$ impulsos periódicos
   \end{itemize}

\item \textbf{Decaimento:}
   \begin{itemize}
   \item Decaimento exponencial $\leftrightarrow$ Lorentziana
   \item Gaussiana $\leftrightarrow$ Gaussiana
   \end{itemize}

\item \textbf{Suavidade:}
   \begin{itemize}
   \item Descontinuidades $\rightarrow$ decaimento lento do espectro ($1/\omega$)
   \item Continuidade $\rightarrow$ decaimento mais rápido ($1/\omega^2$ ou exponencial)
   \end{itemize}
\end{enumerate}

\end{frame}
