\begin{frame}{Série de Fourier: a forma trigonométrica}

Se $f(t)$ é periódico com período $T_0$ e frequência fundamental
$\omega_0 = 2\pi/T_0$:

\vspace{0.35cm}
\[
f(t) \;=\; \tikzmarknode{sfA}{a_0}
\;+\; \sum_{n=1}^{\infty}\Big[\,
\tikzmarknode{sfB}{a_n}\cos\big(\tikzmarknode{sfC}{n\omega_0}\, t\big)
\;+\; b_n \sin\big(n\omega_0\, t\big)\Big]
\]

\anotacoes{%
  \only<2->{
    \node[nota=cGreen, text width=1.6cm] (sfNA) at ($(sfA)+(-2.0,-1.15)$)
      {valor médio de $f(t)$};
    \draw[setinha, cGreen] (sfNA.east) -- (sfA.south);}
  \only<3->{
    \node[nota=cOrange, text width=2.3cm] (sfNC) at ($(sfC)+(2.9,-1.15)$)
      {só múltiplos inteiros de $\omega_0$};
    \draw[setinha, cOrange] (sfNC.west) -- (sfC.south east);}
  \only<4->{
    \node[nota=cBlue, text width=1.9cm] (sfNB) at ($(sfB)+(0,-2.4)$)
      {peso de cada harmônica};
    \draw[setinha, cBlue] (sfNB.north) -- (sfB.south);}
}%

\vspace{2.7cm}

\onslide<5->{\alert{Um sinal periódico só contém frequências discretas.}}

\end{frame}

% --------------------------------------------
\begin{frame}{Euler entra na série}

Duas identidades transformam senos e cossenos em exponenciais:

\vspace{-0.15cm}
\[
\tikzmarknode{euC}{\cos\theta = \frac{e^{j\theta}+e^{-j\theta}}{2}}
\qquad
\tikzmarknode{euS}{\sin\theta = \frac{e^{j\theta}-e^{-j\theta}}{2j}}
\]

\onslide<2->{Aplicamos as duas ao termo de ordem $n$ da série:}

\vspace{-0.1cm}
\onslide<2->{
\[
\tikzmarknode{trA}{a_n\cos(n\omega_0 t)} \;+\; \tikzmarknode{trB}{b_n\sin(n\omega_0 t)}
\]}

\anotacoes{%
  \only<3->{\draw[setinha, cGreen] (euC.south) to[out=-90, in=100] (trA.north);}
  \only<4->{\draw[setinha, cOrange] (euS.south) to[out=-90, in=80] (trB.north);}
}%

\vspace{0.1cm}
\onslide<5->{
\[
=\; \hl[cGreen]{\frac{a_n}{2}}\big(e^{jn\omega_0 t}+e^{-jn\omega_0 t}\big)
\;+\; \hl[cOrange]{\frac{b_n}{2j}}\big(e^{jn\omega_0 t}-e^{-jn\omega_0 t}\big)
\]}

\end{frame}

% --------------------------------------------
\begin{frame}{Agrupando as exponenciais}

Lembrando que $1/j = -j$ e juntando os termos de mesma exponencial:

\vspace{0.1cm}
\[
\tikzmarknode{agA}{\frac{a_n - j b_n}{2}}\, e^{jn\omega_0 t}
\;+\;
\tikzmarknode{agB}{\frac{a_n + j b_n}{2}}\, e^{-jn\omega_0 t}
\]

\anotacoes{%
  \only<2->{
    \node[nota=cRed, text width=1.5cm] (agNA) at ($(agA)+(-2.2,-1.25)$) {chame de $c_n$};
    \draw[setinha, cRed] (agNA.east) -- (agA.south);}
  \only<3->{
    \node[nota=cPurple, text width=2.4cm] (agNB) at ($(agB)+(2.5,-1.25)$)
      {é o conjugado, $c_{-n} = c_n^{*}$};
    \draw[setinha, cPurple] (agNB.west) -- (agB.south);}
}%

\vspace{1.5cm}

\onslide<4->{Somando sobre todos os $n$, positivos e negativos:}

\vspace{-0.1cm}
\onslide<5->{
\[
\boxed{\;f(t) \;=\; \sum_{n=-\infty}^{\infty} c_n\, e^{jn\omega_0 t}\;}
\]}

\end{frame}

% --------------------------------------------
\begin{frame}{Da exponencial para a forma polar}

Escrevendo o coeficiente complexo em módulo e fase:

\vspace{-0.1cm}
\[
c_n = |c_n|\, e^{j\theta_n},
\qquad
c_{-n} = |c_n|\, e^{-j\theta_n}
\]

\onslide<2->{O par conjugado se recombina em um único cosseno real:}

\vspace{-0.1cm}
\onslide<3->{
\[
c_n e^{jn\omega_0 t} + c_{-n} e^{-jn\omega_0 t}
\;=\; 2|c_n|\cos\big(n\omega_0 t + \theta_n\big)
\]}

\vspace{0.1cm}
\onslide<4->{\alert{A soma de um par conjugado de exponenciais é sempre um
cosseno real.}}

\end{frame}

% --------------------------------------------
\begin{frame}{A forma polar da série}

Somando os pares conjugados de $n=1$ até o infinito:

\vspace{0.2cm}
\[
\boxed{\;f(t) \;=\; C_0 \;+\; \sum_{n=1}^{\infty}
\tikzmarknode{poA}{C_n}\cos\big(n\omega_0 t + \tikzmarknode{poB}{\theta_n}\big)\;}
\]

\anotacoes{%
  \only<2->{
    \node[nota=cRed, text width=2.0cm] (poNA) at ($(poA)+(-2.6,-1.3)$)
      {amplitude $C_n = 2|c_n|$};
    \draw[setinha, cRed] (poNA.east) -- (poA.south);}
  \only<3->{
    \node[nota=cBlue, text width=2.0cm] (poNB) at ($(poB)+(2.6,-1.3)$)
      {fase $\theta_n = \arg c_n$};
    \draw[setinha, cBlue] (poNB.west) -- (poB.south);}
}%

\vspace{1.6cm}

\onslide<4->{\alert{Cada harmônica fica descrita por uma amplitude e uma fase.}}

\onslide<5->{São exatamente as duas informações que o espectro de magnitude e o
espectro de fase carregam.}

\end{frame}

% --------------------------------------------
\begin{frame}{Isolando um coeficiente: multiplicar}

Partimos da série:

\vspace{-0.1cm}
\[
f(t) \;=\; \sum_{n=-\infty}^{\infty} c_n\, e^{jn\omega_0 t}
\]

\onslide<2->{Multiplicamos os dois lados por $e^{-jk\omega_0 t}$, com $k$ inteiro:

\vspace{-0.15cm}
\[
f(t)\,e^{-jk\omega_0 t} \;=\; \sum_{n=-\infty}^{\infty} c_n\,
\tikzmarknode{muA}{e^{jn\omega_0 t}\, e^{-jk\omega_0 t}}
\]}

\anotacoes{%
  \only<3->{
    \node[nota=cOrange, text width=3.0cm] (muN) at ($(muA)+(0.2,-1.3)$)
      {mesma base: os expoentes se somam};
    \draw[setinha, cOrange] (muN.north) -- (muA.south);}
}%

\vspace{1.5cm}

\end{frame}

% --------------------------------------------
\begin{frame}{Isolando um coeficiente: integrar}

Somando os expoentes:

\vspace{-0.1cm}
\[
f(t)\,e^{-jk\omega_0 t} \;=\; \sum_{n=-\infty}^{\infty} c_n\, e^{j(n-k)\omega_0 t}
\]

\onslide<2->{Integrando os dois lados ao longo de um período:

\vspace{-0.15cm}
\[
\int_{-T_0/2}^{T_0/2} f(t)\, e^{-jk\omega_0 t}\, dt
\;=\; \sum_{n=-\infty}^{\infty} c_n
\tikzmarknode{orA}{\int_{-T_0/2}^{T_0/2} e^{j(n-k)\omega_0 t}\, dt}
\]}

\anotacoes{%
  \only<3->{
    \node[nota=cRed, text width=3.6cm] (orNA) at ($(orA)+(-0.7,-1.4)$)
      {tudo depende desta integral. Quanto ela vale?};
    \draw[setinha, cRed] (orNA.north) -- (orA.south);}
}%

\vspace{1.5cm}

\onslide<4->{Chame $m = n-k$ e resolva nos dois casos possíveis.}

\end{frame}

% --------------------------------------------
\begin{frame}{Resolvendo a integral: caso $m = n-k \neq 0$}

\[
I_m \;=\; \int_{-T_0/2}^{T_0/2} e^{jm\omega_0 t}\, dt
\]

\onslide<2->{
\[
=\; \left[\frac{e^{jm\omega_0 t}}{jm\omega_0}\right]_{-T_0/2}^{T_0/2}
\]}

\onslide<3->{
\[
=\; \frac{1}{jm\omega_0}\Big(e^{\,jm\,\tikzmarknode{inB}{\omega_0 T_0/2}}
- e^{-jm\,\omega_0 T_0/2}\Big)
\]}

\anotacoes{%
  \only<4->{
    \node[nota=cGreen, text width=2.6cm] (inNB) at ($(inB)+(2.7,-1.1)$)
      {$\omega_0 T_0/2 = \pi$, pois $\omega_0 = 2\pi/T_0$};
    \draw[setinha, cGreen] (inNB.west) -- (inB.south east);}
}%

\vspace{1.5cm}

\end{frame}

% --------------------------------------------
\begin{frame}{Resolvendo a integral: o resultado é zero}

Pela identidade de Euler, a diferença no numerador é um seno:

\vspace{-0.1cm}
\[
I_m \;=\; \frac{e^{jm\pi} - e^{-jm\pi}}{jm\omega_0}
\;=\; \frac{2j\,\tikzmarknode{znA}{\sin(m\pi)}}{jm\omega_0}
\]

\anotacoes{%
  \only<2->{
    \node[nota=cRed, text width=3.4cm] (znNA) at ($(znA)+(0.3,-1.35)$)
      {$m$ é inteiro, logo $\sin(m\pi) = 0$};
    \draw[setinha, cRed] (znNA.north) -- (znA.south);}
}%

\vspace{1.6cm}

\onslide<3->{
\[
\boxed{\;I_m = 0 \quad \text{para todo } m \neq 0\;}
\]}

\onslide<4->{\alert{Exponenciais de harmônicas diferentes se cancelam ao longo
de um período.}}

\end{frame}

% --------------------------------------------
\begin{frame}{Resolvendo a integral: caso $m = 0$}

Quando $n = k$ o expoente desaparece:

\vspace{-0.1cm}
\[
I_0 \;=\; \int_{-T_0/2}^{T_0/2} \tikzmarknode{i0A}{e^{\,j\cdot 0\cdot \omega_0 t}}\, dt
\]

\anotacoes{%
  \only<2->{
    \node[nota=cGreen, text width=2.0cm] (i0NA) at ($(i0A)+(2.6,-1.1)$)
      {vale $1$ para todo $t$};
    \draw[setinha, cGreen] (i0NA.west) -- (i0A.south east);}
}%

\vspace{1.4cm}

\onslide<3->{
\[
=\; \int_{-T_0/2}^{T_0/2} dt \;=\; \frac{T_0}{2} - \left(-\frac{T_0}{2}\right)
\;=\; T_0
\]}

\end{frame}

% --------------------------------------------
\begin{frame}{A soma colapsa}

\vspace{-0.15cm}
\[
\boxed{\;I_{n-k} = \int_{-T_0/2}^{T_0/2} e^{j(n-k)\omega_0 t}\, dt =
\begin{cases} T_0, & n = k \\[2pt] 0, & n \neq k \end{cases}\;}
\]

\onslide<2->{Voltando à soma:}

\vspace{-0.1cm}
\[
\int_{-T_0/2}^{T_0/2} f(t)\, e^{-jk\omega_0 t}\, dt
\;=\; \sum_{n=-\infty}^{\infty} c_n\,
\underbrace{\tikzmarknode{cpA}{I_{n-k}}}_{\text{zero, exceto } n=k}
\]

\anotacoes{%
  \only<3->{
    \node[nota=cRed, text width=3.0cm] (cpNA) at ($(cpA)+(-0.2,-1.5)$)
      {sobra apenas $c_k\,T_0$};
    \draw[setinha, cRed] (cpNA.north) -- (cpA.south);}
}%

\vspace{1.7cm}

\onslide<4->{\alert{A ortogonalidade permite isolar um coeficiente por vez.}}

\end{frame}

% --------------------------------------------

\begin{frame}{Os coeficientes da série}

Sobrando apenas o termo $n=k$:

\[
\int_{-T_0/2}^{T_0/2} f(t)\, e^{-jk\omega_0 t}\, dt \;=\; c_k\, T_0
\]

\onslide<2->{Isolando o coeficiente:

\[
\boxed{\;c_k = \frac{1}{T_0}\int_{-T_0/2}^{T_0/2} f(t)\, e^{-jk\omega_0 t}\, dt\;}
\]}

\vspace{0.3cm}

\onslide<3->{\alert{Analisar é projetar o sinal sobre cada exponencial.}}

\onslide<4->{Sintetizar é somar essas projeções de volta, com pesos $c_k$.}

\end{frame}

\subsection{Derivação da Transformada de Fourier}

% --------------------------------------------
\begin{frame}{Passo 1: colocar $c_n$ dentro da série}

Partimos das duas equações da série de Fourier:

\vspace{-0.15cm}
\[
f(t) = \sum_{n=-\infty}^{\infty} c_n e^{jn\omega_0 t},
\qquad
c_n = \frac{1}{T_0}\int_{-T_0/2}^{T_0/2} f(\tau)\, e^{-jn\omega_0 \tau}\, d\tau
\]

\onslide<2->{Substituindo a segunda na primeira:}

\vspace{-0.1cm}
\onslide<3->{
\[
f(t) = \sum_{n=-\infty}^{\infty}
\hl[cGreen]{\frac{1}{T_0}}
\left[\;\tikzmarknode{p1A}{\int_{-T_0/2}^{T_0/2} f(\tau)\, e^{-jn\omega_0 \tau}\, d\tau}\;\right]
e^{jn\omega_0 t}
\]}

\anotacoes{%
  \only<4->{
    \node[nota=cRed, text width=3.0cm] (p1NA) at ($(p1A)+(0,-1.3)$)
      {depende de $n$ só através de $n\omega_0$};
    \draw[setinha, cRed] (p1NA.north) -- (p1A.south);}
}%

\vspace{1.4cm}

\subsection{Espectro de magnitude e fase}

\begin{frame}{Representação espectral}

A transformada $F(\omega)$ é geralmente complexa:

\[
F(\omega) = |F(\omega)| e^{j\phi(\omega)} = \Real\{F(\omega)\} + j\Imag\{F(\omega)\}
\]

onde:

\begin{itemize}
\item<2-> Espectro de magnitude: $|F(\omega)| = \sqrt{\Real^2\{F(\omega)\} + \Imag^2\{F(\omega)\}}$
\item<3-> Espectro de fase: $\phi(\omega) = \arctan\left(\frac{\Imag\{F(\omega)\}}{\Real\{F(\omega)\}}\right)$
\end{itemize}

\vspace{0.4cm}

\onslide<4->{O espectro completo exige as duas informações, magnitude e fase.}

\onslide<5->{Para sinais reais valem simetrias úteis:
\begin{itemize}
\item $|F(-\omega)| = |F(\omega)|$, magnitude par
\item $\phi(-\omega) = -\phi(\omega)$, fase ímpar
\end{itemize}}

\end{frame}

% ============================================

\subsection{Exemplos}

\begin{frame}{Exemplo 1: exponencial decrescente}

Considere um sinal exponencial causal:

\[
f(t) = \begin{cases}
e^{-at} & t \geq 0 \\
0 & t < 0
\end{cases} = e^{-at} u(t), \quad a > 0
\]

onde $u(t)$ é a função degrau unitário.

\onslide<2->{Cálculo da transformada:

\[
F(\omega) = \int_{-\infty}^{\infty} f(t) e^{-j\omega t} dt = \int_{0}^{\infty} e^{-at} e^{-j\omega t} dt
\]}

\onslide<3->{
\[
= \int_{0}^{\infty} e^{-(a+j\omega)t} dt = \left[ \frac{e^{-(a+j\omega)t}}{-(a+j\omega)} \right]_{0}^{\infty}
\]}

\end{frame}

% ============================================

\begin{frame}{Exemplo 1: exponencial decrescente, continuação}

Como $a > 0$, temos $e^{-(a+j\omega)t} \to 0$ quando $t \to \infty$:

\[
F(\omega) = 0 - \frac{1}{-(a+j\omega)} = \frac{1}{a+j\omega}
\]

\onslide<2->{Racionalizando pelo conjugado:

\[
F(\omega) = \frac{1}{a+j\omega} \cdot \frac{a-j\omega}{a-j\omega} = \frac{a-j\omega}{a^2+\omega^2}
\]}

\end{frame}

% ============================================

\begin{frame}{Exemplo 1: resultado final}

Separando parte real e imaginária:

\[
\boxed{F(\omega) = \frac{a}{a^2+\omega^2} - j\frac{\omega}{a^2+\omega^2}}
\]

\vspace{0.3cm}

\onslide<2->{Espectro de magnitude:
\[
|F(\omega)| = \frac{1}{\sqrt{a^2+\omega^2}}
\]}

\onslide<3->{Espectro de fase:
\[
\phi(\omega) = -\arctan\!\left(\frac{\omega}{a}\right)
\]}

\end{frame}

% ============================================

\begin{frame}{Exemplo 1: análise do resultado}

\[
\boxed{f(t) = e^{-at}u(t) \quad \xleftrightarrow{\ft} \quad F(\omega) = \frac{1}{a+j\omega}}
\]

Características do espectro:

\begin{itemize}
\item<2-> Magnitude $|F(\omega)| = 1/\sqrt{a^2+\omega^2}$, de forma Lorentziana
\item<3-> Fase $\phi(\omega) = -\arctan(\omega/a)$
\item<4-> Em $\omega = 0$ o valor é máximo, $|F(0)| = 1/a$
\item<5-> Em $\omega = \pm a$ há redução de 3\,dB, $|F(\omega)| = 1/(a\sqrt{2})$
\item<6-> Largura de banda $B \approx a$, ligada à constante de decaimento
\end{itemize}

\vspace{0.3cm}

\onslide<7->{\alert{Decaimento rápido no tempo produz espectro largo em frequência.}}

\end{frame}

\end{center}

\item<2-> Decaimento mais rápido no tempo, com $a$ maior, alarga o espectro
\item<3-> A frequência de corte de 3\,dB ocorre em $\omega = a$

\begin{frame}{Exemplo 2: pulso exponencial bilateral}

Considere um pulso exponencial bilateral simétrico:

\[
f(t) = e^{-a|t|}, \quad a > 0
\]

\onslide<2->{Este sinal pode ser decomposto em:
\[
f(t) = e^{-at}u(t) + e^{at}u(-t)
\]}

\onslide<3->{\alert{Cada parcela já tem transformada conhecida do Exemplo 1.}}

\end{frame}

% ============================================

\begin{frame}{Exemplo 2: cálculo da transformada}

Separando a integral em dois intervalos:

\[
F(\omega) = \underbrace{\int_{-\infty}^{0} e^{at} e^{-j\omega t} dt}_{t < 0} + \underbrace{\int_{0}^{\infty} e^{-at} e^{-j\omega t} dt}_{t > 0}
\]

\vspace{0.3cm}

Resolvendo cada integral:

\vspace{0.2cm}

\begin{itemize}
\item<2-> Para $t < 0$: \; $\displaystyle\int_{-\infty}^{0} e^{(a-j\omega)t} dt = \frac{1}{a-j\omega}$

\vspace{0.2cm}

\item<3-> Para $t > 0$: \; $\displaystyle\int_{0}^{\infty} e^{-(a+j\omega)t} dt = \frac{1}{a+j\omega}$
\end{itemize}

\end{frame}

% ============================================

\begin{frame}{Exemplo 2: resultado}

\[
F(\omega) = \frac{1}{a-j\omega} + \frac{1}{a+j\omega} = \frac{(a+j\omega) + (a-j\omega)}{(a-j\omega)(a+j\omega)}
\]

\onslide<2->{
\[
\boxed{F(\omega) = \frac{2a}{a^2+\omega^2}}
\]}

\vspace{0.3cm}

Observações:

\begin{itemize}
\item<3-> $F(\omega)$ é real, de fase zero, porque $f(t)$ é par
\item<4-> Espectro Lorentziano centrado em $\omega = 0$
\item<5-> $F(0) = 2/a$ \; e \; $F(\pm a) = 1/a$
\end{itemize}

\end{frame}

\end{frame}
