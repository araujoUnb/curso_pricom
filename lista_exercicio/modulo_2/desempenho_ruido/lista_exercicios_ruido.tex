\begin{instrucao}[Formulário resumido]
\small
\textbf{SNR de referência (banda base):} $\gamma \equiv \dfrac{P_R}{N_0 W}$.

\textbf{SNR de saída:} \quad
DSB-SC, SSB coerente: $(S/N)_o = \gamma$. \quad
AM convencional, alta SNR: $(S/N)_o = \eta\,\gamma$, com $\eta = \dfrac{\mu^2 P_{m_n}}{1 + \mu^2 P_{m_n}}$.

\textbf{FM (acima do limiar):} $(S/N)_o = 3\beta_f^2 P_{m_n}\,\gamma$, com $\beta_f = \Delta f/W$ e $P_{m_n} = P_M/\max|m|^2$. Para tom único, $(S/N)_o = \tfrac{3}{2}\beta^2\gamma$.

\textbf{Limiar FM:} $\gamma_{\text{th}} = 20(\beta + 1)$ (Proakis eq.~6.2.25); SNR no limiar: $60\,\beta^2(\beta+1)P_{m_n}$.

\textbf{Banda de Carson FM:} $B_c = 2(\Delta f + W) = 2(\beta + 1)W$.

\textbf{Cascata de duas portas (Friis):} $F = F_1 + (F_2 - 1)/G_1 + \cdots$, $F = 1 + T_e/T_0$, com $T_0 = 290$\,K.
\end{instrucao}

\section*{Questão 1 -- Comparação direta: DSB-SC, SSB e AM convencional}

Um canal sem fio apresenta atenuação total $L = 80$\,dB e ruído branco aditivo de PSD bilateral $N_0/2 = 10^{-15}$\,W/Hz. A mensagem é um processo aleatório com largura de banda $W = 5$\,kHz, potência $P_M = 0{,}5\;\text{W}$ e $\max|m(t)| = 2\;\text{V}$. Em todos os esquemas, $A_c = 100\;\text{V}$ e a carga é normalizada.

\begin{enumerate}[label=\alph*)]
  \item Calcule a potência transmitida $P_T$ em cada modulação: DSB-SC, SSB-USB e AM convencional com índice $\mu = 0{,}8$ aplicado ao sinal $m_n(t) = m(t)/2$ normalizado.
  \item Determine a potência recebida $P_R$, a SNR de referência $\gamma$ e a SNR de saída $(S/N)_o$, em decibéis, para os três sistemas. Considere demoduladores coerentes nos casos DSB-SC e SSB, e detector de envelope no AM em alta SNR.
  \item Compare as bandas ocupadas pelo sinal modulado em cada caso. Quem é o vencedor em \emph{eficiência espectral}? E em \emph{eficiência de potência}?
  \item Se o objetivo do projeto for $(S/N)_o \ge 30\;\text{dB}$, qual modulação atinge a meta? Para aquelas que não atingem, calcule o aumento, em dB, que deveria ser dado a $A_c$ para alcançá-la.
  \item Em qual cenário o detector de envelope opera \emph{abaixo do limiar}? Estime, em dB, a SNR mínima de entrada para que a hipótese de alta SNR seja válida.
\end{enumerate}

\section*{Questão 2 -- AM com mensagem real: eficiência e máxima SNR}

Considere um transmissor AM convencional operando com portadora $f_c = 1\;\text{MHz}$ e mensagem de voz cuja PSD ocupa $W = 4\;\text{kHz}$. A mensagem normalizada tem $\max|m_n(t)| = 1$ e potência média $P_{m_n} = 0{,}1$, típico de voz. A SNR de entrada do receptor é $\gamma = 25\;\text{dB}$.

\begin{enumerate}[label=\alph*)]
  \item Determine a eficiência de potência $\eta$ do AM em função do índice de modulação $\mu$ e mostre que $\eta$ é maximizada em $\mu = 1$. Calcule $\eta_{\max}$ para esta mensagem.
  \item Calcule $(S/N)_o$ em dB no receptor para $\mu = 0{,}3$, $\mu = 0{,}7$ e $\mu = 1$. Comente o ganho marginal de SNR ao aumentar $\mu$ próximo de $1$, e o risco operacional de operar em $\mu = 1$.
  \item Compare a eficiência $\eta_{\max}$ desta mensagem de voz com a do tom único ($P_{m_n} = 1/2$, $\eta_{\max} = 1/3$). Quantos dB se perde de SNR ao mudar de tom para voz com a mesma SNR de entrada?
  \item Demonstre que, abaixo do limiar do detector de envelope, o termo de sinal e o termo de ruído na saída deixam de ser aditivos. Use a expansão
  \[
    V_r(t) \approx V_n(t) + \frac{A_c n_c(t)}{V_n(t)}\bigl[1 + \mu m_n(t)\bigr], \quad \text{quando}\;\;A_c \ll V_n,
  \]
  e identifique o termo onde a mensagem aparece \emph{multiplicada} pelo ruído.
  \item Para evitar o limiar, projete a potência da portadora $A_c$ mínima de modo que a SNR de entrada $\gamma_{\text{in}} = P_R/(N_0 B_T) \ge 10\;\text{dB}$, com $B_T = 2W$ e $N_0 = 10^{-13}\;\text{W/Hz}$. Considere $P_R = A_c^2(1 + \mu^2 P_{m_n})/2$ na carga unitária e $\mu = 0{,}8$.
\end{enumerate}

\section*{Questão 3 -- FM e o compromisso banda$\leftrightarrow$SNR}

Um enlace FM transmite áudio com largura de banda $W = 15\;\text{kHz}$ por um canal AWGN. Os equipamentos disponíveis permitem variar o índice $\beta$ de $1$ a $10$. O receptor opera com $\gamma = 15\;\text{dB}$. A mensagem real tem $P_{m_n} = 0{,}2$.

\begin{enumerate}[label=\alph*)]
  \item Para $\beta = 1, 2, 5, 10$, complete a tabela com a banda de Carson $B_c$, a SNR de saída $(S/N)_o$ em dB e o ganho sobre a referência banda base, $G_{\text{FM}} = (S/N)_o/\gamma$ em dB.
  \item Trace, em um esboço $(S/N)_o$ em dB \emph{versus} $B_c$ em kHz, e identifique a inclinação assintótica do trade-off $\big[\text{dB}/\text{oitava de banda}\big]$. Justifique numericamente.
  \item Se a meta de projeto é $(S/N)_o \ge 35\;\text{dB}$, determine o $\beta$ mínimo. Mostre que esse $\beta$ pode \emph{não} ser realizável caso a banda alocada do canal seja $B_{\text{canal}} = 150\;\text{kHz}$.
  \item Repita o item (a) considerando agora $P_{m_n} = 1/2$ (tom único). Calcule a perda em dB de SNR quando se passa de tom para mensagem real, para o mesmo $\beta$.
  \item Demonstre, partindo de $(S/N)_o = 3\beta^2 P_{m_n}\gamma$ e $B_c = 2(\beta+1)W$, que para $\beta \gg 1$ o ganho em SNR cresce com o \emph{quadrado} da expansão de banda $\Omega = B_c/W$. Identifique a constante de proporcionalidade.
\end{enumerate}

\section*{Questão 4 -- Efeito de limiar e projeto de $\beta_{\max}$}

Um sistema FM deve atender à especificação $(S/N)_o = 40\;\text{dB}$, com mensagem de tom único ($P_{m_n} = 1/2$) e largura $W = 10\;\text{kHz}$. A banda do canal é $B_{\text{canal}} = 150\;\text{kHz}$ e o ruído no receptor tem $N_0 = 10^{-8}\;\text{W/Hz}$. A atenuação total do canal é $50\;\text{dB}$.

\begin{enumerate}[label=\alph*)]
  \item Determine $\beta_{\max}^{\text{banda}}$ permitido pela restrição $B_c \le B_{\text{canal}}$.
  \item Determine $\beta_{\max}^{\text{pot}}$ a partir da fórmula do limiar $(S/N)_o\big|_{\text{th}} = 60\beta^2(\beta+1)P_{m_n} \ge 10^4$. Resolva analiticamente a cúbica resultante e indique qual restrição é dominante: banda ou potência.
  \item Para o $\beta$ escolhido, calcule a potência recebida $P_R$ necessária para operar exatamente no limiar, e a potência transmitida correspondente $P_T$. Adicione $10\;\text{dB}$ de margem ao limiar e recalcule $P_T$.
  \item Explique fisicamente o efeito de limiar: o que ocorre com o fasor recebido quando a SNR cai abaixo do limiar? Por que isso gera \emph{cliques} audíveis na saída do FM comercial?
  \item Se incluirmos pré-ênfase com constante de tempo $\tau = 75\;\mu\text{s}$ (padrão FM broadcast), e a mensagem tiver PSD aproximadamente plana, qual o ganho adicional em SNR esperado? Considere o resultado de Proakis para pré-ênfase, $G_{\text{pre}} \approx \dfrac{(W/W_1)^3}{3[W/W_1 - \arctan(W/W_1)]}$, com $W_1 = 1/(2\pi\tau)$.
\end{enumerate}

\section*{Questão 5 -- Orçamento de enlace e ruído de cascata}

Um enlace de microondas em linha-de-visada com $d = 30\;\text{km}$ opera em $f_c = 2\;\text{GHz}$ e transmite áudio FM com $W = 4\;\text{kHz}$. As antenas têm ganho $G_T = G_R = 30\;\text{dBi}$. A potência do transmissor é $P_T = 10\;\text{W}$. A cadeia do receptor é: LNA com $G_1 = 25\;\text{dB}$ e $F_1 = 2\;\text{dB}$, seguido de mixer com $G_2 = -6\;\text{dB}$ e $F_2 = 10\;\text{dB}$, seguido de amplificador IF com $G_3 = 30\;\text{dB}$ e $F_3 = 8\;\text{dB}$. Temperatura ambiente $T_0 = 290\;\text{K}$ e $k = 1{,}38 \times 10^{-23}\;\text{J/K}$.

\begin{enumerate}[label=\alph*)]
  \item Calcule a perda de espaço livre $L_{\text{esp}} = (4\pi d/\lambda)^2$ em dB para $f_c = 2\;\text{GHz}$ e $d = 30\;\text{km}$.
  \item Determine a potência recebida $P_R$ na entrada do LNA, em dBm, considerando $\mathit{EIRP} = P_T G_T$ e $G_R$ na recepção.
  \item Calcule o \emph{noise figure} total $F$ da cadeia LNA $\to$ mixer $\to$ IF pela fórmula de Friis, em dB.
  \item Para uma banda de ruído equivalente $B_{\text{neq}} = B_c$ do sinal FM com $\beta = 5$, calcule a potência de ruído na saída da cadeia $N_o = G k T_0 B_{\text{neq}}(F - 1) + G k T_0 B_{\text{neq}}$, com $G = G_1 G_2 G_3$. Em seguida, determine a SNR $\gamma$ na entrada do demodulador.
  \item Aplique a fórmula $(S/N)_o = \tfrac{3}{2}\beta^2 \gamma$ (tom único) e verifique se o sistema cumpre a meta de $(S/N)_o \ge 50\;\text{dB}$ acima do limiar de FM ($\gamma_{\text{th}} = 20(\beta+1)$). Justifique se a meta é atingida com margem ou se é necessário aumentar $P_T$ ou trocar o LNA por um modelo de menor $F_1$.
\end{enumerate}
